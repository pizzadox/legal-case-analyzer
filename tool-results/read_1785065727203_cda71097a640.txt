    36→// Format ISO timestamp in Russian style: dd.MM.yyyy HH:mm
    37→function formatRussianDateTime(iso: string): string {
    38→  try {
    39→    const d = new Date(iso)
    40→    if (isNaN(d.getTime())) return iso
    41→    const dd = String(d.getDate()).padStart(2, '0')
    42→    const mm = String(d.getMonth() + 1).padStart(2, '0')
    43→    const yyyy = d.getFullYear()
    44→    const hh = String(d.getHours()).padStart(2, '0')
    45→    const min = String(d.getMinutes()).padStart(2, '0')
    46→    return `${dd}.${mm}.${yyyy} ${hh}:${min}`
    47→  } catch {
    48→    return iso
    49→  }
    50→}

    55→function loadAllAnnotations(): Record<string, Annotation[]> {
    56→  if (typeof window === 'undefined') return {}
    57→  const loaded: Record<string, Annotation[]> = {}
    58→  try {
    59→    for (let i = 0; i < window.localStorage.length; i++) {
    60→      const key = window.localStorage.key(i)
    61→      if (key && key.startsWith(ANNOTATIONS_PREFIX)) {
    62→        const docId = key.replace(ANNOTATIONS_PREFIX, '')
    63→        const stored = window.localStorage.getItem(key)
    64→        if (stored) {
    65→          try {
    66→            const parsed = JSON.parse(stored)
    67→            if (Array.isArray(parsed)) loaded[docId] = parsed
    68→          } catch {
    69→            /* ignore malformed entries */
    70→          }
    71→        }
    72→      }
    73→    }
    74→  } catch {
    75→    /* localStorage unavailable */
    76→  }
    77→  return loaded
    78→}

   133→// Integrity score ring (small SVG)
   134→function IntegrityRing({ score }: { score: number }) {
   135→  const radius = 18, circ = 2 * Math.PI * radius, offset = circ - (score / 100) * circ
   136→  const color = score >= 75 ? '#059669' : score >= 50 ? '#d97706' : '#dc2626'
   137→  return (
   138→    <div className="relative w-12 h-12 shrink-0">
   139→      <svg width="48" height="48" className="transform -rotate-90"><circle cx="24" cy="24" r={radius} stroke="#e5e7eb" strokeWidth="4" fill="none" className="dark:stroke-stone-700" /><circle cx="24" cy="24" r={radius} stroke={color} strokeWidth="4" fill="none" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="transition-all duration-700" /></svg>
   140→      <div className="absolute inset-0 flex items-center justify-center text-xs font-bold" style={{ color }}>{score}</div>
   141→    </div>

   193→// Export CSV helper (Russian headers, semicolon-separated for Excel compatibility)
   194→function exportDocumentsCSV(docs: DocumentData[]) {
   195→  const headers = ['Название', 'Тип', 'Статус', 'Дата документа', 'Размер (КБ)', 'Дата загрузки', 'Описание']
   196→  const rows = docs.map(d => [
   197→    d.originalName,
   198→    d.documentType ?? '',
   199→    d.processingStatus,
   200→    d.documentDate ?? '',
   201→    Math.round(d.fileSize / 1024),
   202→    new Date(d.uploadedAt).toLocaleDateString('ru-RU'),
   203→    d.summary ?? '',
   204→  ])
   205→  const csv = [headers.join(';'), ...rows.map(r => r.join(';'))].join('\n')
   206→  const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8' })
   207→  const url = URL.createObjectURL(blob)
   208→  const a = document.createElement('a')
   209→  a.href = url
   210→  a.download = `documents_export_${new Date().toISOString().slice(0,10)}.csv`
   211→  a.click()
   212→  URL.revokeObjectURL(url)
   213→  toast.success('CSV экспорт выполнен')
   214→}

   223→// Document comparison dialog
   224→function DocumentCompareDialog({ doc1, doc2, onClose }: { doc1: DocumentData; doc2: DocumentData; onClose: () => void }) {
   225→  const allFields = [
   226→    { label: 'Название', key1: doc1.originalName, key2: doc2.originalName },
   227→    { label: 'Тип', key1: doc1.documentType ?? '', key2: doc2.documentType ?? '' },
   228→    { label: 'Дата документа', key1: doc1.documentDate ?? '', key2: doc2.documentDate ?? '' },
   229→    { label: 'Статус', key1: doc1.processingStatus, key2: doc2.processingStatus },
   230→    { label: 'Размер', key1: fmtSize(doc1.fileSize), key2: fmtSize(doc2.fileSize) },
   231→    { label: 'Источник', key1: doc1.sourceReference ?? '', key2: doc2.sourceReference ?? '' },
   232→  ]
   233→  const fields = allFields.filter(f => hasValue(f.key1) || hasValue(f.key2))
   234→  const renderSide = (doc: DocumentData, label: string, getValue: (f: typeof fields[number]) => string) => (
   235→    <div className="space-y-3">
   236→      <div className="p-3 rounded-lg bg-muted"><p className="font-medium text-xs mb-1">{label}</p>{doc.documentType && <Badge className={TYPE_BADGE[doc.documentType] ?? 'bg-stone-500 text-white'}>{doc.documentType}</Badge>}</div>
   237→      {fields.map(f => (
   238→        <div key={f.label} className={`p-2 rounded-lg ${f.key1 !== f.key2 ? 'bg-red-100 dark:bg-red-900/30 border border-red-300 dark:border-red-700' : 'bg-muted/50'}`}>
   239→          <p className="text-xs font-medium text-muted-foreground">{f.label}</p><p className="text-xs font-semibold">{getValue(f)}</p>
   240→        </div>
   241→      ))}
   242→      {doc.summary && <div className="p-2 rounded-lg bg-muted"><p className="text-xs font-medium text-muted-foreground">Краткое содержание</p><p className="text-xs">{doc.summary}</p></div>}
   243→    </div>
   244→  )
   245→  return (
   246→    <Dialog open={true} onOpenChange={onClose}>
   247→      <DialogContent className="max-w-3xl max-h-[80vh] overflow-y-auto rounded-xl">
   248→        <DialogHeader><DialogTitle className="text-sm flex items-center gap-2"><GitCompare className="w-4 h-4" /> Сравнение документов</DialogTitle></DialogHeader>
   249→        <div className="grid grid-cols-2 gap-4 text-sm">
   250→          {renderSide(doc1, 'Документ A', f => f.key1)}
   251→          {renderSide(doc2, 'Документ B', f => f.key2)}
   252→        </div>
   253→        <Separator className="mt-3" />
   254→        <p className="text-xs text-muted-foreground">Отличия выделены красным фоном</p>
   255→      </DialogContent>
   256→    </Dialog>

   815→                    <p className="text-xs whitespace-pre-wrap leading-relaxed">
   816→                      {docSearch.trim()
   817→                        ? highlightText(selectedDoc.extractedText, docSearch, docSearchCaseSensitive)
   818→                        : selectedDoc.extractedText}
   819→                    </p>
   820→                  </ScrollArea>
   821→                </CardContent>
   822→              </Card>

   824→

   832→                  </Badge>

   838→                    <div className="flex items-center justify-center w-12 h-12 rounded-full bg-amber-500/10 mx-auto mb-2 ring-4 ring-amber-500/5">
   839→                      <MessageSquare className="w-6 h-6 text-amber-600/70" />
   840→                    </div>
   841→                    <p className="text-xs text-muted-foreground">Пока нет комментариев. Добавьте первый.</p>

   844→                  <div className="space-y-2">
   845→                    {selectedDoc && (annotations[selectedDoc.id] ?? []).map(an => (

   854→                                <User className="w-2.5 h-2.5 text-muted-foreground shrink-0" />
   855→                                {an.author}
   856→                              </p>

   859→                                onClick={() => handleDeleteAnnotation(selectedDoc.id, an.id)}

   868→                              <Calendar className="w-2.5 h-2.5" />
   869→                              {formatRussianDateTime(an.timestamp)}
   870→                            </p>
   871→                          </div>
   872→                        </div>
   873→                      </div>
   874→                    ))}
   875→                  </div>
   876→                )}
   877→
   878→                {/* Add annotation form */}
   879→                <div className="space-y-2 pt-3 border-t">
   880→                  <Textarea

   886→                  />
   887→                  <Button
   888→                    size="sm"
   889→                    className="w-full rounded-xl bg-gradient-to-r from-amber-600 to-amber-700 text-white hover:from-amber-700 hover:to-amber-800"
   890→                    disabled={!newAnnotation.trim()}
   891→                    onClick={() => selectedDoc && handleAddAnnotation(selectedDoc.id)}
   892→                  >
   893→                    <Plus className="w-3 h-3 mr-1" /> Добавить комментарий
   894→                  </Button>
   895→                </div>
   896→              </CardContent>
   897→            </Card>
   898→
   899→            {/* Actions */}
   900→            <Card className="rounded-xl shadow-sm">
   901→              <CardHeader className="pb-2"><CardTitle className="text-sm">Действия</CardTitle></CardHeader>
   902→              <CardContent className="flex flex-wrap gap-2">
   903→                <Button
   904→                  size="sm"
   905→                  variant="outline"
   906→                  className="rounded-xl"
   907→                  onClick={() => shadcnToast({ title: 'Экспорт', description: 'Подготовка экспорта документа...' })}
   908→                >
   909→                  <Download className="w-3 h-3 mr-1" /> Экспорт

   912→                  size="sm"
   913→                  variant="outline"
   914→                  className="rounded-xl"
   915→                  onClick={() => {
   916→                    shadcnToast({ title: 'Переобработка', description: 'Запущена повторная обработка документа.' })

   919→                >
   920→                  <RefreshCw className="w-3 h-3 mr-1" /> Переобработать

   922→                <Button
   923→                  size="sm"
   924→                  variant="outline"

   926→                  onClick={() => {
   927→                    if (selectedDoc) {

   929→                      setSelectedDoc(null)
   930→                      setNewAnnotation('')

   932→                  }}
   933→                >
   934→                  <Trash2 className="w-3 h-3 mr-1" /> Удалить
   935→                </Button>
   936→              </CardContent>
   937→            </Card>
   938→          </div>
   939→        </SheetContent>
   940→      </Sheet>
   941→
   942→      {/* Document Comparison Dialog */}

   944→    </div>
