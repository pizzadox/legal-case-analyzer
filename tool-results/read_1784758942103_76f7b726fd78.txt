   682→function EvidenceLinkPath({

  1030→export function CaseEvidenceChain() {

  1039→    () => new Set(EVIDENCE_LINKS.filter((l) => l.strength === 'strong').map((l) => l.id)),
  1040→    [],
  1041→  )
  1042→  // Все ID узлов, имеющих хотя бы одну сильную связь
  1043→  const strongConnectedIds = useMemo(() => {
  1044→    const ids = new Set<string>()
  1045→    EVIDENCE_LINKS.filter((l) => l.strength === 'strong').forEach((l) => {
  1046→      ids.add(l.sourceId)
  1047→      ids.add(l.targetId)
  1048→    })
  1049→    return ids
  1050→  }, [])
  1051→
  1052→  // Список узлов с учётом фильтра
  1053→  const visibleItems = useMemo(() => {
  1054→    if (filter === 'all') return ALL_EVIDENCE
  1055→    if (filter === 'prosecution') return PROSECUTION_EVIDENCE
  1056→    if (filter === 'defense') return DEFENSE_EVIDENCE
  1057→    if (filter === 'strong') return ALL_EVIDENCE.filter((e) => strongConnectedIds.has(e.id))
  1058→    return ALL_EVIDENCE
  1059→  }, [filter, strongConnectedIds])
  1060→
  1061→  const visibleIds = useMemo(() => new Set(visibleItems.map((i) => i.id)), [visibleItems])
  1062→
  1063→  // Связи, у которых оба конца видимы
  1064→  const visibleLinks = useMemo(() => {
  1065→    return EVIDENCE_LINKS.filter(
  1066→      (l) => visibleIds.has(l.sourceId) && visibleIds.has(l.targetId),
  1067→    )
  1068→  }, [visibleIds])
  1069→
  1070→  // Множество ID узлов и связей, связанных с наведённым элементом
  1071→  const activeIds = useMemo(() => {
  1072→    if (hoveredId) {
  1073→      const linked = new Set<string>([hoveredId])
  1074→      EVIDENCE_LINKS.forEach((l) => {
  1075→        if (l.sourceId === hoveredId) linked.add(l.targetId)
  1076→        if (l.targetId === hoveredId) linked.add(l.sourceId)
  1077→      })
  1078→      return { nodes: linked, links: new Set<string>() }
