    40→function useAnimatedCounter(target: number, ms = 700) {
    41→  const [val, setVal] = useState(0), raf = useRef<number | null>(null)
    42→  useEffect(() => {
    43→    const start = performance.now(), delta = target - val
    44→    if (!delta) return
    45→    const tick = (now: number) => { const t = Math.min(1, (now - start) / ms), eased = 1 - Math.pow(1 - t, 3); setVal(Math.round(val + delta * eased)); if (t < 1) raf.current = requestAnimationFrame(tick) }
    46→    raf.current = requestAnimationFrame(tick)
    47→    return () => { if (raf.current) cancelAnimationFrame(raf.current) }
    48→  }, [target, ms])
    49→  return val
    50→}

    52→function HealthGauge({ score }: { score: number }) {
    53→  const anim = useAnimatedCounter(score, 900), cx = 80, cy = 80, oR = 70, iR = 50, startA = -210, sweep = 240
    54→  const col = score >= 60 ? '#059669' : score >= 40 ? '#d97706' : '#dc2626'
    55→  const segs = [{ s: 0, e: 40, c: '#dc2626' }, { s: 40, e: 60, c: '#d97706' }, { s: 60, e: 100, c: '#059669' }]
    56→  const arc = (s: number, e: number) => { const sa = startA + (s / 100) * sweep, ea = startA + (e / 100) * sweep, sR = (sa * Math.PI) / 180, eR = (ea * Math.PI) / 180, la = ea - sa > 180 ? 1 : 0
    57→    const sx = cx + oR * Math.cos(sR), sy = cy + oR * Math.sin(sR), ex = cx + oR * Math.cos(eR), ey = cy + oR * Math.sin(eR), six = cx + iR * Math.cos(sR), siy = cy + iR * Math.sin(sR), eix = cx + iR * Math.cos(eR), eiy = cy + iR * Math.sin(eR)
    58→    return `M ${sx} ${sy} A ${oR} ${oR} 0 ${la} 1 ${ex} ${ey} L ${eix} ${eiy} A ${iR} ${iR} 0 ${la} 0 ${six} ${siy} Z` }
    59→  const nA = startA + (anim / 100) * sweep, nR = (nA * Math.PI) / 180, nX = cx + 42 * Math.cos(nR), nY = cy + 42 * Math.sin(nR)
    60→  const bgSR = (startA * Math.PI) / 180, bgER = ((startA + sweep) * Math.PI) / 180, la = sweep > 180 ? 1 : 0
    61→  const bgSX = cx + oR * Math.cos(bgSR), bgSY = cy + oR * Math.sin(bgSR), bgEX = cx + oR * Math.cos(bgER), bgEY = cy + oR * Math.sin(bgER), bISX = cx + iR * Math.cos(bgSR), bISY = cy + iR * Math.sin(bgSR), bIEX = cx + iR * Math.cos(bgER), bIEY = cy + iR * Math.sin(bgER)
    62→  return (
    63→    <div className={`flex items-center justify-center ${score < 40 ? 'animate-pulse' : ''}`}>
    64→      <svg width="160" height="120" viewBox="0 0 160 120">
    65→        <path d={`M ${bgSX} ${bgSY} A ${oR} ${oR} 0 ${la} 1 ${bgEX} ${bgEY} L ${bIEX} ${bIEY} A ${iR} ${iR} 0 ${la} 0 ${bISX} ${bISY} Z`} fill="#e7e5e4" className="dark:fill-stone-700" />
    66→        {segs.map(s => <path key={s.c} d={arc(s.s, s.e)} fill={s.c} opacity={0.25} />)}
    67→        <path d={arc(0, anim)} fill={col} opacity={0.8} className="transition-all duration-700" />
