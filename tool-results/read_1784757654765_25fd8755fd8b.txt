   123→function strengthColor(s: number): string {
   124→  if (s >= 70) return '#b91c1c' // strong — red-700 (prosecution) или emerald для defense (определяется вызывающим)
   125→  if (s >= 40) return '#d97706' // moderate — amber-600
   126→  return '#78716c' // weak — stone-500
   127→}
   128→
   129→function strengthColorForSide(s: number, side: EvidenceSide): string {
   130→  if (s >= 70) return side === 'prosecution' ? '#b91c1c' : '#047857'
   131→  if (s >= 40) return '#d97706'
   132→  return '#78716c'
   133→}
   134→
   135→function strengthLabel(s: number): string {
   136→  if (s >= 70) return 'Сильное'
   137→  if (s >= 40) return 'Умеренное'
   138→  return 'Слабое'
   139→}
   140→
   141→function strengthBadge(s: number): string {
   142→  if (s >= 70) return 'bg-emerald-700 text-white'
   143→  if (s >= 40) return 'bg-amber-600 text-white'
   144→  return 'bg-stone-500 text-white'
   145→}
   146→
   147→function strengthBadgeForSide(s: number, side: EvidenceSide): string {
   148→  if (s >= 70) return side === 'prosecution' ? 'bg-red-700 text-white' : 'bg-emerald-700 text-white'
   149→  if (s >= 40) return 'bg-amber-600 text-white'
   150→  return 'bg-stone-500 text-white'
   151→}

   470→const DEFENSE_Y = 380 // центр нижнего узла (немного поднят, чтобы не перекрывать нижний заголовок)

   479→// Рассчитываем x-координату по дате
   480→const dateToX = (dateStr: string): number => {
   481→  const dates = ALL_EVIDENCE.map((e) => new Date(e.date).getTime())
   482→  const min = Math.min(...dates)
   483→  const max = Math.max(...dates)
   484→  const t = (new Date(dateStr).getTime() - min) / (max - min || 1)
   485→  return PADDING_X + t * (SVG_WIDTH - 2 * PADDING_X)
   486→}
   487→
   488→// Предварительно посчитанные позиции узлов (с лёгким «расталкиванием»)
   489→const nodePositions: Record<string, { x: number; y: number }> = (() => {
   490→  const pos: Record<string, { x: number; y: number }> = {}
   491→  // Prosecution: сортируем по дате и расталкиваем
   492→  const pros = [...PROSECUTION_EVIDENCE].sort(
   493→    (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime(),
   494→  )
   495→  const def = [...DEFENSE_EVIDENCE].sort(
   496→    (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime(),
   497→  )
   498→
   499→  const minSpacing = NODE_WIDTH + 24
   500→  const applyDodge = (arr: EvidenceItem[], y: number) => {
   501→    const coords = arr.map((e) => ({ id: e.id, x: dateToX(e.date) }))
   502→    for (let i = 1; i < coords.length; i++) {
   503→      if (coords[i].x - coords[i - 1].x < minSpacing) {
   504→        coords[i].x = coords[i - 1].x + minSpacing
   505→      }
   506→    }
   507→    // Не даём выйти за правую границу
   508→    const maxX = SVG_WIDTH - PADDING_X
   509→    for (let i = coords.length - 1; i >= 0; i--) {
   510→      if (coords[i].x > maxX) coords[i].x = maxX
   511→    }
   512→    coords.forEach((c) => {
   513→      pos[c.id] = { x: c.x, y }
   514→    })
   515→  }
   516→  applyDodge(pros, PROSECUTION_Y)
   517→  applyDodge(def, DEFENSE_Y)
   518→  return pos
   519→})()

   671→interface EvidenceLinkPathProps {
   672→  link: EvidenceLink
   673→  sourceX: number
   674→  sourceY: number
   675→  targetX: number
   676→  targetY: number
   677→  isDimmed: boolean
   678→  isHighlighted: boolean
   679→  onHover: (id: string | null) => void
   680→}
   681→
   682→function EvidenceLinkPath({
   683→  link,
   684→  sourceX,
   685→  sourceY,
   686→  targetX,
   687→  targetY,
   688→  isDimmed,
   689→  isHighlighted,
   690→  onHover,
   691→}: EvidenceLinkPathProps) {
   692→  const typeCfg = LINK_TYPE_CONFIG[link.type]
   693→  const strCfg = LINK_STRENGTH_CONFIG[link.strength]
   694→
   695→  // Источник сверху (обвинение), цель снизу (защита) — всегда так по построению
   696→  const x1 = sourceX
   697→  const y1 = sourceY + NODE_HEIGHT / 2
   698→  const x2 = targetX
   699→  const y2 = targetY - NODE_HEIGHT / 2
   700→  // Контрольные точки для плавной Bezier-кривой
   701→  const midY = (y1 + y2) / 2
   702→  const path = `M ${x1} ${y1} C ${x1} ${midY}, ${x2} ${midY}, ${x2} ${y2}`
   703→
   704→  const opacity = isDimmed ? 0.12 : isHighlighted ? 1 : strCfg.opacity
   705→  const width = isHighlighted ? strCfg.width + 1.2 : strCfg.width
   706→
   707→  return (
   708→    <g
   709→      onMouseEnter={() => onHover(link.id)}
   710→      onMouseLeave={() => onHover(null)}
   711→      className="cursor-pointer"
   712→    >
   713→      {/* Широкая невидимая дорожка для удобного наведения */}
   714→      <path d={path} fill="none" stroke="transparent" strokeWidth={14} />
   715→      {/* Видимая линия */}
   716→      <path
   717→        d={path}
   718→        fill="none"
   719→        stroke={typeCfg.color}
   720→        strokeWidth={width}
   721→        strokeOpacity={opacity}
   722→        strokeDasharray={typeCfg.dashed ? '5 4' : undefined}
   723→        strokeLinecap="round"
   724→        style={{ transition: 'stroke-opacity 200ms ease, stroke-width 200ms ease' }}
   725→      />
   726→      {/* Маркер в средней точке связи */}
   727→      <circle
   728→        cx={(x1 + x2) / 2}
   729→        cy={midY}
   730→        r={isHighlighted ? 4.5 : 3}
   731→        fill={typeCfg.color}
   732→        fillOpacity={opacity}
   733→        stroke="#ffffff"
   734→        strokeWidth={1}
   735→        style={{ transition: 'r 200ms ease, fill-opacity 200ms ease' }}
   736→      />
   737→    </g>
   738→  )
   739→}
