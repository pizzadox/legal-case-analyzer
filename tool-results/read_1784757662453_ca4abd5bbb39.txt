    38→// Format ISO timestamp in Russian style: dd.MM.yyyy HH:mm
    39→function formatRussianDateTime(iso: string): string {
    40→  try {
    41→    const d = new Date(iso)
    42→    if (isNaN(d.getTime())) return iso
    43→    const dd = String(d.getDate()).padStart(2, '0')
    44→    const mm = String(d.getMonth() + 1).padStart(2, '0')
    45→    const yyyy = d.getFullYear()
    46→    const hh = String(d.getHours()).padStart(2, '0')
    47→    const min = String(d.getMinutes()).padStart(2, '0')
    48→    return `${dd}.${mm}.${yyyy} ${hh}:${min}`
    49→  } catch {
    50→    return iso
    51→  }
    52→}

    57→function loadAllAnnotations(): Record<string, Annotation[]> {
    58→  if (typeof window === 'undefined') return {}
    59→  const loaded: Record<string, Annotation[]> = {}
    60→  try {
    61→    for (let i = 0; i < window.localStorage.length; i++) {
    62→      const key = window.localStorage.key(i)
    63→      if (key && key.startsWith(ANNOTATIONS_PREFIX)) {
    64→        const docId = key.replace(ANNOTATIONS_PREFIX, '')
    65→        const stored = window.localStorage.getItem(key)
    66→        if (stored) {
    67→          try {
    68→            const parsed = JSON.parse(stored)
    69→            if (Array.isArray(parsed)) loaded[docId] = parsed
    70→          } catch {
    71→            /* ignore malformed entries */
    72→          }
    73→        }
    74→      }
    75→    }
    76→  } catch {
    77→    /* localStorage unavailable */
    78→  }
    79→  return loaded
    80→}

    91→function fmtSize(b: number) {
    92→  if (b < 1024) return b + ' Б'
    93→  if (b < 1048576) return (b / 1024).toFixed(1) + ' КБ'
    94→  return (b / 1048576).toFixed(1) + ' МБ'
    95→}

   136→function IntegrityRing({ score }: { score: number }) {
   137→  const radius = 18, circ = 2 * Math.PI * radius, offset = circ - (score / 100) * circ
   138→  const color = score >= 75 ? '#059669' : score >= 50 ? '#d97706' : '#dc2626'
   139→  return (
   140→    <div className="relative w-12 h-12 shrink-0">
   141→      <svg width="48" height="48" className="transform -rotate-90"><circle cx="24" cy="24" r={radius} stroke="#e5e7eb" strokeWidth="4" fill="none" className="dark:stroke-stone-700" /><circle cx="24" cy="24" r={radius} stroke={color} strokeWidth="4" fill="none" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="transition-all duration-700" /></svg>
   142→      <div className="absolute inset-0 flex items-center justify-center text-xs font-bold" style={{ color }}>{score}</div>
   143→    </div>

   195→// Export CSV helper
   196→function exportDocumentsCSV(docs: DocumentData[]) {
   197→  const rows = ['Name,Type,Date,Status,Size,Summary']
   198→  docs.forEach(d => {
   199→    rows.push(`"${d.originalName}",${d.documentType ?? ''},${d.documentDate ?? ''},${d.processingStatus},${fmtSize(d.fileSize)},${d.summary ?? ''}`)
   200→  })
   201→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
   202→  const url = URL.createObjectURL(blob)
   203→  const a = document.createElement('a')
   204→  a.href = url
   205→  a.download = 'documents.csv'
   206→  a.click()
   207→  URL.revokeObjectURL(url)
   208→  toast.success('CSV экспорт выполнен')
   209→}
