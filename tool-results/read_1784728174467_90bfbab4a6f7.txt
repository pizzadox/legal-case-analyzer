    28→// Highlight search terms in text
    29→function highlightText(text: string, query: string): React.ReactNode {
    30→  if (!query.trim()) return text
    31→  const regex = new RegExp(`(${query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi')
    32→  const parts = text.split(regex)
    33→  return parts.map((part, i) =>
    34→    regex.test(part) ? (
    35→      <mark key={i} className="bg-orange-500/30 text-orange-200 rounded px-0.5">{part}</mark>
    36→    ) : part
    37→  )
    38→}
