    60→function formatFileSize(bytes: number): string {
    61→  if (bytes < 1024) return bytes + ' Б'
    62→  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' КБ'
    63→  return (bytes / (1024 * 1024)).toFixed(1) + ' МБ'
    64→}

   105→// Animated counter component
   106→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   107→  const [displayValue, setDisplayValue] = useState(0)
   108→
   109→  useEffect(() => {
   110→    let startTime: number | null = null
   111→    let animationFrame: number
   112→
   113→    const animate = (timestamp: number) => {
   114→      if (!startTime) startTime = timestamp
   115→      const progress = Math.min((timestamp - startTime) / duration, 1)
   116→      const eased = 1 - Math.pow(1 - progress, 3)
   117→      setDisplayValue(Math.floor(eased * value))
   118→      if (progress < 1) {
   119→        animationFrame = requestAnimationFrame(animate)
   120→      }
   121→    }
   122→
   123→    animationFrame = requestAnimationFrame(animate)
   124→    return () => cancelAnimationFrame(animationFrame)
   125→  }, [value, duration])
   126→
   127→  return <span>{displayValue}</span>
   128→}

   130→// Case health score calculation
   131→function calculateHealthScore(stats: DashboardStats | null): number {
   132→  if (!stats) return 65
   133→  const processedRatio = stats.totalDocuments > 0 ? stats.processedDocuments / stats.totalDocuments : 0
   134→  const guiltBalance = stats.guiltDistribution.reduce((acc, item) => {
   135→    const weight: Record<string, number> = { 'Высокая': 0.7, 'Средняя': 0.5, 'Низкая': 0.3, 'Нет': 0.9 }
   136→    return acc + (weight[item.level] || 0.5) * item.count
   137→  }, 0) / (stats.guiltDistribution.reduce((acc, item) => acc + item.count, 0) || 1)
   138→  const queueHealth = stats.totalDocuments > 0 ? 1 - (stats.pendingInQueue / stats.totalDocuments) : 1
   139→  return Math.round(processedRatio * 30 + guiltBalance * 40 + queueHealth * 30)
   140→}
