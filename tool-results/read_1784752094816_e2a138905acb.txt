   173→   173→   173→function NotificationCenter({ notifications, setActiveSection }: {
   174→   174→   174→  notifications: NotificationData[]
   175→   175→   175→  setActiveSection: (section: SectionId) => void
   176→   176→   176→}) {
   177→   177→   177→  const [localNotifications, setLocalNotifications] = useState(notifications)
   178→   178→   178→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   179→   179→   179→
   180→   180→   180→  const markAllRead = () => {
   181→   181→   181→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   182→   182→   182→    toast.success('Все уведомления отмечены как прочитанные')
   183→   183→   183→  }
   184→   184→   184→
   185→   185→   185→  const handleClick = (notif: NotificationData) => {
   186→   186→   186→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   187→   187→   187→    if (notif.relatedSection) {
   188→   188→   188→      setActiveSection(notif.relatedSection)
   189→   189→   189→    }
   190→   190→   190→  }
   191→   191→   191→
   192→   192→   192→  return (
   193→   193→   193→    <Popover>
   194→   194→   194→      <PopoverTrigger asChild>
   195→   195→   195→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   196→   196→   196→          <Bell className="h-4 w-4" />
   197→   197→   197→          {unreadCount > 0 && (
   198→   198→   198→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   199→   199→   199→              {unreadCount}
   200→   200→   200→            </span>
   201→   201→   201→          )}
   202→   202→   202→          <span className="sr-only">Уведомления</span>
   203→   203→   203→        </Button>
   204→   204→   204→      </PopoverTrigger>
   205→   205→   205→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   206→   206→   206→        <div className="p-3 border-b flex items-center gap-2">
   207→   207→   207→          <Bell className="w-4 h-4 text-muted-foreground" />
   208→   208→   208→          <p className="font-semibold text-sm">Уведомления</p>
   209→   209→   209→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   210→   210→   210→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   211→   211→   211→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   212→   212→   212→          </Button>
   213→   213→   213→        </div>
   214→   214→   214→        <div className="max-h-64 overflow-y-auto">
   215→   215→   215→          {localNotifications.map(notif => (
   216→   216→   216→            <div key={notif.id}
   217→   217→   217→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   218→   218→   218→              onClick={() => handleClick(notif)}>
   219→   219→   219→              <div className="flex items-center gap-2">
   220→   220→   220→                {NOTIF_TYPE_ICON[notif.type] ?? <Bell className="w-4 h-4" />}

   226→   226→   226→              </div>
