   173→function NotificationCenter({ notifications, setActiveSection }: {
   174→  notifications: NotificationData[]
   175→  setActiveSection: (section: SectionId) => void
   176→}) {
   177→  const [localNotifications, setLocalNotifications] = useState(notifications)
   178→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   179→
   180→  const markAllRead = () => {
   181→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   182→    toast.success('Все уведомления отмечены как прочитанные')
   183→  }
   184→
   185→  const handleClick = (notif: NotificationData) => {
   186→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   187→    if (notif.relatedSection) {
   188→      setActiveSection(notif.relatedSection)
   189→    }
   190→  }
   191→
   192→  return (
   193→    <Popover>
   194→      <PopoverTrigger asChild>
   195→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   196→          <Bell className="h-4 w-4" />
   197→          {unreadCount > 0 && (
   198→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   199→              {unreadCount}
   200→            </span>
   201→          )}
   202→          <span className="sr-only">Уведомления</span>
   203→        </Button>
   204→      </PopoverTrigger>
   205→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   206→        <div className="p-3 border-b flex items-center gap-2">
   207→          <Bell className="w-4 h-4 text-muted-foreground" />
   208→          <p className="font-semibold text-sm">Уведомления</p>
   209→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   210→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   211→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   212→          </Button>
   213→        </div>
   214→        <div className="max-h-64 overflow-y-auto">
   215→          {localNotifications.map(notif => (
   216→            <div key={notif.id}
   217→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   218→              onClick={() => handleClick(notif)}>
   219→              <div className="flex items-center gap-2">
   220→                {NOTIF_TYPE_ICON[notif.type] ?? <Bell className="w-4 h-4" />}
   221→                <div className="flex-1 min-w-0">
   222→                  <p className="text-sm font-medium truncate">{notif.title}</p>
   223→                  <p className="text-xs text-muted-foreground truncate">{notif.description}</p>
   224→                </div>
   225→                {!notif.isRead && <div className="w-2 h-2 rounded-full bg-red-700 shrink-0" />}
   226→              </div>
   227→              <p className="text-xs text-muted-foreground mt-1">{new Date(notif.timestamp).toLocaleString('ru-RU', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' })}</p>
   228→            </div>
   229→          ))}
   230→          {localNotifications.length === 0 && (
   231→            <div className="p-6 text-center">
   232→              <Bell className="w-8 h-8 mx-auto text-muted-foreground" />
   233→              <p className="text-sm text-muted-foreground mt-2">Нет уведомлений</p>
   234→            </div>
   235→          )}
   236→        </div>
   237→      </PopoverContent>
   238→    </Popover>
   239→  )
   240→}

   346→function CaseHealthBadge() {
   347→  // Compact health indicator for sidebar footer
   348→  const [score, setScore] = useState<number | null>(null)
   349→  useEffect(() => {
   350→    fetch('/api/case/health-score')
   351→      .then(r => r.ok ? r.json() : null)
   352→      .then(d => {
   353→        if (d && typeof d.score === 'number') setScore(d.score)
   354→        else if (d && typeof d.overallScore === 'number') setScore(d.overallScore)
   355→      })
   356→      .catch(() => setScore(null))
   357→  }, [])
   358→
   359→  if (score === null) {
   360→    return (
   361→      <div className="flex items-center gap-2 px-2 py-1.5 rounded-md bg-muted/50 text-xs text-muted-foreground">
   362→        <Activity className="w-3.5 h-3.5 animate-pulse" />
   363→        <span className="group-data-[collapsible=icon]:hidden">Загрузка...</span>
   364→      </div>
   365→    )
   366→  }
   367→
   368→  // Color thresholds: <50 red, 50-75 amber, >75 emerald
   369→  const color = score >= 75 ? 'emerald' : score >= 50 ? 'amber' : 'red'
   370→  const colorClasses = {
   371→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900',
   372→    amber: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900',
   373→    red: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900',
   374→  }
   375→  const statusText = score >= 75 ? 'Здоровое' : score >= 50 ? 'Среднее' : 'Проблемное'
   376→
   377→  return (
   378→    <div className={`flex items-center gap-2 px-2 py-1.5 rounded-md border text-xs ${colorClasses[color]}`}>
   379→      <Heart className="w-3.5 h-3.5 fill-current" />
   380→      <div className="flex-1 min-w-0 group-data-[collapsible=icon]:hidden">
   381→        <div className="flex items-center gap-1.5">
   382→          <span className="font-semibold">{score}/100</span>
   383→          <span className="opacity-70">·</span>
   384→          <span className="truncate">{statusText}</span>
   385→        </div>
   386→        <div className="mt-1 h-1 rounded-full bg-black/10 dark:bg-white/10 overflow-hidden">
   387→          <div
   388→            className={`h-full rounded-full transition-all duration-700 ${
   389→              color === 'emerald' ? 'bg-emerald-600' : color === 'amber' ? 'bg-amber-600' : 'bg-red-700'
   390→            }`}
   391→            style={{ width: `${score}%` }}
   392→          />
   393→        </div>
   394→      </div>
   395→    </div>
   396→  )
   397→}

   727→function loadPreferences(): UserPreferences {
   728→  if (typeof window === 'undefined') return DEFAULT_PREFERENCES
   729→  try {
   730→    const raw = localStorage.getItem('case-user-preferences')
   731→    if (!raw) return DEFAULT_PREFERENCES
   732→    const parsed = JSON.parse(raw)
   733→    return { ...DEFAULT_PREFERENCES, ...parsed }
   734→  } catch {
   735→    return DEFAULT_PREFERENCES
   736→  }
   737→}

   739→function savePreferences(prefs: UserPreferences) {
   740→  if (typeof window === 'undefined') return
   741→  localStorage.setItem('case-user-preferences', JSON.stringify(prefs))
   742→}

   744→function applyPreferencesToBody(prefs: UserPreferences) {
   745→  if (typeof document === 'undefined') return
   746→  const body = document.body
   747→  // Density
   748→  body.classList.toggle('compact', prefs.density === 'compact')
   749→  // Font size
   750→  body.classList.remove('text-size-sm', 'text-size-md', 'text-size-lg')
   751→  body.classList.add(`text-size-${prefs.fontSize}`)
   752→  // Animations
   753→  body.classList.toggle('no-animations', !prefs.animations)
   754→}

  1001→export default function Home() {
  1002→  // Lazy-load preferences from localStorage on first render (avoids setState-in-effect)
  1003→  const [preferences, setPreferences] = useState<UserPreferences>(() => {
  1004→    if (typeof window === 'undefined') return DEFAULT_PREFERENCES
  1005→    const loaded = loadPreferences()
  1006→    // Apply to body immediately on first client render
  1007→    if (typeof document !== 'undefined') {
  1008→      applyPreferencesToBody(loaded)
  1009→    }
  1010→    return loaded
  1011→  })
  1012→  const [activeSection, setActiveSection] = useState<SectionId>(() => {
  1013→    if (typeof window === 'undefined') return 'dashboard'
  1014→    const loaded = loadPreferences()
  1015→    return loaded.defaultSection ?? 'dashboard'
  1016→  })
  1017→  const [helpOpen, setHelpOpen] = useState(false)
  1018→  const [commandOpen, setCommandOpen] = useState(false)
  1019→  const [settingsOpen, setSettingsOpen] = useState(false)
  1020→  const notifications = mockNotifications
  1021→
  1022→  // Auto-refresh handler
  1023→  useEffect(() => {
  1024→    if (!preferences.autoRefresh) return
  1025→    const interval = setInterval(() => {
  1026→      // Trigger a refetch by dispatching a custom event
  1027→      window.dispatchEvent(new CustomEvent('case-auto-refresh'))
  1028→    }, 30000) // 30 seconds
  1029→    return () => clearInterval(interval)
  1030→  }, [preferences.autoRefresh])
  1031→
  1032→  // Keyboard shortcuts handler
  1033→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
  1034→    // Ctrl+K opens command palette (highest priority)
  1035→    if (e.ctrlKey && (e.key === 'k' || e.key === 'K' || e.key === 'л' || e.key === 'Л')) {
  1036→      e.preventDefault()
  1037→      setCommandOpen(prev => !prev)
  1038→      return
  1039→    }
  1040→    // Ctrl+, (comma) opens settings
  1041→    if (e.ctrlKey && (e.key === ',' || e.key === 'б' || e.key === 'Б')) {
  1042→      e.preventDefault()
  1043→      setSettingsOpen(true)
  1044→      return
  1045→    }
  1046→    // Ctrl+1-8 for section navigation
  1047→    if (e.ctrlKey && !e.altKey && !e.shiftKey) {
  1048→      const num = parseInt(e.key)
  1049→      if (num >= 1 && num <= 8) {
  1050→        e.preventDefault()
  1051→        setActiveSection(NAV_ITEMS[num - 1].id)
  1052→      }
  1053→      // Ctrl+9 for timeline
  1054→      if (e.key === '9') {
  1055→        e.preventDefault()
  1056→        setActiveSection('timeline')
  1057→      }
  1058→      // Ctrl+0 for risk
  1059→      if (e.key === '0') {
  1060→        e.preventDefault()
  1061→        setActiveSection('risk')
  1062→      }
  1063→      // Ctrl+B for brief
  1064→      if (e.key === 'b' || e.key === 'B' || e.key === 'и' || e.key === 'И') {
  1065→        e.preventDefault()
  1066→        setActiveSection('brief')
  1067→      }
  1068→      // Ctrl+A for analytics
  1069→      if (e.key === 'a' || e.key === 'A' || e.key === 'ф' || e.key === 'Ф') {
  1070→        e.preventDefault()
  1071→        setActiveSection('analytics')
  1072→      }
  1073→      // Ctrl+E for evidence chain
  1074→      if (e.key === 'e' || e.key === 'E' || e.key === 'у' || e.key === 'У') {
  1075→        e.preventDefault()
  1076→        setActiveSection('evidence-chain')
  1077→      }
  1078→      // Ctrl+M for witness matrix
  1079→      if (e.key === 'm' || e.key === 'M' || e.key === 'ь' || e.key === 'Ь') {
  1080→        e.preventDefault()
  1081→        setActiveSection('witness-matrix')
  1082→      }
  1083→    }
  1084→    // ? for help
  1085→    if (e.key === '?' && !e.ctrlKey && !e.altKey) {
  1086→      // Only trigger when not in an input/textarea
  1087→      const target = e.target as HTMLElement
  1088→      if (target.tagName !== 'INPUT' && target.tagName !== 'TEXTAREA') {
  1089→        e.preventDefault()
  1090→        setHelpOpen(true)
  1091→      }
  1092→    }
  1093→    // Escape to close help or command palette
  1094→    if (e.key === 'Escape') {
  1095→      if (helpOpen) setHelpOpen(false)
  1096→      if (commandOpen) setCommandOpen(false)
  1097→      if (settingsOpen) setSettingsOpen(false)
  1098→    }
  1099→  }, [helpOpen, commandOpen, settingsOpen])
  1100→
  1101→  useEffect(() => {
  1102→    window.addEventListener('keydown', handleKeyDown)
  1103→    return () => window.removeEventListener('keydown', handleKeyDown)
  1104→  }, [handleKeyDown])
  1105→
  1106→  return (
  1107→    <SidebarProvider>
  1108→      <AppSidebar activeSection={activeSection} setActiveSection={setActiveSection} preferences={preferences} />
  1109→      <SidebarInset>
  1110→        <TopHeader
  1111→          activeSection={activeSection}
  1112→          notifications={notifications}
  1113→          setActiveSection={setActiveSection}
  1114→          onHelpClick={() => setHelpOpen(true)}
  1115→          onCommandClick={() => setCommandOpen(true)}
  1116→          onSettingsClick={() => setSettingsOpen(true)}
  1117→          preferences={preferences}
  1118→        />
  1119→        <div className="flex flex-1 flex-col min-h-0">
  1120→          <main className="flex-1 overflow-y-auto p-4 md:p-6">
  1121→            <MainContent activeSection={activeSection} />
  1122→          </main>
  1123→          <AppFooter />
  1124→        </div>
  1125→      </SidebarInset>
  1126→      <KeyboardShortcutsHelp open={helpOpen} onClose={() => setHelpOpen(false)} />
  1127→      <CommandPalette
  1128→        open={commandOpen}
  1129→        onOpenChange={setCommandOpen}
  1130→        setActiveSection={setActiveSection}
  1131→      />
  1132→      <SettingsDialog
  1133→        open={settingsOpen}
  1134→        onClose={() => setSettingsOpen(false)}
  1135→        preferences={preferences}
  1136→        setPreferences={setPreferences}
  1137→      />
  1138→    </SidebarProvider>
  1139→  )
  1140→}
