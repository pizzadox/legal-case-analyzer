    31→// Format ISO timestamp in Russian style: dd.MM.yyyy HH:mm
    32→function formatRussianDateTime(iso: string): string {
    33→  try {
    34→    const d = new Date(iso)
    35→    if (isNaN(d.getTime())) return iso
    36→    const dd = String(d.getDate()).padStart(2, '0')
    37→    const mm = String(d.getMonth() + 1).padStart(2, '0')
    38→    const yyyy = d.getFullYear()
    39→    const hh = String(d.getHours()).padStart(2, '0')
    40→    const min = String(d.getMinutes()).padStart(2, '0')
    41→    return `${dd}.${mm}.${yyyy} ${hh}:${min}`
    42→  } catch {
    43→    return iso
    44→  }
    45→}

    50→function loadAllAnnotations(): Record<string, Annotation[]> {
    51→  if (typeof window === 'undefined') return {}
    52→  const loaded: Record<string, Annotation[]> = {}
    53→  try {
    54→    for (let i = 0; i < window.localStorage.length; i++) {
    55→      const key = window.localStorage.key(i)
    56→      if (key && key.startsWith(ANNOTATIONS_PREFIX)) {
    57→        const docId = key.replace(ANNOTATIONS_PREFIX, '')
    58→        const stored = window.localStorage.getItem(key)
    59→        if (stored) {
    60→          try {
    61→            const parsed = JSON.parse(stored)
    62→            if (Array.isArray(parsed)) loaded[docId] = parsed
    63→          } catch {
    64→            /* ignore malformed entries */
    65→          }
    66→        }
    67→      }
    68→    }
    69→  } catch {
    70→    /* localStorage unavailable */
    71→  }
    72→  return loaded
    73→}

   188→// Export CSV helper (Russian headers, semicolon-separated for Excel compatibility)
   189→function exportDocumentsCSV(docs: DocumentData[]) {
   190→  const headers = ['Название', 'Тип', 'Статус', 'Дата документа', 'Размер (КБ)', 'Дата загрузки', 'Описание']
   191→  const rows = docs.map(d => [
   192→    d.originalName,
   193→    d.documentType ?? '',
   194→    d.processingStatus,
   195→    d.documentDate ?? '',
   196→    Math.round(d.fileSize / 1024),
   197→    new Date(d.uploadedAt).toLocaleDateString('ru-RU'),
   198→    d.summary ?? '',
   199→  ])
   200→  const csv = [headers.join(';'), ...rows.map(r => r.join(';'))].join('\n')
   201→  const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8' })
   202→  const url = URL.createObjectURL(blob)
   203→  const a = document.createElement('a')
   204→  a.href = url
   205→  a.download = `documents_export_${new Date().toISOString().slice(0,10)}.csv`
   206→  a.click()
   207→  URL.revokeObjectURL(url)
   208→  toast.success('CSV экспорт выполнен')
   209→}

   768→                          aria-label="Очистить поиск"
   769→                        >

   771→                        </button>
   772→                      )}

   774→                    <Button
   775→                      size="sm"
   776→                      variant={docSearchCaseSensitive ? 'default' : 'outline'}
   777→                      className="h-8 px-2 text-xs font-mono"
   778→                      onClick={() => setDocSearchCaseSensitive(!docSearchCaseSensitive)}

   781→                      Aa
   782→                    </Button>
   783→                  </div>
   784→                  {docSearch.trim() && (
   785→                    <div className="flex items-center gap-2 text-xs">
   786→                      <Badge className={countMatches(selectedDoc.extractedText, docSearch, docSearchCaseSensitive) > 0 ? 'bg-emerald-700 text-white' : 'bg-stone-500 text-white'}>

   788→                      </Badge>
   789→                      {countMatches(selectedDoc.extractedText, docSearch, docSearchCaseSensitive) === 0 && (
   790→                        <span className="text-muted-foreground italic">Совпадений не найдено</span>
   791→                      )}
   792→                    </div>
   793→                  )}
   794→                  <ScrollArea className="max-h-64 w-full rounded-lg border bg-muted/30 p-3">
   795→                    <p className="text-xs whitespace-pre-wrap leading-relaxed">
   796→                      {docSearch.trim()
   797→                        ? highlightText(selectedDoc.extractedText, docSearch, docSearchCaseSensitive)
   798→                        : selectedDoc.extractedText}
   799→                    </p>
   800→                  </ScrollArea>
   801→                </CardContent>
   802→              </Card>
   803→            )}
   804→
   805→            {/* Annotations / Comments */}
   806→            <Card className="rounded-xl shadow-sm border-t-2 border-t-amber-500">

   808→                <CardTitle className="text-sm flex items-center gap-2">
   809→                  <MessageSquare className="w-4 h-4 text-amber-600" /> Комментарии и заметки
   810→                  <Badge variant="outline" className="text-xs ml-auto">
   811→                    {selectedDoc ? (annotations[selectedDoc.id]?.length ?? 0) : 0}
   812→                  </Badge>
   813→                </CardTitle>
   814→              </CardHeader>
   815→              <CardContent className="space-y-3">
   816→                {selectedDoc && (annotations[selectedDoc.id]?.length ?? 0) === 0 ? (

   818→                    <div className="flex items-center justify-center w-12 h-12 rounded-full bg-amber-500/10 mx-auto mb-2 ring-4 ring-amber-500/5">
   819→                      <MessageSquare className="w-6 h-6 text-amber-600/70" />
   820→                    </div>
   821→                    <p className="text-xs text-muted-foreground">Пока нет комментариев. Добавьте первый.</p>
   822→                  </div>
   823→                ) : (
   824→                  <div className="space-y-2">
   825→                    {selectedDoc && (annotations[selectedDoc.id] ?? []).map(an => (
   826→                      <div key={an.id} className="border-l-4 border-l-amber-500 bg-muted/30 rounded-r-lg p-2.5">
   827→                        <div className="flex items-start gap-2">
   828→                          <Avatar className="w-7 h-7 shrink-0">
   829→                            <AvatarFallback className="text-[10px] font-bold bg-amber-100 text-amber-800">АП</AvatarFallback>
   830→                          </Avatar>
   831→                          <div className="flex-1 min-w-0">
   832→                            <div className="flex items-center justify-between gap-2">
   833→                              <p className="text-xs font-semibold flex items-center gap-1 truncate">
   834→                                <User className="w-2.5 h-2.5 text-muted-foreground shrink-0" />
   835→                                {an.author}
   836→                              </p>
   837→                              <button
   838→                                type="button"
   839→                                onClick={() => handleDeleteAnnotation(selectedDoc.id, an.id)}

   842→                              >
   843→                                <Trash2 className="w-3 h-3" />
   844→                              </button>
   845→                            </div>
   846→                            <p className="text-xs mt-1 break-words">{an.text}</p>
   847→                            <p className="text-[10px] text-muted-foreground mt-1.5 flex items-center gap-1">
   848→                              <Calendar className="w-2.5 h-2.5" />
   849→                              {formatRussianDateTime(an.timestamp)}
   850→                            </p>
   851→                          </div>
   852→                        </div>
   853→                      </div>
   854→                    ))}
   855→                  </div>
   856→                )}
   857→
   858→                {/* Add annotation form */}
   859→                <div className="space-y-2 pt-3 border-t">
   860→                  <Textarea
   861→                    value={newAnnotation}
   862→                    onChange={(e) => setNewAnnotation(e.target.value)}
   863→                    placeholder="Введите ваш комментарий или заметку..."
   864→                    className="min-h-[80px] text-sm resize-none"
   865→                    maxLength={1000}
   866→                  />
   867→                  <Button
   868→                    size="sm"
   869→                    className="w-full rounded-xl bg-gradient-to-r from-amber-600 to-amber-700 text-white hover:from-amber-700 hover:to-amber-800"
   870→                    disabled={!newAnnotation.trim()}
   871→                    onClick={() => selectedDoc && handleAddAnnotation(selectedDoc.id)}
   872→                  >
   873→                    <Plus className="w-3 h-3 mr-1" /> Добавить комментарий
   874→                  </Button>
   875→                </div>
   876→              </CardContent>
   877→            </Card>
   878→
   879→            {/* Actions */}
   880→            <Card className="rounded-xl shadow-sm">
   881→              <CardHeader className="pb-2"><CardTitle className="text-sm">Действия</CardTitle></CardHeader>
   882→              <CardContent className="flex flex-wrap gap-2">
   883→                <Button
   884→                  size="sm"
   885→                  variant="outline"
   886→                  className="rounded-xl"
   887→                  onClick={() => shadcnToast({ title: 'Экспорт', description: 'Подготовка экспорта документа...' })}
   888→                >
   889→                  <Download className="w-3 h-3 mr-1" /> Экспорт
   890→                </Button>
   891→                <Button
   892→                  size="sm"
   893→                  variant="outline"
   894→                  className="rounded-xl"
   895→                  onClick={() => {
   896→                    shadcnToast({ title: 'Переобработка', description: 'Запущена повторная обработка документа.' })

   898→                  }}
   899→                >
   900→                  <RefreshCw className="w-3 h-3 mr-1" /> Переобработать
   901→                </Button>
   902→                <Button
   903→                  size="sm"
   904→                  variant="outline"
   905→                  className="rounded-xl text-red-700 hover:bg-red-500/10 hover:text-red-800 border-red-300"
   906→                  onClick={() => {
   907→                    if (selectedDoc) {
   908→                      handleDelete(selectedDoc.id)
   909→                      setSelectedDoc(null)
   910→                      setNewAnnotation('')
   911→                    }
   912→                  }}
   913→                >
   914→                  <Trash2 className="w-3 h-3 mr-1" /> Удалить
   915→                </Button>
   916→              </CardContent>
   917→            </Card>
   918→          </div>
   919→        </SheetContent>
   920→      </Sheet>
   921→
   922→      {/* Document Comparison Dialog */}
   923→      {compareDocs && (<DocumentCompareDialog doc1={compareDocs[0]} doc2={compareDocs[1]} onClose={() => setCompareDocs(null)} />)}
   924→    </div>
