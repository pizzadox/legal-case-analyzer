    29→function formatFileSize(bytes: number): string {
    30→  if (bytes < 1024) return bytes + ' Б'
    31→  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' КБ'
    32→  return (bytes / (1024 * 1024)).toFixed(1) + ' МБ'
    33→}

    35→function getStatusConfig(status: string) {
    36→  switch (status) {
    37→    case 'completed':
    38→      return { icon: <CheckCircle className="w-3.5 h-3.5" />, label: 'Завершён', className: 'bg-emerald-700 text-white border-emerald-600', pulse: false }
    39→    case 'processing':
    40→      return { icon: <Clock className="w-3.5 h-3.5" />, label: 'Обработка', className: 'bg-orange-600 text-white border-orange-500 animate-pulse', pulse: true }
    41→    case 'pending':
    42→      return { icon: <Clock className="w-3.5 h-3.5" />, label: 'Ожидание', className: 'bg-stone-500 text-white border-stone-400', pulse: false }
    43→    case 'failed':
    44→      return { icon: <AlertTriangle className="w-3.5 h-3.5" />, label: 'Ошибка', className: 'bg-red-700 text-white border-red-600', pulse: false }
    45→    default:
    46→      return { icon: <Clock className="w-3.5 h-3.5" />, label: status, className: 'bg-stone-400 text-white border-stone-300', pulse: false }
    47→  }
    48→}

    50→function getDocTypeConfig(type: string | null) {
    51→  if (!type) return { className: 'bg-stone-500/70 text-stone-200', label: 'Не определён' }
    52→  const map: Record<string, { className: string; label: string }> = {
    53→    'обвинение': { className: 'bg-red-700 text-white', label: 'Обвинение' },
    54→    'показание': { className: 'bg-orange-600 text-white', label: 'Показание' },
    55→    'протокол': { className: 'bg-amber-600 text-white', label: 'Протокол' },
    56→    'доказательство': { className: 'bg-emerald-700 text-white', label: 'Доказательство' },
    57→    'заключение': { className: 'bg-stone-600 text-white', label: 'Заключение' },
    58→  }
    59→  return map[type] || { className: 'bg-stone-500 text-white', label: type }
    60→}
