   143→function formatRussianDateTime(iso: string): string {
   144→  try {
   145→    return new Intl.DateTimeFormat('ru-RU', { day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' }).format(new Date(iso))
   146→  } catch { return iso }
   147→}
   148→
   149→function formatFileSize(bytes: number): string {
   150→  if (bytes < 1024) return `${bytes} Б`
   151→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} КБ`
   152→  return `${(bytes / 1048576).toFixed(1)} МБ`
   153→}
   154→
   155→function truncateText(text: string | null, maxLen: number = 100): string {
   156→  if (!text) return ''
   157→  return text.length > maxLen ? text.slice(0, maxLen) + '…' : text
   158→}

   160→function exportSearchCSV(results: SearchResultData) {
   161→  const rows: string[] = ['Category,Name,Type,Status']
   162→  results.documents.forEach(d => rows.push(`Document,"${d.originalName}",${d.documentType ?? ''},${d.processingStatus}`))
   163→  results.persons.forEach(p => rows.push(`Person,"${p.fullName}",${p.role ?? ''},${p.status ?? ''}`))
   164→  results.episodes.forEach(e => rows.push(`Episode,"${e.title}",${e.severity ?? ''},${e.status ?? ''}`))
   165→  results.crossReferences.forEach(cr => rows.push(`CrossRef,"${cr.referenceText}",${cr.referenceType ?? ''},"${cr.sourceDocument.originalName} → ${cr.targetDocument.originalName}"`))
   166→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
   167→  const url = URL.createObjectURL(blob)
   168→  const a = document.createElement('a')
   169→  a.href = url
   170→  a.download = 'search_results.csv'
   171→  a.click()
   172→  URL.revokeObjectURL(url)
   173→}

   175→// ──── Cross-Reference Graph Mini Widget (SVG) ────

   177→function CrossRefGraphMini({ nodes, highlightedDocId, onNodeClick }: {
   178→  nodes: CrossRefNode[]
   179→  highlightedDocId: string | null
   180→  onNodeClick: (docId: string) => void
   181→}) {
   182→  // Layout nodes in a circle
   183→  const svgW = 420
   184→  const svgH = 280
   185→  const cx = svgW / 2
   186→  const cy = svgH / 2
   187→  const radius = Math.min(svgW, svgH) * 0.35
   188→
   189→  const nodePositions = useMemo(() => {
   190→    const positions: Record<string, { x: number; y: number }> = {}
   191→    const total = nodes.length
   192→    nodes.forEach((node, i) => {
   193→      const angle = (2 * Math.PI * i) / total - Math.PI / 2
   194→      positions[node.documentId] = {
   195→        x: cx + radius * Math.cos(angle),
   196→        y: cy + radius * Math.sin(angle),
   197→      }
   198→    })
   199→    return positions
   200→  }, [nodes, cx, cy, radius])
   201→
   202→  // Build edges from linkedDocuments
   203→  const edges = useMemo(() => {
   204→    const edgeList: { source: string; target: string; refType: string | null }[] = []
   205→    nodes.forEach(node => {
   206→      node.linkedDocuments.forEach(link => {
   207→        // Avoid duplicate edges
   208→        const exists = edgeList.some(e =>
   209→          (e.source === node.documentId && e.target === link.id) ||
   210→          (e.source === link.id && e.target === node.documentId)
   211→        )
   212→        if (!exists) {
   213→          edgeList.push({ source: node.documentId, target: link.id, refType: link.refType })
   214→        }
   215→      })
   216→    })
   217→    return edgeList
   218→  }, [nodes])
   219→
   220→  // Find related docs when one is highlighted
   221→  const relatedDocs = useMemo(() => {
   222→    if (!highlightedDocId) return new Set<string>()
   223→    const related = new Set<string>()
   224→    related.add(highlightedDocId)
   225→    edges.forEach(e => {
   226→      if (e.source === highlightedDocId) related.add(e.target)
   227→      if (e.target === highlightedDocId) related.add(e.source)
   228→    })
   229→    return related
   230→  }, [highlightedDocId, edges])
   231→
   232→  return (
   233→    <Card className="rounded-xl shadow-sm">
   234→      <CardHeader className="pb-2">
   235→        <CardTitle className="text-sm flex items-center gap-2">
   236→          <Network className="w-4 h-4 text-amber-600" /> Граф перекрёстных ссылок
   237→          <Badge className="bg-stone-600 text-white text-xs">{nodes.length} документов</Badge>
   238→        </CardTitle>
   239→      </CardHeader>
   240→      <CardContent className="p-4">
   241→        <svg width={svgW} height={svgH} viewBox={`0 0 ${svgW} ${svgH}`} className="w-full h-auto">
   242→          {/* Edges */}
   243→          {edges.map((edge, i) => {
   244→            const src = nodePositions[edge.source]
   245→            const tgt = nodePositions[edge.target]
   246→            if (!src || !tgt) return null
   247→            const isHighlighted = relatedDocs.has(edge.source) && relatedDocs.has(edge.target)
   248→            const edgeColor = LINK_TYPE_COLOR[edge.refType ?? ''] ?? '#78716c'
   249→            return (
   250→              <line
   251→                key={`edge-${i}`}
   252→                x1={src.x} y1={src.y} x2={tgt.x} y2={tgt.y}
   253→                stroke={isHighlighted ? edgeColor : '#d6d3d1'}
   254→                strokeWidth={isHighlighted ? 2.5 : 1.5}
   255→                strokeDasharray={isHighlighted ? '' : '4 2'}
   256→                opacity={isHighlighted ? 1 : 0.5}
   257→              />
   258→            )
   259→          })}
   260→          {/* Nodes */}
   261→          {nodes.map(node => {
   262→            const pos = nodePositions[node.documentId]
   263→            if (!pos) return null
   264→            const isHighlighted = relatedDocs.has(node.documentId)
   265→            const nodeColor = NODE_TYPE_COLOR[node.documentType ?? ''] ?? '#78716c'
   266→            return (
   267→              <g key={node.documentId} onClick={() => onNodeClick(node.documentId)} className="cursor-pointer">
   268→                <circle
   269→                  cx={pos.x} cy={pos.y} r={isHighlighted ? 18 : 14}
   270→                  fill={isHighlighted ? nodeColor : '#f5f5f4'}
   271→                  stroke={nodeColor}
   272→                  strokeWidth={isHighlighted ? 3 : 2}
   273→                  opacity={isHighlighted ? 1 : 0.8}
   274→                />
   275→                <text
   276→                  x={pos.x} y={pos.y + 4}
   277→                  textAnchor="middle"
   278→                  fontSize={isHighlighted ? 10 : 8}
   279→                  fill={isHighlighted ? '#ffffff' : nodeColor}
   280→                  fontWeight={isHighlighted ? 'bold' : 'normal'}
   281→                >
   282→                  {node.documentName.slice(0, 12)}
   283→                </text>
   284→              </g>
   285→            )
   286→          })}
   287→        </svg>
   288→        {/* Legend */}
   289→        <div className="flex flex-wrap gap-3 mt-3">
   290→          {Object.entries(LINK_TYPE_COLOR).map(([type, color]) => (
   291→            <div key={type} className="flex items-center gap-1.5">
   292→              <div className="w-3 h-0.5 rounded" style={{ backgroundColor: color }} />
   293→              <span className="text-xs text-muted-foreground">{type}</span>
   294→            </div>
   295→          ))}
   296→        </div>
   297→        {highlightedDocId && (
   298→          <div className="mt-2 flex items-center gap-2 text-xs">
   299→            <Sparkles className="w-3 h-3 text-amber-600" />
   300→            <span className="text-amber-700 font-medium">
   301→              Выделен: {nodes.find(n => n.documentId === highlightedDocId)?.documentName ?? '—'}
   302→            </span>
   303→            <Button size="sm" variant="ghost" className="h-5 px-2 text-xs" onClick={() => onNodeClick(highlightedDocId)}>
   304→              <X className="w-3 h-3" /> Сбросить
   305→            </Button>
   306→          </div>
   307→        )}
   308→      </CardContent>
   309→    </Card>
   310→  )
   311→}
