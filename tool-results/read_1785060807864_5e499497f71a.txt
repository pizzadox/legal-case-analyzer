    26→function personLabel(p: { shortName?: string | null; fullName?: string | null; person?: { shortName?: string | null; fullName?: string | null } }): string {

    40→function exportEpisodesCSV(episodes: EpisodeData[]) {
    41→  const rows = ['Title,Severity,Status,Date,Persons,Articles']
    42→  episodes.forEach(e => {
    43→    const persons = e.persons.map(p => `${personLabel(p)} (${p.involvement ?? ''})`).join('; ')
    44→    const articles = e.articles.map(a => articleCode(a)).join('; ')
    45→    rows.push(`"${e.title}",${e.severity ?? ''},${e.status ?? ''},${e.date ?? ''},"${persons}","${articles}"`)
    46→  })
    47→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
    48→  const url = URL.createObjectURL(blob)
    49→  const a = document.createElement('a')
    50→  a.href = url
    51→  a.download = 'episodes.csv'
    52→  a.click()
    53→  URL.revokeObjectURL(url)
    54→  toast.success('CSV экспорт выполнен')
    55→}

   119→// Heat map cell color based on count intensity
   120→function heatColor(count: number, max: number): string {
   121→  if (max === 0) return 'bg-stone-50 text-stone-400'
   122→  const ratio = count / max
   123→  if (ratio === 0) return 'bg-stone-50 text-stone-400'
   124→  if (ratio <= 0.25) return 'bg-red-100 text-red-700'
   125→  if (ratio <= 0.5) return 'bg-red-200 text-red-800'
   126→  if (ratio <= 0.75) return 'bg-red-400 text-white'
   127→  return 'bg-red-700 text-white'
   128→}

   130→// Severity score calculation
   131→function severityScore(severity: string | null): number {
   132→  return SEVERITY_ORDER[severity ?? ''] ?? 0
   133→}
