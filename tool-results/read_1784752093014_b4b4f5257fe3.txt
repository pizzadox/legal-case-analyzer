   173→   173→function NotificationCenter({ notifications, setActiveSection }: {
   174→   174→  notifications: NotificationData[]
   175→   175→  setActiveSection: (section: SectionId) => void
   176→   176→}) {
   177→   177→  const [localNotifications, setLocalNotifications] = useState(notifications)
   178→   178→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   179→   179→
   180→   180→  const markAllRead = () => {
   181→   181→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   182→   182→    toast.success('Все уведомления отмечены как прочитанные')
   183→   183→  }
   184→   184→
   185→   185→  const handleClick = (notif: NotificationData) => {
   186→   186→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   187→   187→    if (notif.relatedSection) {
   188→   188→      setActiveSection(notif.relatedSection)
   189→   189→    }
   190→   190→  }
   191→   191→
   192→   192→  return (
   193→   193→    <Popover>
   194→   194→      <PopoverTrigger asChild>
   195→   195→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   196→   196→          <Bell className="h-4 w-4" />
   197→   197→          {unreadCount > 0 && (
   198→   198→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   199→   199→              {unreadCount}
   200→   200→            </span>
   201→   201→          )}
   202→   202→          <span className="sr-only">Уведомления</span>
   203→   203→        </Button>
   204→   204→      </PopoverTrigger>
   205→   205→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   206→   206→        <div className="p-3 border-b flex items-center gap-2">
   207→   207→          <Bell className="w-4 h-4 text-muted-foreground" />
   208→   208→          <p className="font-semibold text-sm">Уведомления</p>
   209→   209→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   210→   210→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   211→   211→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   212→   212→          </Button>
   213→   213→        </div>
   214→   214→        <div className="max-h-64 overflow-y-auto">
   215→   215→          {localNotifications.map(notif => (
   216→   216→            <div key={notif.id}
   217→   217→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   218→   218→              onClick={() => handleClick(notif)}>
   219→   219→              <div className="flex items-center gap-2">

   222→   222→                  <p className="text-sm font-medium truncate">{notif.title}</p>
   223→   223→                  <p className="text-xs text-muted-foreground truncate">{notif.description}</p>

   346→   346→function CaseHealthBadge() {
   347→   347→  // Compact health indicator for sidebar footer
   348→   348→  const [score, setScore] = useState<number | null>(null)
   349→   349→  useEffect(() => {
   350→   350→    fetch('/api/case/health-score')
   351→   351→      .then(r => r.ok ? r.json() : null)
   352→   352→      .then(d => {
   353→   353→        if (d && typeof d.score === 'number') setScore(d.score)
   354→   354→        else if (d && typeof d.overallScore === 'number') setScore(d.overallScore)
   355→   355→      })
   356→   356→      .catch(() => setScore(null))
   357→   357→  }, [])
   358→   358→
   359→   359→  if (score === null) {
   360→   360→    return (
   361→   361→      <div className="flex items-center gap-2 px-2 py-1.5 rounded-md bg-muted/50 text-xs text-muted-foreground">
   362→   362→        <Activity className="w-3.5 h-3.5 animate-pulse" />
   363→   363→        <span className="group-data-[collapsible=icon]:hidden">Загрузка...</span>
   364→   364→      </div>
   365→   365→    )
   366→   366→  }
   367→   367→
   368→   368→  // Color thresholds: <50 red, 50-75 amber, >75 emerald
   369→   369→  const color = score >= 75 ? 'emerald' : score >= 50 ? 'amber' : 'red'
   370→   370→  const colorClasses = {
   371→   371→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900',
   372→   372→    amber: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900',
   373→   373→    red: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900',
   374→   374→  }
   375→   375→  const statusText = score >= 75 ? 'Здоровое' : score >= 50 ? 'Среднее' : 'Проблемное'
   376→   376→
   377→   377→  return (
   378→   378→    <div className={`flex items-center gap-2 px-2 py-1.5 rounded-md border text-xs ${colorClasses[color]}`}>
   379→   379→      <Heart className="w-3.5 h-3.5 fill-current" />

   724→   724→function loadPreferences(): UserPreferences {

   728→   728→    if (!raw) return DEFAULT_PREFERENCES
   729→   729→    const parsed = JSON.parse(raw)
   730→   730→    return { ...DEFAULT_PREFERENCES, ...parsed }
   731→   731→  } catch {
   732→   732→    return DEFAULT_PREFERENCES
   733→   733→  }

   739→   739→}

   742→   742→  if (typeof document === 'undefined') return

   744→   744→  // Density
   745→   745→  body.classList.toggle('compact', prefs.density === 'compact')

   999→   999→  // Lazy-load preferences from localStorage on first render (avoids setState-in-effect)
  1000→  1000→  const [preferences, setPreferences] = useState<UserPreferences>(() => {
  1001→  1001→    if (typeof window === 'undefined') return DEFAULT_PREFERENCES
  1002→  1002→    const loaded = loadPreferences()
  1003→  1003→    // Apply to body immediately on first client render
  1004→  1004→    if (typeof document !== 'undefined') {
  1005→  1005→      applyPreferencesToBody(loaded)
  1006→  1006→    }
  1007→  1007→    return loaded
  1008→  1008→  })
  1009→  1009→  const [activeSection, setActiveSection] = useState<SectionId>(() => {
  1010→  1010→    if (typeof window === 'undefined') return 'dashboard'
  1011→  1011→    const loaded = loadPreferences()
  1012→  1012→    return loaded.defaultSection ?? 'dashboard'
  1013→  1013→  })
  1014→  1014→  const [helpOpen, setHelpOpen] = useState(false)
  1015→  1015→  const [commandOpen, setCommandOpen] = useState(false)
  1016→  1016→  const [settingsOpen, setSettingsOpen] = useState(false)
  1017→  1017→  const notifications = mockNotifications
  1018→  1018→
  1019→  1019→  // Auto-refresh handler
  1020→  1020→  useEffect(() => {
  1021→  1021→    if (!preferences.autoRefresh) return
  1022→  1022→    const interval = setInterval(() => {
  1023→  1023→      // Trigger a refetch by dispatching a custom event
  1024→  1024→      window.dispatchEvent(new CustomEvent('case-auto-refresh'))
  1025→  1025→    }, 30000) // 30 seconds
  1026→  1026→    return () => clearInterval(interval)
  1027→  1027→  }, [preferences.autoRefresh])
  1028→  1028→
  1029→  1029→  // Keyboard shortcuts handler
  1030→  1030→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
  1031→  1031→    // Ctrl+K opens command palette (highest priority)
  1032→  1032→    if (e.ctrlKey && (e.key === 'k' || e.key === 'K' || e.key === 'л' || e.key === 'Л')) {
  1033→  1033→      e.preventDefault()
  1034→  1034→      setCommandOpen(prev => !prev)
  1035→  1035→      return
  1036→  1036→    }
  1037→  1037→    // Ctrl+, (comma) opens settings
  1038→  1038→    if (e.ctrlKey && (e.key === ',' || e.key === 'б' || e.key === 'Б')) {
  1039→  1039→      e.preventDefault()
  1040→  1040→      setSettingsOpen(true)
  1041→  1041→      return
  1042→  1042→    }
  1043→  1043→    // Ctrl+1-8 for section navigation
  1044→  1044→    if (e.ctrlKey && !e.altKey && !e.shiftKey) {
  1045→  1045→      const num = parseInt(e.key)
  1046→  1046→      if (num >= 1 && num <= 8) {
  1047→  1047→        e.preventDefault()
  1048→  1048→        setActiveSection(NAV_ITEMS[num - 1].id)
  1049→  1049→      }
  1050→  1050→      // Ctrl+9 for timeline
  1051→  1051→      if (e.key === '9') {
  1052→  1052→        e.preventDefault()
  1053→  1053→        setActiveSection('timeline')
  1054→  1054→      }
  1055→  1055→      // Ctrl+0 for risk
  1056→  1056→      if (e.key === '0') {
  1057→  1057→        e.preventDefault()
  1058→  1058→        setActiveSection('risk')
  1059→  1059→      }
  1060→  1060→      // Ctrl+B for brief
  1061→  1061→      if (e.key === 'b' || e.key === 'B' || e.key === 'и' || e.key === 'И') {
  1062→  1062→        e.preventDefault()
  1063→  1063→        setActiveSection('brief')
  1064→  1064→      }
  1065→  1065→      // Ctrl+A for analytics
  1066→  1066→      if (e.key === 'a' || e.key === 'A' || e.key === 'ф' || e.key === 'Ф') {
  1067→  1067→        e.preventDefault()
  1068→  1068→        setActiveSection('analytics')
  1069→  1069→      }
  1070→  1070→      // Ctrl+E for evidence chain
  1071→  1071→      if (e.key === 'e' || e.key === 'E' || e.key === 'у' || e.key === 'У') {
  1072→  1072→        e.preventDefault()
  1073→  1073→        setActiveSection('evidence-chain')
  1074→  1074→      }
  1075→  1075→      // Ctrl+M for witness matrix
  1076→  1076→      if (e.key === 'm' || e.key === 'M' || e.key === 'ь' || e.key === 'Ь') {

  1078→  1078→        setActiveSection('witness-matrix')

  1080→  1080→    }

  1084→  1084→      const target = e.target as HTMLElement
