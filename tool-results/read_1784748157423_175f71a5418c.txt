   168→function NotificationCenter({ notifications, setActiveSection }: {
   169→  notifications: NotificationData[]
   170→  setActiveSection: (section: SectionId) => void
   171→}) {
   172→  const [localNotifications, setLocalNotifications] = useState(notifications)
   173→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   174→
   175→  const markAllRead = () => {
   176→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   177→    toast.success('Все уведомления отмечены как прочитанные')
   178→  }
   179→
   180→  const handleClick = (notif: NotificationData) => {
   181→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   182→    if (notif.relatedSection) {
   183→      setActiveSection(notif.relatedSection)
   184→    }
   185→  }
   186→
   187→  return (
   188→    <Popover>
   189→      <PopoverTrigger asChild>
   190→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   191→          <Bell className="h-4 w-4" />
   192→          {unreadCount > 0 && (
   193→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   194→              {unreadCount}
   195→            </span>
   196→          )}
   197→          <span className="sr-only">Уведомления</span>
   198→        </Button>
   199→      </PopoverTrigger>
   200→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   201→        <div className="p-3 border-b flex items-center gap-2">
   202→          <Bell className="w-4 h-4 text-muted-foreground" />
   203→          <p className="font-semibold text-sm">Уведомления</p>
   204→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   205→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   206→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   207→          </Button>
   208→        </div>
   209→        <div className="max-h-64 overflow-y-auto">
   210→          {localNotifications.map(notif => (
   211→            <div key={notif.id}
   212→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   213→              onClick={() => handleClick(notif)}>
   214→              <div className="flex items-center gap-2">
   215→                {NOTIF_TYPE_ICON[notif.type] ?? <Bell className="w-4 h-4" />}
   216→                <div className="flex-1 min-w-0">
   217→                  <p className="text-sm font-medium truncate">{notif.title}</p>
   218→                  <p className="text-xs text-muted-foreground truncate">{notif.description}</p>
   219→                </div>
   220→                {!notif.isRead && <div className="w-2 h-2 rounded-full bg-red-700 shrink-0" />}
   221→              </div>
   222→              <p className="text-xs text-muted-foreground mt-1">{new Date(notif.timestamp).toLocaleString('ru-RU', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' })}</p>
   223→            </div>
   224→          ))}
   225→          {localNotifications.length === 0 && (
   226→            <div className="p-6 text-center">
   227→              <Bell className="w-8 h-8 mx-auto text-muted-foreground" />
   228→              <p className="text-sm text-muted-foreground mt-2">Нет уведомлений</p>
   229→            </div>
   230→          )}
   231→        </div>
   232→      </PopoverContent>
   233→    </Popover>
   234→  )
   235→}

   340→function CaseHealthBadge() {
   341→  // Compact health indicator for sidebar footer
   342→  const [score, setScore] = useState<number | null>(null)
   343→  useEffect(() => {
   344→    fetch('/api/case/health-score')
   345→      .then(r => r.ok ? r.json() : null)
   346→      .then(d => {
   347→        if (d && typeof d.score === 'number') setScore(d.score)
   348→        else if (d && typeof d.overallScore === 'number') setScore(d.overallScore)
   349→      })
   350→      .catch(() => setScore(null))
   351→  }, [])
   352→
   353→  if (score === null) {
   354→    return (
   355→      <div className="flex items-center gap-2 px-2 py-1.5 rounded-md bg-muted/50 text-xs text-muted-foreground">
   356→        <Activity className="w-3.5 h-3.5 animate-pulse" />
   357→        <span className="group-data-[collapsible=icon]:hidden">Загрузка...</span>
   358→      </div>
   359→    )
   360→  }
   361→
   362→  // Color thresholds: <50 red, 50-75 amber, >75 emerald
   363→  const color = score >= 75 ? 'emerald' : score >= 50 ? 'amber' : 'red'
   364→  const colorClasses = {
   365→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900',
   366→    amber: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900',
   367→    red: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900',
   368→  }
   369→  const statusText = score >= 75 ? 'Здоровое' : score >= 50 ? 'Среднее' : 'Проблемное'
   370→
   371→  return (
   372→    <div className={`flex items-center gap-2 px-2 py-1.5 rounded-md border text-xs ${colorClasses[color]}`}>
   373→      <Heart className="w-3.5 h-3.5 fill-current" />
   374→      <div className="flex-1 min-w-0 group-data-[collapsible=icon]:hidden">
   375→        <div className="flex items-center gap-1.5">
   376→          <span className="font-semibold">{score}/100</span>
   377→          <span className="opacity-70">·</span>
   378→          <span className="truncate">{statusText}</span>
   379→        </div>
   380→        <div className="mt-1 h-1 rounded-full bg-black/10 dark:bg-white/10 overflow-hidden">
   381→          <div
   382→            className={`h-full rounded-full transition-all duration-700 ${
   383→              color === 'emerald' ? 'bg-emerald-600' : color === 'amber' ? 'bg-amber-600' : 'bg-red-700'
   384→            }`}
   385→            style={{ width: `${score}%` }}
   386→          />
   387→        </div>
   388→      </div>
   389→    </div>
   390→  )
   391→}

   719→function loadPreferences(): UserPreferences {
   720→  if (typeof window === 'undefined') return DEFAULT_PREFERENCES
   721→  try {
   722→    const raw = localStorage.getItem('case-user-preferences')
   723→    if (!raw) return DEFAULT_PREFERENCES
   724→    const parsed = JSON.parse(raw)
   725→    return { ...DEFAULT_PREFERENCES, ...parsed }
   726→  } catch {
   727→    return DEFAULT_PREFERENCES
   728→  }
   729→}
   730→
   731→function savePreferences(prefs: UserPreferences) {
   732→  if (typeof window === 'undefined') return
   733→  localStorage.setItem('case-user-preferences', JSON.stringify(prefs))
   734→}
   735→
   736→function applyPreferencesToBody(prefs: UserPreferences) {
   737→  if (typeof document === 'undefined') return
   738→  const body = document.body
   739→  // Density
   740→  body.classList.toggle('compact', prefs.density === 'compact')
   741→  // Font size
   742→  body.classList.remove('text-size-sm', 'text-size-md', 'text-size-lg')
   743→  body.classList.add(`text-size-${prefs.fontSize}`)
   744→  // Animations
   745→  body.classList.toggle('no-animations', !prefs.animations)
   746→}

   993→export default function Home() {
   994→  // Lazy-load preferences from localStorage on first render (avoids setState-in-effect)
   995→  const [preferences, setPreferences] = useState<UserPreferences>(() => {
   996→    if (typeof window === 'undefined') return DEFAULT_PREFERENCES
   997→    const loaded = loadPreferences()
   998→    // Apply to body immediately on first client render
   999→    if (typeof document !== 'undefined') {
  1000→      applyPreferencesToBody(loaded)
  1001→    }
  1002→    return loaded
  1003→  })
  1004→  const [activeSection, setActiveSection] = useState<SectionId>(() => {
  1005→    if (typeof window === 'undefined') return 'dashboard'
  1006→    const loaded = loadPreferences()
  1007→    return loaded.defaultSection ?? 'dashboard'
  1008→  })
  1009→  const [helpOpen, setHelpOpen] = useState(false)
  1010→  const [commandOpen, setCommandOpen] = useState(false)
  1011→  const [settingsOpen, setSettingsOpen] = useState(false)
  1012→  const notifications = mockNotifications
  1013→
  1014→  // Auto-refresh handler
  1015→  useEffect(() => {
  1016→    if (!preferences.autoRefresh) return
  1017→    const interval = setInterval(() => {
  1018→      // Trigger a refetch by dispatching a custom event
  1019→      window.dispatchEvent(new CustomEvent('case-auto-refresh'))
  1020→    }, 30000) // 30 seconds
  1021→    return () => clearInterval(interval)
  1022→  }, [preferences.autoRefresh])
  1023→
  1024→  // Keyboard shortcuts handler
  1025→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
  1026→    // Ctrl+K opens command palette (highest priority)
  1027→    if (e.ctrlKey && (e.key === 'k' || e.key === 'K' || e.key === 'л' || e.key === 'Л')) {
  1028→      e.preventDefault()
  1029→      setCommandOpen(prev => !prev)
  1030→      return
  1031→    }
  1032→    // Ctrl+, (comma) opens settings
  1033→    if (e.ctrlKey && (e.key === ',' || e.key === 'б' || e.key === 'Б')) {
  1034→      e.preventDefault()
  1035→      setSettingsOpen(true)
  1036→      return
  1037→    }
  1038→    // Ctrl+1-8 for section navigation
  1039→    if (e.ctrlKey && !e.altKey && !e.shiftKey) {
  1040→      const num = parseInt(e.key)
  1041→      if (num >= 1 && num <= 8) {
  1042→        e.preventDefault()
  1043→        setActiveSection(NAV_ITEMS[num - 1].id)
  1044→      }
  1045→      // Ctrl+9 for timeline
  1046→      if (e.key === '9') {
  1047→        e.preventDefault()
  1048→        setActiveSection('timeline')
  1049→      }
  1050→      // Ctrl+0 for risk
  1051→      if (e.key === '0') {
  1052→        e.preventDefault()
  1053→        setActiveSection('risk')
  1054→      }
  1055→      // Ctrl+B for brief
  1056→      if (e.key === 'b' || e.key === 'B' || e.key === 'и' || e.key === 'И') {
  1057→        e.preventDefault()
  1058→        setActiveSection('brief')
  1059→      }
  1060→      // Ctrl+A for analytics
  1061→      if (e.key === 'a' || e.key === 'A' || e.key === 'ф' || e.key === 'Ф') {
  1062→        e.preventDefault()
  1063→        setActiveSection('analytics')
  1064→      }
  1065→    }
  1066→    // ? for help
  1067→    if (e.key === '?' && !e.ctrlKey && !e.altKey) {
  1068→      // Only trigger when not in an input/textarea
  1069→      const target = e.target as HTMLElement
  1070→      if (target.tagName !== 'INPUT' && target.tagName !== 'TEXTAREA') {
  1071→        e.preventDefault()
  1072→        setHelpOpen(true)
  1073→      }
  1074→    }
  1075→    // Escape to close help or command palette
  1076→    if (e.key === 'Escape') {
  1077→      if (helpOpen) setHelpOpen(false)
  1078→      if (commandOpen) setCommandOpen(false)
  1079→      if (settingsOpen) setSettingsOpen(false)
  1080→    }
  1081→  }, [helpOpen, commandOpen, settingsOpen])
  1082→
  1083→  useEffect(() => {
  1084→    window.addEventListener('keydown', handleKeyDown)
  1085→    return () => window.removeEventListener('keydown', handleKeyDown)
  1086→  }, [handleKeyDown])
  1087→
  1088→  return (
  1089→    <SidebarProvider>
  1090→      <AppSidebar activeSection={activeSection} setActiveSection={setActiveSection} preferences={preferences} />
  1091→      <SidebarInset>
  1092→        <TopHeader
  1093→          activeSection={activeSection}
  1094→          notifications={notifications}
  1095→          setActiveSection={setActiveSection}
  1096→          onHelpClick={() => setHelpOpen(true)}
  1097→          onCommandClick={() => setCommandOpen(true)}
  1098→          onSettingsClick={() => setSettingsOpen(true)}
  1099→          preferences={preferences}
  1100→        />
  1101→        <div className="flex flex-1 flex-col min-h-0">
  1102→          <main className="flex-1 overflow-y-auto p-4 md:p-6">
  1103→            <MainContent activeSection={activeSection} />
  1104→          </main>
  1105→          <AppFooter />
  1106→        </div>
  1107→      </SidebarInset>
  1108→      <KeyboardShortcutsHelp open={helpOpen} onClose={() => setHelpOpen(false)} />
  1109→      <CommandPalette
  1110→        open={commandOpen}
  1111→        onOpenChange={setCommandOpen}
  1112→        setActiveSection={setActiveSection}
  1113→      />
  1114→      <SettingsDialog
  1115→        open={settingsOpen}
  1116→        onClose={() => setSettingsOpen(false)}
  1117→        preferences={preferences}
  1118→        setPreferences={setPreferences}
  1119→      />
  1120→    </SidebarProvider>
  1121→  )
