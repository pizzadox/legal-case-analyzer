   411→// Animated counter hook — smoothly increments from 0 to target value over ~700ms
   412→function useAnimatedCounter(target: number, durationMs: number = 700) {
   413→  const [value, setValue] = useState(0)
   414→  const rafRef = useRef<number | null>(null)
   415→  useEffect(() => {
   416→    const start = performance.now()
   417→    const startValue = value
   418→    const delta = target - startValue
   419→    if (delta === 0) return
   420→    const tick = (now: number) => {
   421→      const elapsed = now - start
   422→      const t = Math.min(1, elapsed / durationMs)
   423→      // ease-out cubic
   424→      const eased = 1 - Math.pow(1 - t, 3)
   425→      setValue(Math.round(startValue + delta * eased))
   426→      if (t < 1) {
   427→        rafRef.current = requestAnimationFrame(tick)
   428→      }
   429→    }
   430→    rafRef.current = requestAnimationFrame(tick)
   431→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current) }
   432→  }, [target, durationMs])
   433→  return value
   434→}

   652→function ProceduralDeadlinesTracker() {
   653→  const now = new Date()
   654→  const sorted = [...PROCEDURAL_DEADLINES].sort((a, b) => new Date(a.deadline).getTime() - new Date(b.deadline).getTime())
   655→  const upcoming = sorted.filter(d => new Date(d.deadline).getTime() > now.getTime())
   656→  const overdue = sorted.filter(d => new Date(d.deadline).getTime() <= now.getTime())
   657→  const nextDeadline = upcoming[0]
   658→  const daysToNext = nextDeadline
   659→    ? Math.ceil((new Date(nextDeadline.deadline).getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   660→    : 0
   661→
   662→  const fmtDate = (iso: string) => new Date(iso).toLocaleDateString('ru-RU', { day: '2-digit', month: 'long', year: 'numeric' })
   663→  const daysTo = (iso: string) => {
   664→    const diff = Math.ceil((new Date(iso).getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   665→    if (diff < 0) return `просрочено на ${Math.abs(diff)} дн.`
   666→    if (diff === 0) return 'сегодня'
   667→    if (diff === 1) return 'завтра'
   668→    return `через ${diff} дн.`
   669→  }
   670→
   671→  return (
   672→    <Card className="rounded-xl shadow-sm bg-gradient-to-br from-card via-card to-red-500/5 border-t-2 border-t-red-500 transition-shadow hover:shadow-md">
