   176→function NotificationCenter({ notifications, setActiveSection }: {
   177→  notifications: NotificationData[]
   178→  setActiveSection: (section: SectionId) => void
   179→}) {
   180→  const [localNotifications, setLocalNotifications] = useState(notifications)
   181→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   182→
   183→  const markAllRead = () => {
   184→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   185→    toast.success('Все уведомления отмечены как прочитанные')
   186→  }
   187→
   188→  const handleClick = (notif: NotificationData) => {
   189→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   190→    if (notif.relatedSection) {
   191→      setActiveSection(notif.relatedSection)
   192→    }
   193→  }
   194→
   195→  return (
   196→    <Popover>
   197→      <PopoverTrigger asChild>
   198→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   199→          <Bell className="h-4 w-4" />
   200→          {unreadCount > 0 && (
   201→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   202→              {unreadCount}
   203→            </span>
   204→          )}
   205→          <span className="sr-only">Уведомления</span>
   206→        </Button>
   207→      </PopoverTrigger>
   208→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   209→        <div className="p-3 border-b flex items-center gap-2">
   210→          <Bell className="w-4 h-4 text-muted-foreground" />
   211→          <p className="font-semibold text-sm">Уведомления</p>
   212→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   213→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   214→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   215→          </Button>
   216→        </div>
   217→        <div className="max-h-64 overflow-y-auto">
   218→          {localNotifications.map(notif => (
   219→            <div key={notif.id}
   220→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   221→              onClick={() => handleClick(notif)}>
   222→              <div className="flex items-center gap-2">
   223→                {NOTIF_TYPE_ICON[notif.type] ?? <Bell className="w-4 h-4" />}
   224→                <div className="flex-1 min-w-0">
   225→                  <p className="text-sm font-medium truncate">{notif.title}</p>
   226→                  <p className="text-xs text-muted-foreground truncate">{notif.description}</p>
   227→                </div>
   228→                {!notif.isRead && <div className="w-2 h-2 rounded-full bg-red-700 shrink-0" />}
   229→              </div>
   230→              <p className="text-xs text-muted-foreground mt-1">{new Date(notif.timestamp).toLocaleString('ru-RU', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' })}</p>
   231→            </div>
   232→          ))}
   233→          {localNotifications.length === 0 && (
   234→            <div className="p-6 text-center">
   235→              <Bell className="w-8 h-8 mx-auto text-muted-foreground" />
   236→              <p className="text-sm text-muted-foreground mt-2">Нет уведомлений</p>
   237→            </div>
   238→          )}
   239→        </div>
   240→      </PopoverContent>
   241→    </Popover>
   242→  )
   243→}

   349→function CaseHealthBadge() {
   350→  // Compact health indicator for sidebar footer
   351→  const [score, setScore] = useState<number | null>(null)
   352→  useEffect(() => {
   353→    fetch('/api/case/health-score')
   354→      .then(r => r.ok ? r.json() : null)
   355→      .then(d => {
   356→        if (d && typeof d.score === 'number') setScore(d.score)
   357→        else if (d && typeof d.overallScore === 'number') setScore(d.overallScore)
   358→      })
   359→      .catch(() => setScore(null))
   360→  }, [])
   361→
   362→  if (score === null) {
   363→    return (
   364→      <div className="flex items-center gap-2 px-2 py-1.5 rounded-md bg-muted/50 text-xs text-muted-foreground">
   365→        <Activity className="w-3.5 h-3.5 animate-pulse" />
   366→        <span className="group-data-[collapsible=icon]:hidden">Загрузка...</span>
   367→      </div>
   368→    )
   369→  }
   370→
   371→  // Color thresholds: <50 red, 50-75 amber, >75 emerald
   372→  const color = score >= 75 ? 'emerald' : score >= 50 ? 'amber' : 'red'
   373→  const colorClasses = {
   374→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900',
   375→    amber: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900',
   376→    red: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900',
   377→  }
   378→  const statusText = score >= 75 ? 'Здоровое' : score >= 50 ? 'Среднее' : 'Проблемное'
   379→
   380→  return (
   381→    <div className={`flex items-center gap-2 px-2 py-1.5 rounded-md border text-xs ${colorClasses[color]}`}>
   382→      <Heart className="w-3.5 h-3.5 fill-current" />
   383→      <div className="flex-1 min-w-0 group-data-[collapsible=icon]:hidden">
   384→        <div className="flex items-center gap-1.5">
   385→          <span className="font-semibold">{score}/100</span>
   386→          <span className="opacity-70">·</span>
   387→          <span className="truncate">{statusText}</span>
   388→        </div>
   389→        <div className="mt-1 h-1 rounded-full bg-black/10 dark:bg-white/10 overflow-hidden">
   390→          <div
   391→            className={`h-full rounded-full transition-all duration-700 ${
   392→              color === 'emerald' ? 'bg-emerald-600' : color === 'amber' ? 'bg-amber-600' : 'bg-red-700'
   393→            }`}
   394→            style={{ width: `${score}%` }}
   395→          />
   396→        </div>
   397→      </div>
   398→    </div>
   399→  )
   400→}

   728→function loadPreferences(): UserPreferences {
   729→  if (typeof window === 'undefined') return DEFAULT_PREFERENCES
   730→  try {
   731→    const raw = localStorage.getItem('case-user-preferences')
   732→    if (!raw) return DEFAULT_PREFERENCES
   733→    const parsed = JSON.parse(raw)
   734→    return { ...DEFAULT_PREFERENCES, ...parsed }
   735→  } catch {
   736→    return DEFAULT_PREFERENCES
   737→  }
   738→}
   739→
   740→function savePreferences(prefs: UserPreferences) {
   741→  if (typeof window === 'undefined') return
   742→  localStorage.setItem('case-user-preferences', JSON.stringify(prefs))
   743→}
   744→
   745→function applyPreferencesToBody(prefs: UserPreferences) {
   746→  if (typeof document === 'undefined') return
   747→  const body = document.body
   748→  // Density
   749→  body.classList.toggle('compact', prefs.density === 'compact')
   750→  // Font size
   751→  body.classList.remove('text-size-sm', 'text-size-md', 'text-size-lg')
   752→  body.classList.add(`text-size-${prefs.fontSize}`)
   753→  // Animations
   754→  body.classList.toggle('no-animations', !prefs.animations)
   755→}

  1002→export default function Home() {
  1003→  // Lazy-load preferences from localStorage on first render (avoids setState-in-effect)
  1004→  const [preferences, setPreferences] = useState<UserPreferences>(() => {
  1005→    if (typeof window === 'undefined') return DEFAULT_PREFERENCES
  1006→    const loaded = loadPreferences()
  1007→    // Apply to body immediately on first client render
  1008→    if (typeof document !== 'undefined') {
  1009→      applyPreferencesToBody(loaded)
  1010→    }
  1011→    return loaded
  1012→  })
  1013→  const [activeSection, setActiveSection] = useState<SectionId>(() => {
  1014→    if (typeof window === 'undefined') return 'dashboard'
  1015→    const loaded = loadPreferences()
  1016→    return loaded.defaultSection ?? 'dashboard'
  1017→  })
  1018→  const [helpOpen, setHelpOpen] = useState(false)
  1019→  const [commandOpen, setCommandOpen] = useState(false)
  1020→  const [settingsOpen, setSettingsOpen] = useState(false)
  1021→  const notifications = mockNotifications
  1022→
  1023→  // Auto-refresh handler
  1024→  useEffect(() => {
  1025→    if (!preferences.autoRefresh) return
  1026→    const interval = setInterval(() => {
  1027→      // Trigger a refetch by dispatching a custom event
  1028→      window.dispatchEvent(new CustomEvent('case-auto-refresh'))
  1029→    }, 30000) // 30 seconds
  1030→    return () => clearInterval(interval)
  1031→  }, [preferences.autoRefresh])
  1032→
  1033→  // Keyboard shortcuts handler
  1034→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
  1035→    // Ctrl+K opens command palette (highest priority)
  1036→    if (e.ctrlKey && (e.key === 'k' || e.key === 'K' || e.key === 'л' || e.key === 'Л')) {
  1037→      e.preventDefault()
  1038→      setCommandOpen(prev => !prev)
  1039→      return
  1040→    }
  1041→    // Ctrl+, (comma) opens settings
  1042→    if (e.ctrlKey && (e.key === ',' || e.key === 'б' || e.key === 'Б')) {
  1043→      e.preventDefault()
  1044→      setSettingsOpen(true)
  1045→      return
  1046→    }
  1047→    // Ctrl+1-8 for section navigation
  1048→    if (e.ctrlKey && !e.altKey && !e.shiftKey) {
  1049→      const num = parseInt(e.key)
  1050→      if (num >= 1 && num <= 8) {
  1051→        e.preventDefault()
  1052→        setActiveSection(NAV_ITEMS[num - 1].id)
  1053→      }
  1054→      // Ctrl+9 for timeline
  1055→      if (e.key === '9') {
  1056→        e.preventDefault()
  1057→        setActiveSection('timeline')
  1058→      }
  1059→      // Ctrl+0 for risk
  1060→      if (e.key === '0') {
  1061→        e.preventDefault()
  1062→        setActiveSection('risk')
  1063→      }
  1064→      // Ctrl+B for brief
  1065→      if (e.key === 'b' || e.key === 'B' || e.key === 'и' || e.key === 'И') {
  1066→        e.preventDefault()
  1067→        setActiveSection('brief')
  1068→      }
  1069→      // Ctrl+A for analytics
  1070→      if (e.key === 'a' || e.key === 'A' || e.key === 'ф' || e.key === 'Ф') {
  1071→        e.preventDefault()
  1072→        setActiveSection('analytics')
  1073→      }
  1074→      // Ctrl+E for evidence chain
  1075→      if (e.key === 'e' || e.key === 'E' || e.key === 'у' || e.key === 'У') {
  1076→        e.preventDefault()
  1077→        setActiveSection('evidence-chain')
  1078→      }
  1079→      // Ctrl+M for witness matrix
  1080→      if (e.key === 'm' || e.key === 'M' || e.key === 'ь' || e.key === 'Ь') {
  1081→        e.preventDefault()
  1082→        setActiveSection('witness-matrix')
  1083→      }
  1084→      // Ctrl+X for export center
  1085→      if (e.key === 'x' || e.key === 'X' || e.key === 'ч' || e.key === 'Ч') {
  1086→        e.preventDefault()
  1087→        setActiveSection('export-center')
  1088→      }
  1089→    }
  1090→    // ? for help
  1091→    if (e.key === '?' && !e.ctrlKey && !e.altKey) {
  1092→      // Only trigger when not in an input/textarea
  1093→      const target = e.target as HTMLElement
  1094→      if (target.tagName !== 'INPUT' && target.tagName !== 'TEXTAREA') {
  1095→        e.preventDefault()
  1096→        setHelpOpen(true)
  1097→      }
  1098→    }
  1099→    // Escape to close help or command palette
  1100→    if (e.key === 'Escape') {
  1101→      if (helpOpen) setHelpOpen(false)
  1102→      if (commandOpen) setCommandOpen(false)
  1103→      if (settingsOpen) setSettingsOpen(false)
  1104→    }
  1105→  }, [helpOpen, commandOpen, settingsOpen])
  1106→
  1107→  useEffect(() => {
  1108→    window.addEventListener('keydown', handleKeyDown)
  1109→    return () => window.removeEventListener('keydown', handleKeyDown)
  1110→  }, [handleKeyDown])
  1111→
  1112→  return (
  1113→    <SidebarProvider>
  1114→      <AppSidebar activeSection={activeSection} setActiveSection={setActiveSection} preferences={preferences} />
  1115→      <SidebarInset>
  1116→        <TopHeader
  1117→          activeSection={activeSection}
  1118→          notifications={notifications}
  1119→          setActiveSection={setActiveSection}
  1120→          onHelpClick={() => setHelpOpen(true)}
  1121→          onCommandClick={() => setCommandOpen(true)}
  1122→          onSettingsClick={() => setSettingsOpen(true)}
  1123→          preferences={preferences}
  1124→        />
  1125→        <div className="flex flex-1 flex-col min-h-0">
  1126→          <main className="flex-1 overflow-y-auto p-4 md:p-6">
  1127→            <MainContent activeSection={activeSection} />
  1128→          </main>
  1129→          <AppFooter />
  1130→        </div>
  1131→      </SidebarInset>
  1132→      <KeyboardShortcutsHelp open={helpOpen} onClose={() => setHelpOpen(false)} />
  1133→      <CommandPalette
  1134→        open={commandOpen}
  1135→        onOpenChange={setCommandOpen}
  1136→        setActiveSection={setActiveSection}
  1137→      />
  1138→      <SettingsDialog
  1139→        open={settingsOpen}
  1140→        onClose={() => setSettingsOpen(false)}
  1141→        preferences={preferences}
  1142→        setPreferences={setPreferences}
  1143→      />
  1144→    </SidebarProvider>
  1145→  )
  1146→}
