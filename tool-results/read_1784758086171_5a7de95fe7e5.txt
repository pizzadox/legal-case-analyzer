   145→// Export helper
   146→function exportPersonsCSV(persons: PersonData[]) {
   147→  const rows = ['Name,Role,Status,GuiltLevel,Occupation']
   148→  persons.forEach(p => {
   149→    rows.push(`"${p.fullName}",${p.role ?? ''},${p.status ?? ''},${p.guiltLevel ?? 'none'},${p.occupation ?? ''}`)
   150→  })
   151→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
   152→  const url = URL.createObjectURL(blob)
   153→  const a = document.createElement('a')
   154→  a.href = url
   155→  a.download = 'persons.csv'
   156→  a.click()
   157→  URL.revokeObjectURL(url)
   158→  toast.success('CSV экспорт выполнен')
   159→}

   195→// === Person Summary Stats Bar ===
   196→function PersonSummaryStats({ persons }: { persons: PersonData[] }) {
   197→  const stats = useMemo(() => {
   198→    const total = persons.length
   199→    const byRole: Record<string, number> = {}
   200→    persons.forEach(p => {
   201→      const role = p.role ?? 'не указана'
   202→      byRole[role] = (byRole[role] ?? 0) + 1
   203→    })
   204→    const avgGuilt = total > 0
   205→      ? Math.round(persons.reduce((sum, p) => sum + (GUILT[p.guiltLevel ?? 'none'].pct), 0) / total)
   206→      : 0
   207→    const defenseCoverage = total > 0
   208→      ? Math.round(persons.filter(p => p.defenseStrategy).length / total * 100)
   209→      : 0
   210→    return { total, byRole, avgGuilt, defenseCoverage }
   211→  }, [persons])
   212→
   213→  const pieData = Object.entries(stats.byRole).map(([role, count]) => ({
   214→    role: ROLE_LABEL[role] ?? role,
   215→    count,
   216→    fill: ROLE_BADGE[role]?.replace('bg-', '').replace(' text-white', '') || 'stone-500',
   217→    color: role === 'обвиняемый' ? '#dc2626'
   218→      : role === 'соучастник' ? '#ea580c'
   219→      : role === 'свидетель' ? '#ca8a04'
   220→      : role === 'потерпевшая' || role === 'потерпевший' ? '#047857'
   221→      : role === 'следователь' ? '#78716c'
   222→      : '#a8a29e',
   223→  }))
   224→
   225→  const pieConfig = Object.fromEntries(
   226→    pieData.map(d => [d.role, { label: d.role, color: d.color }])
   227→  )
   228→
   229→  return (
   230→    <Card className="rounded-xl shadow-sm border-l-4 border-amber-600">
   231→      <CardContent className="p-4">
   232→        <div className="flex items-center gap-2 mb-3">
   233→          <div className="flex items-center justify-center w-8 h-8 rounded-xl bg-amber-600/20">
   234→            <Network className="w-4 h-4 text-amber-600" />
   235→          </div>
   236→          <p className="font-semibold text-sm">Сводка по участникам</p>
   237→        </div>
   238→        <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   239→          {/* Total persons */}
   240→          <div className="flex flex-col items-center justify-center p-3 rounded-lg bg-muted/30 border">
   241→            <Users className="w-5 h-5 text-amber-600 mb-1" />
   242→            <p className="text-2xl font-bold">{stats.total}</p>
   243→            <p className="text-xs text-muted-foreground">Участников</p>
   244→          </div>
   245→          {/* Role pie chart */}
   246→          <div className="flex flex-col items-center justify-center p-2 rounded-lg bg-muted/30 border">
   247→            <ChartContainer config={pieConfig} className="h-[80px] w-[100px]">
   248→              <PieChart>
   249→                <Pie
   250→                  data={pieData}
   251→                  dataKey="count"

   253→                  cx="50%"

   332→// Relationship map section (enhanced)
   333→function RelationshipMap({ relationships, persons }: { relationships: PersonRelationship[]; persons: PersonData[] }) {
   334→  const personRelMap = useMemo(() => {
   335→    const map: Record<string, PersonRelationship[]> = {}
   336→    relationships.forEach(r => {
   337→      if (!map[r.sourcePersonId]) map[r.sourcePersonId] = []
   338→      map[r.sourcePersonId].push(r)
   339→    })
   340→    return map
   341→  }, [relationships])
   342→
   343→  const relCount = useMemo(() => {
   344→    const counts: Record<string, number> = {}
   345→    relationships.forEach(r => {
   346→      counts[r.sourcePersonId] = (counts[r.sourcePersonId] ?? 0) + 1
   347→      counts[r.targetPersonId] = (counts[r.targetPersonId] ?? 0) + 1
   348→    })
   349→    return counts
   350→  }, [relationships])
   351→
   352→  return (
   353→    <Card className="rounded-xl shadow-sm">
   354→      <CardHeader className="pb-2">
   355→        <CardTitle className="text-sm flex items-center gap-2">
   356→          <Link2 className="w-4 h-4 text-amber-600" /> Связи между участниками
   357→          <Badge variant="outline" className="text-xs">{relationships.length} связей</Badge>
   358→        </CardTitle>
   359→      </CardHeader>
   360→      <CardContent className="p-4">
   361→        <div className="grid sm:grid-cols-2 gap-3 max-h-96 overflow-y-auto scrollbar-thin pr-1">
   362→          {persons.map(person => {
   363→            const rels = personRelMap[person.id] ?? []
   364→            const count = relCount[person.id] ?? 0
   365→            const heatClass = count >= 3 ? 'border-l-red-700 bg-red-50/40 dark:bg-red-950/20'
   366→              : count === 2 ? 'border-l-amber-600 bg-amber-50/40 dark:bg-amber-950/20'
   367→              : count === 1 ? 'border-l-stone-400 bg-stone-50/40 dark:bg-stone-900/20'
   368→              : 'border-l-transparent'
   369→            return (
   370→              <Card key={person.id} className={`rounded-xl border border-l-4 ${heatClass} shadow-sm transition-all duration-200 hover:shadow-md hover:translate-x-0.5`}>
   371→                <CardContent className="p-3">
   372→                  <div className="flex items-center justify-between gap-2 mb-2">
   373→                    <div className="flex items-center gap-2 min-w-0 flex-1">
   374→                      <Users className="w-4 h-4 text-muted-foreground shrink-0" />
   375→                      <p className="font-medium text-sm truncate">{person.shortName ?? person.fullName}</p>
   376→                    </div>
   377→                    <Badge className={`${ROLE_BADGE[person.role ?? ''] ?? 'bg-stone-500 text-white'} text-xs shrink-0`}>{ROLE_LABEL[person.role ?? ''] ?? person.role}</Badge>
   378→                  </div>
   379→                  {rels.length > 0 ? (
   380→                    <div className="space-y-1.5">
   381→                      {rels.map(rel => {
   382→                        const edgeColor = REL_EDGE_COLOR[rel.relationshipType] ?? '#78716c'
   383→                        const edgeCategory = REL_EDGE_CATEGORY[rel.relationshipType] ?? 'professional'
   384→                        return (
   385→                          <TooltipProvider key={rel.id}>
   386→                            <Tooltip>
   387→                              <TooltipTrigger asChild>
   388→                                <div className="flex items-center gap-1.5 text-xs bg-muted/40 rounded-md px-2 py-1.5 cursor-help">
   389→                                  <ArrowRight className="w-3 h-3 shrink-0" style={{ color: edgeColor }} />
   390→                                  <span className="font-medium truncate flex-1 min-w-0">{rel.targetPersonName}</span>
   391→                                  <Badge className={`${REL_TYPE_BADGE[rel.relationshipType] ?? 'bg-stone-500 text-white'} text-[10px] shrink-0 leading-tight`}>{rel.relationshipType}</Badge>
   392→                                </div>
   393→                              </TooltipTrigger>
   394→                              <TooltipContent className="max-w-[250px]">
   395→                                <p className="font-medium">{rel.sourcePersonName} → {rel.targetPersonName}</p>
   396→                                <p className="text-xs">{rel.description}</p>
   397→                                <p className="text-xs mt-1">Тип связи: {REL_CATEGORY_COLOR[edgeCategory]?.label ?? edgeCategory}</p>
   398→                              </TooltipContent>
   399→                            </Tooltip>
   400→                          </TooltipProvider>
   401→                        )
   402→                      })}
   403→                    </div>
   404→                  ) : (
   405→                    <p className="text-xs text-muted-foreground italic">Нет исходящих связей</p>
   406→                  )}
   407→                </CardContent>
   408→              </Card>
   409→            )
   410→          })}
   411→        </div>
