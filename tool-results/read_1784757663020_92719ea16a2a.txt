   128→// ─── Animated Counter Hook ───────────────────────────────────────────
   129→
   130→function useAnimatedCounter(target: number, duration: number = 1200) {
   131→  const [value, setValue] = useState(0)
   132→  useEffect(() => {
   133→    let start = 0
   134→    const startTime = performance.now()
   135→    const animate = (currentTime: number) => {
   136→      const elapsed = currentTime - startTime
   137→      const progress = Math.min(elapsed / duration, 1)
   138→      // Ease-out cubic
   139→      const eased = 1 - Math.pow(1 - progress, 3)
   140→      setValue(Math.round(start + (target - start) * eased))
   141→      if (progress < 1) requestAnimationFrame(animate)
   142→    }
   143→    requestAnimationFrame(animate)
   144→    return () => { setValue(0) }
   145→  }, [target, duration])
   146→  return value
   147→}

   149→// ─── Animated Number Component ───────────────────────────────────────

   151→function AnimatedNumber({ value, className, duration = 1200 }: { value: number; className?: string; duration?: number }) {
   152→  const animated = useAnimatedCounter(value, duration)
   153→  return <span className={className}>{animated}</span>
   154→}

   156→// ─── Complexity Gauge SVG ────────────────────────────────────────────
   157→
   158→function ComplexityGauge({ score }: { score: number }) {
   159→  const animatedScore = useAnimatedCounter(score, 1500)
   160→
   161→  // Gauge segments: 0-25 emerald, 25-50 amber, 50-75 orange-red, 75-100 red
   162→  const segments = [
   163→    { start: 0, end: 25, color: '#059669', label: 'Низкая' },
   164→    { start: 25, end: 50, color: '#d97706', label: 'Умеренная' },
   165→    { start: 50, end: 75, color: '#ea580c', label: 'Высокая' },
   166→    { start: 75, end: 100, color: '#dc2626', label: 'Экстремальная' },
   167→  ]
   168→
   169→  const cx = 120, cy = 120, r = 100, strokeWidth = 14
   170→  const circumference = 2 * Math.PI * r
   171→
   172→  // Get the current segment color
   173→  const currentColor = score >= 75 ? '#dc2626' : score >= 50 ? '#ea580c' : score >= 25 ? '#d97706' : '#059669'
   174→
   175→  return (
   176→    <div className="flex flex-col items-center gap-2 relative">
   177→      <svg width="240" height="240" viewBox="0 0 240 240" className="transform -rotate-90">
   178→        {/* Background segments */}
   179→        {segments.map((seg, i) => {
   180→          const segLength = ((seg.end - seg.start) / 100) * circumference
   181→          const segOffset = ((seg.start / 100) * circumference)
   182→          return (
   183→            <circle
   184→              key={i}
   185→              cx={cx}
   186→              cy={cy}
   187→              r={r}
   188→              stroke={seg.color}
   189→              strokeWidth={strokeWidth}
   190→              fill="none"
   191→              strokeDasharray={`${segLength} ${circumference - segLength}`}
   192→              strokeDashoffset={-segOffset}
   193→              opacity={0.2}
   194→              className="transition-all duration-500"
   195→            />
   196→          )
   197→        })}
   198→        {/* Active arc (animated) */}
   199→        <circle
   200→          cx={cx}
   201→          cy={cy}
   202→          r={r}
   203→          stroke={currentColor}
   204→          strokeWidth={strokeWidth}
   205→          fill="none"
   206→          strokeDasharray={circumference}
   207→          strokeDashoffset={circumference - (animatedScore / 100) * circumference}
   208→          strokeLinecap="round"
   209→          className="transition-all duration-300"
   210→          style={{ filter: `drop-shadow(0 0 6px ${currentColor}40)` }}
   211→        />
   212→      </svg>
   213→      {/* Center label - positioned over the SVG */}
   214→      <div className="absolute flex flex-col items-center justify-center" style={{ width: 240, height: 240 }}>
   215→        <AnimatedNumber value={score} className="text-4xl font-bold" duration={1500} />
   216→        <span className="text-xs text-muted-foreground mt-1">из 100</span>
   217→      </div>
   218→      {/* Segment labels */}
   219→      <div className="flex items-center justify-center gap-3 text-xs mt-1">
   220→        {segments.map((seg, i) => (
   221→          <div key={i} className="flex items-center gap-1">
   222→            <span className="w-2 h-2 rounded-full" style={{ backgroundColor: seg.color }} />
   223→            <span className="text-muted-foreground">{seg.label}</span>
   224→          </div>
   225→        ))}
   226→      </div>
   227→    </div>
   228→  )
   229→}

   413→export function CaseAnalytics() {

   416→    queryFn: getAnalytics,

   420→  const analytics = data ?? mockAnalytics
   421→
   422→  // Ensure outcome predictions have isMostLikely flag
   423→  const enrichedPredictions = useMemo(() => {
   424→    const maxProb = Math.max(...analytics.outcomePrediction.map(o => o.probability))
   425→    return analytics.outcomePrediction.map(o => ({
   426→      ...o,
   427→      riskAdjustedProbability: o.riskAdjustedProbability ?? Math.round(o.probability * 0.85),
   428→      defenseImpact: o.defenseImpact ?? '',
   429→      isMostLikely: o.isMostLikely ?? o.probability === maxProb,
   430→    }))
   431→  }, [analytics.outcomePrediction])
   432→
   433→  // Ensure insights have relatedEntities
   434→  const enrichedInsights = useMemo(() => {
   435→    return analytics.insights.map(insight => ({
   436→      ...insight,
   437→      relatedEntities: insight.relatedEntities ?? [],
   438→      actionRecommendation: insight.actionRecommendation ?? '',
   439→    }))
   440→  }, [analytics.insights])
   441→
   442→  // Processing trend with trend line
   443→  const trendData = useMemo(() => {
