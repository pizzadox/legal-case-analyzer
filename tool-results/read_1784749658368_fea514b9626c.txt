   171→function NotificationCenter({ notifications, setActiveSection }: {
   172→  notifications: NotificationData[]
   173→  setActiveSection: (section: SectionId) => void
   174→}) {
   175→  const [localNotifications, setLocalNotifications] = useState(notifications)
   176→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   177→
   178→  const markAllRead = () => {
   179→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   180→    toast.success('Все уведомления отмечены как прочитанные')
   181→  }
   182→
   183→  const handleClick = (notif: NotificationData) => {
   184→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   185→    if (notif.relatedSection) {
   186→      setActiveSection(notif.relatedSection)
   187→    }
   188→  }
   189→
   190→  return (
   191→    <Popover>
   192→      <PopoverTrigger asChild>
   193→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   194→          <Bell className="h-4 w-4" />
   195→          {unreadCount > 0 && (
   196→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   197→              {unreadCount}
   198→            </span>
   199→          )}
   200→          <span className="sr-only">Уведомления</span>
   201→        </Button>
   202→      </PopoverTrigger>
   203→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   204→        <div className="p-3 border-b flex items-center gap-2">
   205→          <Bell className="w-4 h-4 text-muted-foreground" />
   206→          <p className="font-semibold text-sm">Уведомления</p>
   207→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   208→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   209→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   210→          </Button>
   211→        </div>
   212→        <div className="max-h-64 overflow-y-auto">
   213→          {localNotifications.map(notif => (
   214→            <div key={notif.id}
   215→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   216→              onClick={() => handleClick(notif)}>
   217→              <div className="flex items-center gap-2">
   218→                {NOTIF_TYPE_ICON[notif.type] ?? <Bell className="w-4 h-4" />}
   219→                <div className="flex-1 min-w-0">
   220→                  <p className="text-sm font-medium truncate">{notif.title}</p>
   221→                  <p className="text-xs text-muted-foreground truncate">{notif.description}</p>
   222→                </div>
   223→                {!notif.isRead && <div className="w-2 h-2 rounded-full bg-red-700 shrink-0" />}
   224→              </div>
   225→              <p className="text-xs text-muted-foreground mt-1">{new Date(notif.timestamp).toLocaleString('ru-RU', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' })}</p>
   226→            </div>
   227→          ))}
   228→          {localNotifications.length === 0 && (
   229→            <div className="p-6 text-center">
   230→              <Bell className="w-8 h-8 mx-auto text-muted-foreground" />
   231→              <p className="text-sm text-muted-foreground mt-2">Нет уведомлений</p>
   232→            </div>
   233→          )}
   234→        </div>
   235→      </PopoverContent>
   236→    </Popover>
   237→  )
   238→}

   343→function CaseHealthBadge() {
   344→  // Compact health indicator for sidebar footer
   345→  const [score, setScore] = useState<number | null>(null)
   346→  useEffect(() => {
   347→    fetch('/api/case/health-score')
   348→      .then(r => r.ok ? r.json() : null)
   349→      .then(d => {
   350→        if (d && typeof d.score === 'number') setScore(d.score)
   351→        else if (d && typeof d.overallScore === 'number') setScore(d.overallScore)
   352→      })
   353→      .catch(() => setScore(null))
   354→  }, [])
   355→
   356→  if (score === null) {
   357→    return (
   358→      <div className="flex items-center gap-2 px-2 py-1.5 rounded-md bg-muted/50 text-xs text-muted-foreground">
   359→        <Activity className="w-3.5 h-3.5 animate-pulse" />
   360→        <span className="group-data-[collapsible=icon]:hidden">Загрузка...</span>
   361→      </div>
   362→    )
   363→  }
   364→
   365→  // Color thresholds: <50 red, 50-75 amber, >75 emerald
   366→  const color = score >= 75 ? 'emerald' : score >= 50 ? 'amber' : 'red'
   367→  const colorClasses = {
   368→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900',
   369→    amber: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900',
   370→    red: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900',
   371→  }
   372→  const statusText = score >= 75 ? 'Здоровое' : score >= 50 ? 'Среднее' : 'Проблемное'
   373→
   374→  return (
   375→    <div className={`flex items-center gap-2 px-2 py-1.5 rounded-md border text-xs ${colorClasses[color]}`}>
   376→      <Heart className="w-3.5 h-3.5 fill-current" />
   377→      <div className="flex-1 min-w-0 group-data-[collapsible=icon]:hidden">
   378→        <div className="flex items-center gap-1.5">
   379→          <span className="font-semibold">{score}/100</span>
   380→          <span className="opacity-70">·</span>
   381→          <span className="truncate">{statusText}</span>
   382→        </div>
   383→        <div className="mt-1 h-1 rounded-full bg-black/10 dark:bg-white/10 overflow-hidden">
   384→          <div
   385→            className={`h-full rounded-full transition-all duration-700 ${
   386→              color === 'emerald' ? 'bg-emerald-600' : color === 'amber' ? 'bg-amber-600' : 'bg-red-700'
   387→            }`}
   388→            style={{ width: `${score}%` }}
   389→          />
   390→        </div>
   391→      </div>
   392→    </div>
   393→  )
   394→}

   723→function loadPreferences(): UserPreferences {
   724→  if (typeof window === 'undefined') return DEFAULT_PREFERENCES
   725→  try {
   726→    const raw = localStorage.getItem('case-user-preferences')
   727→    if (!raw) return DEFAULT_PREFERENCES
   728→    const parsed = JSON.parse(raw)
   729→    return { ...DEFAULT_PREFERENCES, ...parsed }
   730→  } catch {
   731→    return DEFAULT_PREFERENCES
   732→  }
   733→}
   734→
   735→function savePreferences(prefs: UserPreferences) {
   736→  if (typeof window === 'undefined') return
   737→  localStorage.setItem('case-user-preferences', JSON.stringify(prefs))
   738→}
   739→
   740→function applyPreferencesToBody(prefs: UserPreferences) {
   741→  if (typeof document === 'undefined') return
   742→  const body = document.body
   743→  // Density
   744→  body.classList.toggle('compact', prefs.density === 'compact')
   745→  // Font size
   746→  body.classList.remove('text-size-sm', 'text-size-md', 'text-size-lg')
   747→  body.classList.add(`text-size-${prefs.fontSize}`)
   748→  // Animations
   749→  body.classList.toggle('no-animations', !prefs.animations)
   750→}

   997→export default function Home() {
   998→  // Lazy-load preferences from localStorage on first render (avoids setState-in-effect)
   999→  const [preferences, setPreferences] = useState<UserPreferences>(() => {
  1000→    if (typeof window === 'undefined') return DEFAULT_PREFERENCES
  1001→    const loaded = loadPreferences()
  1002→    // Apply to body immediately on first client render
  1003→    if (typeof document !== 'undefined') {
  1004→      applyPreferencesToBody(loaded)
  1005→    }
  1006→    return loaded
  1007→  })
  1008→  const [activeSection, setActiveSection] = useState<SectionId>(() => {
  1009→    if (typeof window === 'undefined') return 'dashboard'
  1010→    const loaded = loadPreferences()
  1011→    return loaded.defaultSection ?? 'dashboard'
  1012→  })
  1013→  const [helpOpen, setHelpOpen] = useState(false)
  1014→  const [commandOpen, setCommandOpen] = useState(false)
  1015→  const [settingsOpen, setSettingsOpen] = useState(false)
  1016→  const notifications = mockNotifications
  1017→
  1018→  // Auto-refresh handler
  1019→  useEffect(() => {
  1020→    if (!preferences.autoRefresh) return
  1021→    const interval = setInterval(() => {
  1022→      // Trigger a refetch by dispatching a custom event
  1023→      window.dispatchEvent(new CustomEvent('case-auto-refresh'))
  1024→    }, 30000) // 30 seconds
  1025→    return () => clearInterval(interval)
  1026→  }, [preferences.autoRefresh])
  1027→
  1028→  // Keyboard shortcuts handler
  1029→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
  1030→    // Ctrl+K opens command palette (highest priority)
  1031→    if (e.ctrlKey && (e.key === 'k' || e.key === 'K' || e.key === 'л' || e.key === 'Л')) {
  1032→      e.preventDefault()
  1033→      setCommandOpen(prev => !prev)
  1034→      return
  1035→    }
  1036→    // Ctrl+, (comma) opens settings
  1037→    if (e.ctrlKey && (e.key === ',' || e.key === 'б' || e.key === 'Б')) {
  1038→      e.preventDefault()
  1039→      setSettingsOpen(true)
  1040→      return
  1041→    }
  1042→    // Ctrl+1-8 for section navigation
  1043→    if (e.ctrlKey && !e.altKey && !e.shiftKey) {
  1044→      const num = parseInt(e.key)
  1045→      if (num >= 1 && num <= 8) {
  1046→        e.preventDefault()
  1047→        setActiveSection(NAV_ITEMS[num - 1].id)
  1048→      }
  1049→      // Ctrl+9 for timeline
  1050→      if (e.key === '9') {
  1051→        e.preventDefault()
  1052→        setActiveSection('timeline')
  1053→      }
  1054→      // Ctrl+0 for risk
  1055→      if (e.key === '0') {
  1056→        e.preventDefault()
  1057→        setActiveSection('risk')
  1058→      }
  1059→      // Ctrl+B for brief
  1060→      if (e.key === 'b' || e.key === 'B' || e.key === 'и' || e.key === 'И') {
  1061→        e.preventDefault()
  1062→        setActiveSection('brief')
  1063→      }
  1064→      // Ctrl+A for analytics
  1065→      if (e.key === 'a' || e.key === 'A' || e.key === 'ф' || e.key === 'Ф') {
  1066→        e.preventDefault()
  1067→        setActiveSection('analytics')
  1068→      }
  1069→      // Ctrl+E for evidence chain
  1070→      if (e.key === 'e' || e.key === 'E' || e.key === 'у' || e.key === 'У') {
  1071→        e.preventDefault()
  1072→        setActiveSection('evidence-chain')
  1073→      }
  1074→    }
  1075→    // ? for help
  1076→    if (e.key === '?' && !e.ctrlKey && !e.altKey) {
  1077→      // Only trigger when not in an input/textarea
  1078→      const target = e.target as HTMLElement
  1079→      if (target.tagName !== 'INPUT' && target.tagName !== 'TEXTAREA') {
  1080→        e.preventDefault()
  1081→        setHelpOpen(true)
  1082→      }
  1083→    }
  1084→    // Escape to close help or command palette
  1085→    if (e.key === 'Escape') {
  1086→      if (helpOpen) setHelpOpen(false)
  1087→      if (commandOpen) setCommandOpen(false)
  1088→      if (settingsOpen) setSettingsOpen(false)
  1089→    }
  1090→  }, [helpOpen, commandOpen, settingsOpen])
  1091→
  1092→  useEffect(() => {
  1093→    window.addEventListener('keydown', handleKeyDown)
  1094→    return () => window.removeEventListener('keydown', handleKeyDown)
  1095→  }, [handleKeyDown])
  1096→
  1097→  return (
  1098→    <SidebarProvider>
  1099→      <AppSidebar activeSection={activeSection} setActiveSection={setActiveSection} preferences={preferences} />
  1100→      <SidebarInset>
  1101→        <TopHeader
  1102→          activeSection={activeSection}
  1103→          notifications={notifications}
  1104→          setActiveSection={setActiveSection}
  1105→          onHelpClick={() => setHelpOpen(true)}
  1106→          onCommandClick={() => setCommandOpen(true)}
  1107→          onSettingsClick={() => setSettingsOpen(true)}
  1108→          preferences={preferences}
  1109→        />
  1110→        <div className="flex flex-1 flex-col min-h-0">
  1111→          <main className="flex-1 overflow-y-auto p-4 md:p-6">
  1112→            <MainContent activeSection={activeSection} />
  1113→          </main>
  1114→          <AppFooter />
  1115→        </div>
  1116→      </SidebarInset>
  1117→      <KeyboardShortcutsHelp open={helpOpen} onClose={() => setHelpOpen(false)} />
  1118→      <CommandPalette
  1119→        open={commandOpen}
  1120→        onOpenChange={setCommandOpen}
  1121→        setActiveSection={setActiveSection}
  1122→      />
  1123→      <SettingsDialog
  1124→        open={settingsOpen}
  1125→        onClose={() => setSettingsOpen(false)}
  1126→        preferences={preferences}
  1127→        setPreferences={setPreferences}
  1128→      />
  1129→    </SidebarProvider>
  1130→  )
  1131→}
