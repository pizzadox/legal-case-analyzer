    72→function getContextBadge(contextType: string | null) {
    73→  if (!contextType) return null
    74→  const contextMap: Record<string, { icon: React.ReactNode; label: string; className: string }> = {
    75→    'general': { icon: <Sparkles className="w-3 h-3" />, label: 'Общий', className: 'bg-stone-600 text-white' },
    76→    'article_specific': { icon: <Scale className="w-3 h-3" />, label: 'Статья', className: 'bg-red-700 text-white' },
    77→    'person_specific': { icon: <Users className="w-3 h-3" />, label: 'Участник', className: 'bg-orange-600 text-white' },
    78→    'episode_specific': { icon: <BookOpen className="w-3 h-3" />, label: 'Эпизод', className: 'bg-amber-600 text-white' },
    79→  }
    80→  const info = contextMap[contextType]
    81→  if (!info) return <Badge variant="outline">{contextType}</Badge>
    82→  return <Badge className={info.className}>{info.icon} {info.label}</Badge>

   127→    mutationFn: caseApi.askQuestion,
   128→    onSuccess: (data) => {
   129→      const mockFollowUp = generateMockResponse(currentQuestion)
   130→      const aiMessage: ChatMessageExtended = {
   131→        ...data,
