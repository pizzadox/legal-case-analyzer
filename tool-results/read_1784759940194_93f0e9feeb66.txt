   912→function scoreColor(s: number): string {
   913→  if (s >= 75) return '#991b1b'

   934→function matrixColor(likelihood: number, impact: number): string {
   935→  const sum = Math.ceil(likelihood / 20) + Math.ceil(impact / 20)
   936→  if (sum <= 3) return 'bg-emerald-700/60'
   937→  if (sum <= 5) return 'bg-amber-500/60'
   938→  if (sum <= 7) return 'bg-orange-500/70'
   939→  return 'bg-red-700/80'
   940→}

   947→      <svg width="140" height="140" className="transform -rotate-90">

   949→        <circle cx="70" cy="70" r={r} stroke={color} strokeWidth="8" fill="none" strokeDasharray={c} strokeDashoffset={off} strokeLinecap="round" className="transition-all duration-700" />
   950→      </svg>
   951→      <div className="absolute flex flex-col items-center">
   952→        <span className="text-2xl font-bold" style={{ color }}>{score}</span>
   953→        <span className="text-xs text-muted-foreground">{LEVEL_LABEL[level] ?? level}</span>
   954→      </div>
   955→    </div>
