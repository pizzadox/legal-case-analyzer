   183→function NotificationCenter({ notifications, setActiveSection }: {
   184→  notifications: NotificationData[]
   185→  setActiveSection: (section: SectionId) => void
   186→}) {
   187→  const [localNotifications, setLocalNotifications] = useState(notifications)
   188→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   189→
   190→  const markAllRead = () => {
   191→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   192→    toast.success('Все уведомления отмечены как прочитанные')
   193→  }
   194→
   195→  const handleClick = (notif: NotificationData) => {
   196→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   197→    if (notif.relatedSection) {
   198→      setActiveSection(notif.relatedSection)
   199→    }
   200→  }
   201→
   202→  return (
   203→    <Popover>
   204→      <PopoverTrigger asChild>
   205→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   206→          <Bell className="h-4 w-4" />
   207→          {unreadCount > 0 && (
   208→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   209→              {unreadCount}
   210→            </span>
   211→          )}
   212→          <span className="sr-only">Уведомления</span>
   213→        </Button>
   214→      </PopoverTrigger>
   215→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   216→        <div className="p-3 border-b flex items-center gap-2">
   217→          <Bell className="w-4 h-4 text-muted-foreground" />
   218→          <p className="font-semibold text-sm">Уведомления</p>
   219→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   220→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   221→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   222→          </Button>
   223→        </div>
   224→        <div className="max-h-64 overflow-y-auto">
   225→          {localNotifications.map(notif => (
   226→            <div key={notif.id}
   227→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   228→              onClick={() => handleClick(notif)}>
   229→              <div className="flex items-center gap-2">
   230→                {NOTIF_TYPE_ICON[notif.type] ?? <Bell className="w-4 h-4" />}
   231→                <div className="flex-1 min-w-0">
   232→                  <p className="text-sm font-medium truncate">{notif.title}</p>
   233→                  <p className="text-xs text-muted-foreground truncate">{notif.description}</p>
   234→                </div>
   235→                {!notif.isRead && <div className="w-2 h-2 rounded-full bg-red-700 shrink-0" />}
   236→              </div>
   237→              <p className="text-xs text-muted-foreground mt-1">{new Date(notif.timestamp).toLocaleString('ru-RU', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' })}</p>
   238→            </div>
   239→          ))}
   240→          {localNotifications.length === 0 && (
   241→            <div className="p-6 text-center">
   242→              <Bell className="w-8 h-8 mx-auto text-muted-foreground" />
   243→              <p className="text-sm text-muted-foreground mt-2">Нет уведомлений</p>
   244→            </div>
   245→          )}
   246→        </div>
   247→      </PopoverContent>
   248→    </Popover>
   249→  )
   250→}

   356→function CaseHealthBadge() {
   357→  // Compact health indicator for sidebar footer
   358→  const [score, setScore] = useState<number | null>(null)
   359→  useEffect(() => {
   360→    fetch('/api/case/health-score')
   361→      .then(r => r.ok ? r.json() : null)
   362→      .then(d => {
   363→        if (d && typeof d.score === 'number') setScore(d.score)
   364→        else if (d && typeof d.overallScore === 'number') setScore(d.overallScore)
   365→      })
   366→      .catch(() => setScore(null))
   367→  }, [])
   368→
   369→  if (score === null) {
   370→    return (
   371→      <div className="flex items-center gap-2 px-2 py-1.5 rounded-md bg-muted/50 text-xs text-muted-foreground">
   372→        <Activity className="w-3.5 h-3.5 animate-pulse" />
   373→        <span className="group-data-[collapsible=icon]:hidden">Загрузка...</span>
   374→      </div>
   375→    )
   376→  }
   377→
   378→  // Color thresholds: <50 red, 50-75 amber, >75 emerald
   379→  const color = score >= 75 ? 'emerald' : score >= 50 ? 'amber' : 'red'
   380→  const colorClasses = {
   381→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900',
   382→    amber: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900',
   383→    red: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900',
   384→  }
   385→  const statusText = score >= 75 ? 'Здоровое' : score >= 50 ? 'Среднее' : 'Проблемное'
   386→
   387→  return (
   388→    <div className={`flex items-center gap-2 px-2 py-1.5 rounded-md border text-xs ${colorClasses[color]}`}>
   389→      <Heart className="w-3.5 h-3.5 fill-current" />
   390→      <div className="flex-1 min-w-0 group-data-[collapsible=icon]:hidden">
   391→        <div className="flex items-center gap-1.5">
   392→          <span className="font-semibold">{score}/100</span>
   393→          <span className="opacity-70">·</span>
   394→          <span className="truncate">{statusText}</span>
   395→        </div>
   396→        <div className="mt-1 h-1 rounded-full bg-black/10 dark:bg-white/10 overflow-hidden">
   397→          <div
   398→            className={`h-full rounded-full transition-all duration-700 ${
   399→              color === 'emerald' ? 'bg-emerald-600' : color === 'amber' ? 'bg-amber-600' : 'bg-red-700'
   400→            }`}
   401→            style={{ width: `${score}%` }}
   402→          />
   403→        </div>
   404→      </div>
   405→    </div>
   406→  )
   407→}

   759→function loadPreferences(): UserPreferences {
   760→  if (typeof window === 'undefined') return DEFAULT_PREFERENCES
   761→  try {
   762→    const raw = localStorage.getItem('case-user-preferences')
   763→    if (!raw) return DEFAULT_PREFERENCES
   764→    const parsed = JSON.parse(raw)
   765→    return { ...DEFAULT_PREFERENCES, ...parsed }
   766→  } catch {
   767→    return DEFAULT_PREFERENCES
   768→  }
   769→}

   771→function savePreferences(prefs: UserPreferences) {
   772→  if (typeof window === 'undefined') return
   773→  localStorage.setItem('case-user-preferences', JSON.stringify(prefs))
   774→}

   776→function applyPreferencesToBody(prefs: UserPreferences) {
   777→  if (typeof document === 'undefined') return
   778→  const body = document.body
   779→  // Density
   780→  body.classList.toggle('compact', prefs.density === 'compact')
   781→  // Font size
   782→  body.classList.remove('text-size-sm', 'text-size-md', 'text-size-lg')
   783→  body.classList.add(`text-size-${prefs.fontSize}`)
   784→  // Animations
   785→  body.classList.toggle('no-animations', !prefs.animations)
   786→}

  1033→export default function Home() {
  1034→  // Lazy-load preferences from localStorage on first render (avoids setState-in-effect)
  1035→  const [preferences, setPreferences] = useState<UserPreferences>(() => {
  1036→    if (typeof window === 'undefined') return DEFAULT_PREFERENCES
  1037→    const loaded = loadPreferences()
  1038→    // Apply to body immediately on first client render
  1039→    if (typeof document !== 'undefined') {
  1040→      applyPreferencesToBody(loaded)
  1041→    }
  1042→    return loaded
  1043→  })
  1044→  const [activeSection, setActiveSection] = useState<SectionId>(() => {
  1045→    if (typeof window === 'undefined') return 'dashboard'
  1046→    const loaded = loadPreferences()
  1047→    return loaded.defaultSection ?? 'dashboard'
  1048→  })
  1049→  const [helpOpen, setHelpOpen] = useState(false)
  1050→  const [commandOpen, setCommandOpen] = useState(false)
  1051→  const [settingsOpen, setSettingsOpen] = useState(false)
  1052→  const notifications = mockNotifications
  1053→
  1054→  // Auto-refresh handler
  1055→  useEffect(() => {
  1056→    if (!preferences.autoRefresh) return
  1057→    const interval = setInterval(() => {
  1058→      // Trigger a refetch by dispatching a custom event
  1059→      window.dispatchEvent(new CustomEvent('case-auto-refresh'))
  1060→    }, 30000) // 30 seconds
  1061→    return () => clearInterval(interval)
  1062→  }, [preferences.autoRefresh])
  1063→
  1064→  // Keyboard shortcuts handler
  1065→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
  1066→    // Ctrl+K opens command palette (highest priority)
  1067→    if (e.ctrlKey && (e.key === 'k' || e.key === 'K' || e.key === 'л' || e.key === 'Л')) {
  1068→      e.preventDefault()
  1069→      setCommandOpen(prev => !prev)
  1070→      return
  1071→    }
  1072→    // Ctrl+, (comma) opens settings
  1073→    if (e.ctrlKey && (e.key === ',' || e.key === 'б' || e.key === 'Б')) {
  1074→      e.preventDefault()
  1075→      setSettingsOpen(true)
  1076→      return
  1077→    }
  1078→    // Ctrl+1-8 for section navigation
  1079→    if (e.ctrlKey && !e.altKey && !e.shiftKey) {
  1080→      const num = parseInt(e.key)
  1081→      if (num >= 1 && num <= 8) {
  1082→        e.preventDefault()
  1083→        setActiveSection(NAV_ITEMS[num - 1].id)
  1084→      }
  1085→      // Ctrl+9 for timeline
  1086→      if (e.key === '9') {
  1087→        e.preventDefault()
  1088→        setActiveSection('timeline')
  1089→      }
  1090→      // Ctrl+0 for risk
  1091→      if (e.key === '0') {
  1092→        e.preventDefault()
  1093→        setActiveSection('risk')
  1094→      }
  1095→      // Ctrl+B for brief
  1096→      if (e.key === 'b' || e.key === 'B' || e.key === 'и' || e.key === 'И') {
  1097→        e.preventDefault()
  1098→        setActiveSection('brief')
  1099→      }
  1100→      // Ctrl+A for analytics
  1101→      if (e.key === 'a' || e.key === 'A' || e.key === 'ф' || e.key === 'Ф') {
  1102→        e.preventDefault()
  1103→        setActiveSection('analytics')
  1104→      }
  1105→      // Ctrl+E for evidence chain
  1106→      if (e.key === 'e' || e.key === 'E' || e.key === 'у' || e.key === 'У') {
  1107→        e.preventDefault()
  1108→        setActiveSection('evidence-chain')
  1109→      }
  1110→      // Ctrl+M for witness matrix
  1111→      if (e.key === 'm' || e.key === 'M' || e.key === 'ь' || e.key === 'Ь') {
  1112→        e.preventDefault()
  1113→        setActiveSection('witness-matrix')
  1114→      }
  1115→      // Ctrl+X for export center
  1116→      if (e.key === 'x' || e.key === 'X' || e.key === 'ч' || e.key === 'Ч') {
  1117→        e.preventDefault()
  1118→        setActiveSection('export-center')
  1119→      }
  1120→      // Ctrl+G for battle plan (Боевой план)
  1121→      if (e.key === 'g' || e.key === 'G' || e.key === 'п' || e.key === 'П') {
  1122→        e.preventDefault()
  1123→        setActiveSection('battle-plan')
  1124→      }
  1125→      // Ctrl+V for violations (Нарушения УПК)
  1126→      if (e.key === 'v' || e.key === 'V' || e.key === 'м' || e.key === 'М') {
  1127→        e.preventDefault()
  1128→        setActiveSection('violations')
  1129→      }
  1130→    }
  1131→    // ? for help
  1132→    if (e.key === '?' && !e.ctrlKey && !e.altKey) {
  1133→      // Only trigger when not in an input/textarea
  1134→      const target = e.target as HTMLElement
  1135→      if (target.tagName !== 'INPUT' && target.tagName !== 'TEXTAREA') {
  1136→        e.preventDefault()
  1137→        setHelpOpen(true)
  1138→      }
  1139→    }
  1140→    // Escape to close help or command palette
  1141→    if (e.key === 'Escape') {
  1142→      if (helpOpen) setHelpOpen(false)
  1143→      if (commandOpen) setCommandOpen(false)
  1144→      if (settingsOpen) setSettingsOpen(false)
  1145→    }
  1146→  }, [helpOpen, commandOpen, settingsOpen])
  1147→
  1148→  useEffect(() => {
  1149→    window.addEventListener('keydown', handleKeyDown)
  1150→    return () => window.removeEventListener('keydown', handleKeyDown)
  1151→  }, [handleKeyDown])
  1152→
  1153→  return (
  1154→    <SidebarProvider>
  1155→      <AppSidebar activeSection={activeSection} setActiveSection={setActiveSection} preferences={preferences} />
  1156→      <SidebarInset>
  1157→        <TopHeader
  1158→          activeSection={activeSection}
  1159→          notifications={notifications}
  1160→          setActiveSection={setActiveSection}
  1161→          onHelpClick={() => setHelpOpen(true)}
  1162→          onCommandClick={() => setCommandOpen(true)}
  1163→          onSettingsClick={() => setSettingsOpen(true)}
  1164→          preferences={preferences}
  1165→        />
  1166→        <div className="flex flex-1 flex-col min-h-0">
  1167→          <main className="flex-1 overflow-y-auto p-4 md:p-6">
  1168→            <MainContent activeSection={activeSection} />
  1169→          </main>
  1170→          <AppFooter />
  1171→        </div>
  1172→      </SidebarInset>
  1173→      <KeyboardShortcutsHelp open={helpOpen} onClose={() => setHelpOpen(false)} />
  1174→      <CommandPalette
  1175→        open={commandOpen}
  1176→        onOpenChange={setCommandOpen}
  1177→        setActiveSection={setActiveSection}
  1178→      />
  1179→      <SettingsDialog
  1180→        open={settingsOpen}
  1181→        onClose={() => setSettingsOpen(false)}
  1182→        preferences={preferences}
  1183→        setPreferences={setPreferences}
  1184→      />
