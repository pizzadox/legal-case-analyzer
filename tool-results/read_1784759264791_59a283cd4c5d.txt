    44→// ──── Animated counter hook — smoothly increments from 0 to target value over ~700ms ────
    45→function useAnimatedCounter(target: number, durationMs: number = 700) {
    46→  const [value, setValue] = useState(0)
    47→  const rafRef = useRef<number | null>(null)
    48→  useEffect(() => {
    49→    const start = performance.now()
    50→    const startValue = value
    51→    const delta = target - startValue
    52→    if (delta === 0) return
    53→    const tick = (now: number) => {
    54→      const elapsed = now - start
    55→      const t = Math.min(1, elapsed / durationMs)
    56→      // ease-out cubic
    57→      const eased = 1 - Math.pow(1 - t, 3)
    58→      setValue(Math.round(startValue + delta * eased))
    59→      if (t < 1) {
    60→        rafRef.current = requestAnimationFrame(tick)
    61→      }
    62→    }
    63→    rafRef.current = requestAnimationFrame(tick)
    64→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current) }
    65→  }, [target, durationMs])
    66→  return value
    67→}

    69→// ──── ENHANCED: Speedometer-style Health Gauge with color segments and pulsing ────
    70→function HealthScoreSpeedometer({ score }: { score: number }) {
    71→  const animatedScore = useAnimatedCounter(score, 900)
    72→  const isLow = score < 40
    73→  // Gauge dimensions
    74→  const cx = 80
    75→  const cy = 80
    76→  const outerR = 70
    77→  const innerR = 50
    78→  // Arc spans from -210° to 30° (240° sweep, like a speedometer)
    79→  const startAngle = -210
    80→  const sweepAngle = 240
    81→  // Background arc
    82→  const bgStartRad = (startAngle * Math.PI) / 180
    83→  const bgEndRad = ((startAngle + sweepAngle) * Math.PI) / 180
    84→  const bgStartX = cx + outerR * Math.cos(bgStartRad)
    85→  const bgStartY = cy + outerR * Math.sin(bgStartRad)
    86→  const bgEndX = cx + outerR * Math.cos(bgEndRad)
    87→  const bgEndY = cy + outerR * Math.sin(bgEndRad)
    88→  const bgInnerStartX = cx + innerR * Math.cos(bgStartRad)
    89→  const bgInnerStartY = cy + innerR * Math.sin(bgStartRad)
    90→  const bgInnerEndX = cx + innerR * Math.cos(bgEndRad)
    91→  const bgInnerEndY = cy + innerR * Math.sin(bgEndRad)
    92→  const bgLargeArc = sweepAngle > 180 ? 1 : 0
    93→
    94→  // Color segments: red 0-40, amber 40-60, emerald 60-100
    95→  const segments = [
    96→    { start: 0, end: 40, color: '#dc2626' },
    97→    { start: 40, end: 60, color: '#d97706' },
    98→    { start: 60, end: 100, color: '#059669' },
    99→  ]
   100→
   101→  function arcPath(segStart: number, segEnd: number) {
   102→    const sa = startAngle + (segStart / 100) * sweepAngle
   103→    const ea = startAngle + (segEnd / 100) * sweepAngle
   104→    const saRad = (sa * Math.PI) / 180
   105→    const eaRad = (ea * Math.PI) / 180
   106→    const sOuterX = cx + outerR * Math.cos(saRad)
   107→    const sOuterY = cy + outerR * Math.sin(saRad)
   108→    const eOuterX = cx + outerR * Math.cos(eaRad)
   109→    const eOuterY = cy + outerR * Math.sin(eaRad)
   110→    const sInnerX = cx + innerR * Math.cos(saRad)
   111→    const sInnerY = cy + innerR * Math.sin(saRad)
   112→    const eInnerX = cx + innerR * Math.cos(eaRad)
   113→    const eInnerY = cy + innerR * Math.sin(eaRad)
   114→    const arcSweep = ea - sa
   115→    const largeArc = arcSweep > 180 ? 1 : 0
   116→    return `M ${sOuterX} ${sOuterY} A ${outerR} ${outerR} 0 ${largeArc} 1 ${eOuterX} ${eOuterY} L ${eInnerX} ${eInnerY} A ${innerR} ${innerR} 0 ${largeArc} 0 ${sInnerX} ${sInnerY} Z`
   117→  }
   118→
   119→  // Needle angle
   120→  const needleAngle = startAngle + (animatedScore / 100) * sweepAngle
   121→  const needleRad = (needleAngle * Math.PI) / 180
   122→  const needleLen = 42
   123→  const needleX = cx + needleLen * Math.cos(needleRad)
   124→  const needleY = cy + needleLen * Math.sin(needleRad)
   125→
   126→  // Current color based on score
   127→  const currentColor = score >= 60 ? '#059669' : score >= 40 ? '#d97706' : '#dc2626'
   128→
   129→  return (
   130→    <div className={`flex items-center justify-center ${isLow ? 'animate-pulse' : ''}`}>
   131→      <svg width="160" height="120" viewBox="0 0 160 120">
   132→        {/* Background arc */}
   133→        <path
   134→          d={`M ${bgStartX} ${bgStartY} A ${outerR} ${outerR} 0 ${bgLargeArc} 1 ${bgEndX} ${bgEndY} L ${bgInnerEndX} ${bgInnerEndY} A ${innerR} ${innerR} 0 ${bgLargeArc} 0 ${bgInnerStartX} ${bgInnerStartY} Z`}
   135→          fill="#e7e5e4"
   136→          className="dark:fill-stone-700"
   137→        />
   138→        {/* Color segments */}
   139→        {segments.map(seg => (
   140→          <path
   141→            key={seg.color}
   142→            d={arcPath(seg.start, seg.end)}
   143→            fill={seg.color}
   144→            opacity={0.25}
   145→          />
   146→        ))}
   147→        {/* Active segment up to score */}
   148→        <path
   149→          d={arcPath(0, animatedScore)}
   150→          fill={currentColor}
   151→          opacity={0.8}
   152→          className="transition-all duration-700"
   153→        />
   154→        {/* Needle */}
   155→        <line
   156→          x1={cx}
   157→          y1={cy}
   158→          x2={needleX}
   159→          y2={needleY}
   160→          stroke={currentColor}
   161→          strokeWidth="3"
   162→          strokeLinecap="round"
   163→          className="transition-all duration-700"
   164→        />
   165→        {/* Center circle */}
   166→        <circle cx={cx} cy={cy} r="6" fill={currentColor} className="transition-all duration-700" />
   167→        {/* Score text */}
   168→        <text x={cx} y={cy + 28} textAnchor="middle" className="text-sm font-bold fill-foreground" style={{ fontSize: '14px' }}>
   169→          {animatedScore}
   170→        </text>
   171→        <text x={cx} y={cy + 40} textAnchor="middle" className="fill-muted-foreground" style={{ fontSize: '10px' }}>
   172→          из 100
   173→        </text>
   174→        {/* Scale labels */}
   175→        <text x={bgStartX - 2} y={bgStartY + 4} textAnchor="end" className="fill-muted-foreground" style={{ fontSize: '9px' }}>0</text>
   176→        <text x={bgEndX + 2} y={bgEndY + 4} textAnchor="start" className="fill-muted-foreground" style={{ fontSize: '9px' }}>100</text>
   177→        {/* Color labels at midpoints */}
   178→        <text x={cx - 52} y={cy + 52} textAnchor="middle" className="fill-red-600" style={{ fontSize: '8px', fontWeight: 'bold' }}>Крит.</text>
   179→        <text x={cx - 8} y={cy + 52} textAnchor="middle" className="fill-amber-600" style={{ fontSize: '8px', fontWeight: 'bold' }}>Умер.</text>
   180→        <text x={cx + 42} y={cy + 52} textAnchor="middle" className="fill-emerald-600" style={{ fontSize: '8px', fontWeight: 'bold' }}>Хорошо</text>
   181→      </svg>
   182→    </div>
   183→  )
   184→}

   773→  const overdue = sorted.filter(d => new Date(d.deadline).getTime() <= now.getTime())
   774→  const nextDeadline = upcoming[0]
   775→  const daysToNext = nextDeadline
   776→    ? Math.ceil((new Date(nextDeadline.deadline).getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   777→    : 0
   778→
   779→  const fmtDate = (iso: string) => new Date(iso).toLocaleDateString('ru-RU', { day: '2-digit', month: 'long', year: 'numeric' })
   780→  const daysTo = (iso: string) => {
   781→    const diff = Math.ceil((new Date(iso).getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   782→    if (diff < 0) return `просрочено на ${Math.abs(diff)} дн.`
   783→    if (diff === 0) return 'сегодня'
   784→    if (diff === 1) return 'завтра'
   785→    return `через ${diff} дн.`
   786→  }
   787→
   788→  return (

   821→                    <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-red-400 opacity-75" />
   822→                    <span className="relative inline-flex rounded-full h-3 w-3 bg-red-600" />
   823→                  </span>
