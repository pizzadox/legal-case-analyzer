    23→function personLabel(p: { shortName?: string | null; fullName?: string | null; person?: { shortName?: string | null; fullName?: string | null } }): string {
    24→  if (p.person) return p.person.shortName ?? p.person.fullName ?? '—'
    25→  return (p.shortName ?? p.fullName ?? '—') as string
    26→}
    27→function articleCode(a: { code?: string; article?: { code?: string } }): string {
    28→  return (a.article?.code ?? a.code ?? '—') as string
    29→}
    30→function locationName(l: { name?: string | null; location?: { name?: string | null } }): string {
    31→  return (l.location?.name ?? l.name ?? '—') as string
    32→}
    33→function locationAddress(l: { address?: string | null; location?: { address?: string | null } }): string {
    34→  return (l.location?.address ?? l.address ?? '—') as string
    35→}

    37→function exportEpisodesCSV(episodes: EpisodeData[]) {
    38→  const rows = ['Title,Severity,Status,Date,Persons,Articles']
    39→  episodes.forEach(e => {
    40→    const persons = e.persons.map(p => `${personLabel(p)} (${p.involvement ?? ''})`).join('; ')
    41→    const articles = e.articles.map(a => articleCode(a)).join('; ')
    42→    rows.push(`"${e.title}",${e.severity ?? ''},${e.status ?? ''},${e.date ?? ''},"${persons}","${articles}"`)
    43→  })
    44→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
    45→  const url = URL.createObjectURL(blob)
    46→  const a = document.createElement('a')
    47→  a.href = url
    48→  a.download = 'episodes.csv'
    49→  a.click()
    50→  URL.revokeObjectURL(url)
    51→  toast.success('CSV экспорт выполнен')
    52→}

   116→// Heat map cell color based on count intensity
   117→function heatColor(count: number, max: number): string {
   118→  if (max === 0) return 'bg-stone-50 text-stone-400'
   119→  const ratio = count / max
   120→  if (ratio === 0) return 'bg-stone-50 text-stone-400'
   121→  if (ratio <= 0.25) return 'bg-red-100 text-red-700'
   122→  if (ratio <= 0.5) return 'bg-red-200 text-red-800'
   123→  if (ratio <= 0.75) return 'bg-red-400 text-white'
   124→  return 'bg-red-700 text-white'
   125→}
   126→
   127→// Severity score calculation
   128→function severityScore(severity: string | null): number {
   129→  return SEVERITY_ORDER[severity ?? ''] ?? 0
   130→}
