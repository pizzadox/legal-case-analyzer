   182→function NotificationCenter({ notifications, setActiveSection }: {
   183→  notifications: NotificationData[]
   184→  setActiveSection: (section: SectionId) => void
   185→}) {
   186→  const [localNotifications, setLocalNotifications] = useState(notifications)
   187→  const unreadCount = localNotifications.filter(n => !n.isRead).length
   188→
   189→  const markAllRead = () => {
   190→    setLocalNotifications(localNotifications.map(n => ({ ...n, isRead: true })))
   191→    toast.success('Все уведомления отмечены как прочитанные')
   192→  }
   193→
   194→  const handleClick = (notif: NotificationData) => {
   195→    setLocalNotifications(localNotifications.map(n => n.id === notif.id ? { ...n, isRead: true } : n))
   196→    if (notif.relatedSection) {
   197→      setActiveSection(notif.relatedSection)
   198→    }
   199→  }
   200→
   201→  return (
   202→    <Popover>
   203→      <PopoverTrigger asChild>
   204→        <Button variant="ghost" size="icon" className="h-8 w-8 relative">
   205→          <Bell className="h-4 w-4" />
   206→          {unreadCount > 0 && (
   207→            <span className="absolute -top-1 -right-1 flex items-center justify-center w-4 h-4 rounded-full bg-red-700 text-white text-xs font-bold">
   208→              {unreadCount}
   209→            </span>
   210→          )}
   211→          <span className="sr-only">Уведомления</span>
   212→        </Button>
   213→      </PopoverTrigger>
   214→      <PopoverContent align="end" className="w-80 p-0 rounded-xl">
   215→        <div className="p-3 border-b flex items-center gap-2">
   216→          <Bell className="w-4 h-4 text-muted-foreground" />
   217→          <p className="font-semibold text-sm">Уведомления</p>
   218→          <Badge className="bg-stone-600 text-white text-xs">{unreadCount} новых</Badge>
   219→          <Button size="sm" variant="ghost" className="ml-auto text-xs" onClick={markAllRead}>
   220→            <CheckCircle className="w-3 h-3 mr-1" />Прочитать все
   221→          </Button>
   222→        </div>
   223→        <div className="max-h-64 overflow-y-auto">
   224→          {localNotifications.map(notif => (
   225→            <div key={notif.id}
   226→              className={`p-3 border-b last:border-b-0 cursor-pointer transition-colors hover:bg-muted ${!notif.isRead ? 'bg-muted/30' : ''}`}
   227→              onClick={() => handleClick(notif)}>
   228→              <div className="flex items-center gap-2">
   229→                {NOTIF_TYPE_ICON[notif.type] ?? <Bell className="w-4 h-4" />}
   230→                <div className="flex-1 min-w-0">
   231→                  <p className="text-sm font-medium truncate">{notif.title}</p>
   232→                  <p className="text-xs text-muted-foreground truncate">{notif.description}</p>
   233→                </div>
   234→                {!notif.isRead && <div className="w-2 h-2 rounded-full bg-red-700 shrink-0" />}
   235→              </div>
   236→              <p className="text-xs text-muted-foreground mt-1">{new Date(notif.timestamp).toLocaleString('ru-RU', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' })}</p>
   237→            </div>
   238→          ))}
   239→          {localNotifications.length === 0 && (
   240→            <div className="p-6 text-center">
   241→              <Bell className="w-8 h-8 mx-auto text-muted-foreground" />
   242→              <p className="text-sm text-muted-foreground mt-2">Нет уведомлений</p>
   243→            </div>
   244→          )}
   245→        </div>
   246→      </PopoverContent>
   247→    </Popover>
   248→  )
   249→}

   355→function CaseHealthBadge() {
   356→  // Compact health indicator for sidebar footer
   357→  const [score, setScore] = useState<number | null>(null)
   358→  useEffect(() => {
   359→    fetch('/api/case/health-score')
   360→      .then(r => r.ok ? r.json() : null)
   361→      .then(d => {
   362→        if (d && typeof d.score === 'number') setScore(d.score)
   363→        else if (d && typeof d.overallScore === 'number') setScore(d.overallScore)
   364→      })
   365→      .catch(() => setScore(null))
   366→  }, [])
   367→
   368→  if (score === null) {
   369→    return (
   370→      <div className="flex items-center gap-2 px-2 py-1.5 rounded-md bg-muted/50 text-xs text-muted-foreground">
   371→        <Activity className="w-3.5 h-3.5 animate-pulse" />
   372→        <span className="group-data-[collapsible=icon]:hidden">Загрузка...</span>
   373→      </div>
   374→    )
   375→  }
   376→
   377→  // Color thresholds: <50 red, 50-75 amber, >75 emerald
   378→  const color = score >= 75 ? 'emerald' : score >= 50 ? 'amber' : 'red'
   379→  const colorClasses = {
   380→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900',
   381→    amber: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900',
   382→    red: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900',
   383→  }
   384→  const statusText = score >= 75 ? 'Здоровое' : score >= 50 ? 'Среднее' : 'Проблемное'
   385→
   386→  return (
   387→    <div className={`flex items-center gap-2 px-2 py-1.5 rounded-md border text-xs ${colorClasses[color]}`}>
   388→      <Heart className="w-3.5 h-3.5 fill-current" />
   389→      <div className="flex-1 min-w-0 group-data-[collapsible=icon]:hidden">
   390→        <div className="flex items-center gap-1.5">
   391→          <span className="font-semibold">{score}/100</span>
   392→          <span className="opacity-70">·</span>
   393→          <span className="truncate">{statusText}</span>
   394→        </div>
   395→        <div className="mt-1 h-1 rounded-full bg-black/10 dark:bg-white/10 overflow-hidden">
   396→          <div
   397→            className={`h-full rounded-full transition-all duration-700 ${
   398→              color === 'emerald' ? 'bg-emerald-600' : color === 'amber' ? 'bg-amber-600' : 'bg-red-700'
   399→            }`}
   400→            style={{ width: `${score}%` }}
   401→          />
   402→        </div>
   403→      </div>
   404→    </div>
   405→  )
   406→}

   736→function loadPreferences(): UserPreferences {
   737→  if (typeof window === 'undefined') return DEFAULT_PREFERENCES
   738→  try {
   739→    const raw = localStorage.getItem('case-user-preferences')
   740→    if (!raw) return DEFAULT_PREFERENCES
   741→    const parsed = JSON.parse(raw)
   742→    return { ...DEFAULT_PREFERENCES, ...parsed }
   743→  } catch {
   744→    return DEFAULT_PREFERENCES
   745→  }
   746→}

   748→function savePreferences(prefs: UserPreferences) {
   749→  if (typeof window === 'undefined') return
   750→  localStorage.setItem('case-user-preferences', JSON.stringify(prefs))
   751→}

   753→function applyPreferencesToBody(prefs: UserPreferences) {
   754→  if (typeof document === 'undefined') return
   755→  const body = document.body
   756→  // Density
   757→  body.classList.toggle('compact', prefs.density === 'compact')
   758→  // Font size
   759→  body.classList.remove('text-size-sm', 'text-size-md', 'text-size-lg')
   760→  body.classList.add(`text-size-${prefs.fontSize}`)
   761→  // Animations
   762→  body.classList.toggle('no-animations', !prefs.animations)
   763→}

  1010→export default function Home() {
  1011→  // Lazy-load preferences from localStorage on first render (avoids setState-in-effect)
  1012→  const [preferences, setPreferences] = useState<UserPreferences>(() => {
  1013→    if (typeof window === 'undefined') return DEFAULT_PREFERENCES
  1014→    const loaded = loadPreferences()
  1015→    // Apply to body immediately on first client render
  1016→    if (typeof document !== 'undefined') {
  1017→      applyPreferencesToBody(loaded)
  1018→    }
  1019→    return loaded
  1020→  })
  1021→  const [activeSection, setActiveSection] = useState<SectionId>(() => {
  1022→    if (typeof window === 'undefined') return 'dashboard'
  1023→    const loaded = loadPreferences()
  1024→    return loaded.defaultSection ?? 'dashboard'
  1025→  })
  1026→  const [helpOpen, setHelpOpen] = useState(false)
  1027→  const [commandOpen, setCommandOpen] = useState(false)
  1028→  const [settingsOpen, setSettingsOpen] = useState(false)
  1029→  const notifications = mockNotifications
  1030→
  1031→  // Auto-refresh handler
  1032→  useEffect(() => {
  1033→    if (!preferences.autoRefresh) return
  1034→    const interval = setInterval(() => {
  1035→      // Trigger a refetch by dispatching a custom event
  1036→      window.dispatchEvent(new CustomEvent('case-auto-refresh'))
  1037→    }, 30000) // 30 seconds
  1038→    return () => clearInterval(interval)
  1039→  }, [preferences.autoRefresh])
  1040→
  1041→  // Keyboard shortcuts handler
  1042→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
  1043→    // Ctrl+K opens command palette (highest priority)
  1044→    if (e.ctrlKey && (e.key === 'k' || e.key === 'K' || e.key === 'л' || e.key === 'Л')) {
  1045→      e.preventDefault()
  1046→      setCommandOpen(prev => !prev)
  1047→      return
  1048→    }
  1049→    // Ctrl+, (comma) opens settings
  1050→    if (e.ctrlKey && (e.key === ',' || e.key === 'б' || e.key === 'Б')) {
  1051→      e.preventDefault()
  1052→      setSettingsOpen(true)
  1053→      return
  1054→    }
  1055→    // Ctrl+1-8 for section navigation
  1056→    if (e.ctrlKey && !e.altKey && !e.shiftKey) {
  1057→      const num = parseInt(e.key)
  1058→      if (num >= 1 && num <= 8) {
  1059→        e.preventDefault()
  1060→        setActiveSection(NAV_ITEMS[num - 1].id)
  1061→      }
  1062→      // Ctrl+9 for timeline
  1063→      if (e.key === '9') {
  1064→        e.preventDefault()
  1065→        setActiveSection('timeline')
  1066→      }
  1067→      // Ctrl+0 for risk
  1068→      if (e.key === '0') {
  1069→        e.preventDefault()
  1070→        setActiveSection('risk')
  1071→      }
  1072→      // Ctrl+B for brief
  1073→      if (e.key === 'b' || e.key === 'B' || e.key === 'и' || e.key === 'И') {
  1074→        e.preventDefault()
  1075→        setActiveSection('brief')
  1076→      }
  1077→      // Ctrl+A for analytics
  1078→      if (e.key === 'a' || e.key === 'A' || e.key === 'ф' || e.key === 'Ф') {
  1079→        e.preventDefault()
  1080→        setActiveSection('analytics')
  1081→      }
  1082→      // Ctrl+E for evidence chain
  1083→      if (e.key === 'e' || e.key === 'E' || e.key === 'у' || e.key === 'У') {
  1084→        e.preventDefault()
  1085→        setActiveSection('evidence-chain')
  1086→      }
  1087→      // Ctrl+M for witness matrix
  1088→      if (e.key === 'm' || e.key === 'M' || e.key === 'ь' || e.key === 'Ь') {
  1089→        e.preventDefault()
  1090→        setActiveSection('witness-matrix')
  1091→      }
  1092→      // Ctrl+X for export center
  1093→      if (e.key === 'x' || e.key === 'X' || e.key === 'ч' || e.key === 'Ч') {
  1094→        e.preventDefault()
  1095→        setActiveSection('export-center')
  1096→      }
  1097→      // Ctrl+G for battle plan (Боевой план)
  1098→      if (e.key === 'g' || e.key === 'G' || e.key === 'п' || e.key === 'П') {
  1099→        e.preventDefault()
  1100→        setActiveSection('battle-plan')
  1101→      }
  1102→      // Ctrl+V for violations (Нарушения УПК)
  1103→      if (e.key === 'v' || e.key === 'V' || e.key === 'м' || e.key === 'М') {
  1104→        e.preventDefault()
  1105→        setActiveSection('violations')
  1106→      }
  1107→    }
  1108→    // ? for help
  1109→    if (e.key === '?' && !e.ctrlKey && !e.altKey) {
  1110→      // Only trigger when not in an input/textarea
  1111→      const target = e.target as HTMLElement
  1112→      if (target.tagName !== 'INPUT' && target.tagName !== 'TEXTAREA') {
  1113→        e.preventDefault()
  1114→        setHelpOpen(true)
  1115→      }
  1116→    }
  1117→    // Escape to close help or command palette
  1118→    if (e.key === 'Escape') {
  1119→      if (helpOpen) setHelpOpen(false)
  1120→      if (commandOpen) setCommandOpen(false)
  1121→      if (settingsOpen) setSettingsOpen(false)
  1122→    }
  1123→  }, [helpOpen, commandOpen, settingsOpen])
  1124→
  1125→  useEffect(() => {
  1126→    window.addEventListener('keydown', handleKeyDown)
  1127→    return () => window.removeEventListener('keydown', handleKeyDown)
  1128→  }, [handleKeyDown])
  1129→
  1130→  return (
  1131→    <SidebarProvider>
  1132→      <AppSidebar activeSection={activeSection} setActiveSection={setActiveSection} preferences={preferences} />
  1133→      <SidebarInset>
  1134→        <TopHeader
  1135→          activeSection={activeSection}
  1136→          notifications={notifications}
  1137→          setActiveSection={setActiveSection}
  1138→          onHelpClick={() => setHelpOpen(true)}
  1139→          onCommandClick={() => setCommandOpen(true)}
  1140→          onSettingsClick={() => setSettingsOpen(true)}
  1141→          preferences={preferences}
  1142→        />
  1143→        <div className="flex flex-1 flex-col min-h-0">
  1144→          <main className="flex-1 overflow-y-auto p-4 md:p-6">
  1145→            <MainContent activeSection={activeSection} />
  1146→          </main>
  1147→          <AppFooter />
  1148→        </div>
  1149→      </SidebarInset>
  1150→      <KeyboardShortcutsHelp open={helpOpen} onClose={() => setHelpOpen(false)} />
  1151→      <CommandPalette
  1152→        open={commandOpen}
  1153→        onOpenChange={setCommandOpen}
  1154→        setActiveSection={setActiveSection}
  1155→      />
  1156→      <SettingsDialog
  1157→        open={settingsOpen}
  1158→        onClose={() => setSettingsOpen(false)}
  1159→        preferences={preferences}
  1160→        setPreferences={setPreferences}
  1161→      />
  1162→    </SidebarProvider>
  1163→  )
  1164→}
