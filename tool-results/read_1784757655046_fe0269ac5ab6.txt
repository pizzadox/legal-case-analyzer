   630→function ConflictsSummary({
   631→  witnesses,
   632→  facts,
   633→}: {
   634→  witnesses: Witness[]
   635→  facts: Fact[]
   636→}) {
   637→  const totalConflicts = useMemo(() => {
   638→    return witnesses.reduce((sum, w) => sum + countContradictions(w), 0)
   639→  }, [witnesses])
   640→
   641→  const worstOffender = useMemo(() => {
   642→    return [...witnesses].sort((a, b) => {
   643→      const ca = countContradictions(a)
   644→      const cb = countContradictions(b)
   645→      if (cb !== ca) return cb - ca
   646→      return a.name.localeCompare(b.name, 'ru')
   647→    })[0]
   648→  }, [witnesses])
   649→
   650→  const mostDisputedFact = useMemo(() => {
   651→    let best: { fact: Fact; count: number } | null = null
   652→    for (const f of facts) {
   653→      const count = witnesses.filter(
   654→        w => w.positions[f.id]?.contradictsOthers,
   655→      ).length
   656→      if (!best || count > best.count) {
   657→        best = { fact: f, count }
   658→      }
   659→    }
   660→    return best
   661→  }, [witnesses, facts])
   662→
   663→  const stats = [
   664→    {
   665→      icon: <Flame className="w-5 h-5 text-red-700" />,
   666→      bgIcon: 'bg-red-100 dark:bg-red-950/40',
   667→      label: 'Всего противоречий',
   668→      value: totalConflicts,
   669→      hint: 'Сумма конфликтующих ячеек',
   670→      tone: 'text-red-700',
   671→    },
   672→    {
   673→      icon: <TrendingDown className="w-5 h-5 text-orange-600" />,
   674→      bgIcon: 'bg-orange-100 dark:bg-orange-950/40',
   675→      label: 'Худший свидетель',
   676→      value: worstOffender?.name ?? '—',
   677→      hint: `${worstOffender ? countContradictions(worstOffender) : 0} противоречий`,
   678→      tone: 'text-orange-700',
   679→    },
   680→    {
   681→      icon: <AlertTriangle className="w-5 h-5 text-amber-600" />,
   682→      bgIcon: 'bg-amber-100 dark:bg-amber-950/40',
   683→      label: 'Самый спорный факт',
   684→      value: mostDisputedFact?.fact.text ?? '—',
   685→      hint: `${mostDisputedFact?.count ?? 0} свидетелей в конфликте`,
   686→      tone: 'text-amber-700',
   687→    },
   688→  ]
   689→
   690→  return (
   691→    <Card className="rounded-xl shadow-sm border-l-4 border-red-700 hover:shadow-md hover:-translate-y-0.5 transition-all">
   692→      <CardHeader className="pb-2">
   693→        <CardTitle className="text-sm flex items-center gap-2">
   694→          <AlertTriangle className="w-4 h-4 text-red-700" />
   695→          Сводка противоречий
   696→        </CardTitle>
   697→      </CardHeader>
   698→      <CardContent className="p-4">
   699→        <div className="grid sm:grid-cols-3 gap-3">
   700→          {stats.map((s, i) => (
   701→            <div
   702→              key={i}
   703→              className="flex items-start gap-3 p-3 rounded-lg bg-muted/40 hover:bg-muted/70 transition-colors"
   704→            >
   705→              <div
   706→                className={`flex items-center justify-center w-9 h-9 rounded-lg shrink-0 ${s.bgIcon}`}
   707→              >
   708→                {s.icon}
   709→              </div>
   710→              <div className="min-w-0 flex-1">
   711→                <p className="text-xs text-muted-foreground">{s.label}</p>
   712→                <p className={`font-bold break-words leading-snug ${s.tone}`} title={String(s.value)}>
   713→                  {s.value}
   714→                </p>
   715→                <p className="text-xs text-muted-foreground mt-0.5">{s.hint}</p>
   716→              </div>
   717→            </div>
   718→          ))}
   719→        </div>
   720→      </CardContent>
   721→    </Card>
   722→  )
   723→}

   741→function MatrixCell({ witness, fact, isSelected, conflictsOnly, onClick }: CellProps) {
   742→  const pos = witness.positions[fact.id]
   743→  if (!pos) return null
   744→  const cfg = POSITION_CONFIG[pos.position]
   745→  const isContradiction = pos.contradictsOthers
   746→
   747→  // Determine if cell should be dimmed (when filter is on, only show conflicts highlighted)
   748→  const dimmed = conflictsOnly && !isContradiction
   749→
   750→  // Background class
   751→  let bgClass = cfg.cellClass
   752→  if (dimmed) {
   753→    bgClass = 'bg-stone-50 text-stone-400 dark:bg-stone-900 dark:text-stone-600'
   754→  } else if (isContradiction && pos.position !== 'no-data') {
   755→    // For contradictions, use red-700 background regardless of position
   756→    bgClass = 'bg-red-700 text-white'
   757→  }
   758→
   759→  const ringClass = isSelected
   760→    ? 'ring-2 ring-purple-700 ring-offset-1'
   761→    : isContradiction && !dimmed
   762→      ? 'ring-1 ring-red-900/40'
   763→      : ''
   764→
   765→  const pulseClass =
   766→    isContradiction && !dimmed && pos.position === 'deny'
   767→      ? 'animate-pulse'
   768→      : ''
   769→
   770→  return (
   771→    <button
   772→      type="button"
   773→      onClick={onClick}
   774→      title={`${witness.name} — ${fact.text}: ${cfg.label}${
   775→        isContradiction ? ' (противоречит другим)' : ''
   776→      }`}
   777→      className={`relative h-12 min-w-[64px] flex items-center justify-center transition-all duration-150 ${bgClass} ${ringClass} ${pulseClass} hover:brightness-110 hover:z-10 cursor-pointer`}
   778→    >
   779→      <CellIcon position={pos.position} conflictsOnly={conflictsOnly} />
   780→      {isContradiction && !dimmed && pos.position !== 'no-data' && (
   781→        <span className="absolute top-0.5 right-0.5 w-1.5 h-1.5 rounded-full bg-white/90 shadow-sm" />
   782→      )}
   783→      {isSelected && (
   784→        <span className="absolute inset-0 ring-2 ring-purple-700 rounded-sm pointer-events-none" />
   785→      )}
   786→    </button>
   787→  )
   788→}

  1128→function AgreementAnalysis({ witnesses, facts }: { witnesses: Witness[]; facts: Fact[] }) {
  1129→  const analysis = useMemo(() => {
  1130→    return facts.map(f => {
  1131→      const counts = { confirm: 0, deny: 0, 'dont-remember': 0, 'no-data': 0 }
  1132→      for (const w of witnesses) {
  1133→        const p = w.positions[f.id]?.position
  1134→        if (p) counts[p]++
  1135→      }
  1136→      const total = counts.confirm + counts.deny + counts['dont-remember']
  1137→      const consensus =
  1138→        total > 0
  1139→          ? Math.max(counts.confirm, counts.deny, counts['dont-remember']) / total
  1140→          : 0
  1141→      return { fact: f, counts, consensus, total }
  1142→    })
  1143→  }, [witnesses, facts])
  1144→
  1145→  return (
  1146→    <Card className="rounded-xl shadow-sm hover:shadow-md transition-shadow h-full">
  1147→      <CardHeader className="pb-2">
  1148→        <CardTitle className="text-sm flex items-center gap-2">

  1151→        </CardTitle>

  1154→        {analysis.map(a => {
  1155→          const maxCount = Math.max(
  1156→            a.counts.confirm,
  1157→            a.counts.deny,

  1159→            a.counts['no-data'],
