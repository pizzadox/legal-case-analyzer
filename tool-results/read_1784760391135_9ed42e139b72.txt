   124→// ─── CIRCULAR GAUGE SVG ────────────────────────────────────────────
   125→function CircularGauge({ value, size = 120 }: { value: number; size?: number }) {
   126→  const radius = (size - 16) / 2
   127→  const circumference = 2 * Math.PI * radius
   128→  const offset = circumference - (value / 100) * circumference
   129→  const color = value >= 70 ? '#059669' : value >= 40 ? '#d97706' : '#b91c1c'
   130→  return (
   131→    <svg width={size} height={size} className="shrink-0">
   132→      <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth="8" className="text-stone-200 dark:text-stone-700" />
   133→      <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth="8" strokeLinecap="round" strokeDasharray={circumference} strokeDashoffset={offset} transform={`rotate(-90 ${size / 2} ${size / 2})`} style={{ transition: 'stroke-dashoffset 0.8s ease' }} />
   134→      <text x={size / 2} y={size / 2 - 6} textAnchor="middle" className="fill-foreground font-bold" fontSize="24">{value}%</text>
   135→      <text x={size / 2} y={size / 2 + 14} textAnchor="middle" className="fill-muted-foreground" fontSize="10">соответствие</text>
   136→    </svg>
   137→  )
   138→}

   140→// ─── COMPLIANCE MATRIX HEATMAP ─────────────────────────────────────
   141→function ComplianceMatrixHeatmap({ compliance, documents }: { compliance: LegalComplianceData[]; documents: { id: string; originalName: string }[] }) {
   142→  // Build a map: docId → checkType → worst status
   143→  const matrix = useMemo(() => {
   144→    const map: Record<string, Record<string, string>> = {}
   145→    compliance.forEach(c => {
   146→      if (!map[c.documentId]) map[c.documentId] = {}
   147→      // Take the worst status for each cell
   148→      const existing = map[c.documentId][c.checkType]
   149→      if (!existing || severityRank(c.status) > severityRank(existing)) {
   150→        map[c.documentId][c.checkType] = c.status
   151→      }
   152→    })
   153→    return map
   154→  }, [compliance])
   155→
   156→  function severityRank(s: string): number {
   157→    if (s === 'violation') return 3
   158→    if (s === 'warning') return 2
   159→    if (s === 'needs_review') return 1
   160→    return 0
   161→  }
   162→
   163→  function cellColor(status: string | undefined): string {
   164→    if (!status) return 'bg-stone-100 dark:bg-stone-800'
   165→    if (status === 'violation') return 'bg-red-500/80 hover:bg-red-500'
   166→    if (status === 'warning') return 'bg-amber-400/80 hover:bg-amber-400'
   167→    if (status === 'compliant') return 'bg-emerald-500/80 hover:bg-emerald-500'
   168→    if (status === 'needs_review') return 'bg-stone-300/80 hover:bg-stone-300 dark:bg-stone-600/80'
   169→    return 'bg-stone-200'
   170→  }
   171→
   172→  function cellIcon(status: string | undefined): React.ReactNode {
   173→    if (!status) return <span className="text-stone-400 text-xs">—</span>
   174→    if (status === 'violation') return <XCircle className="w-3.5 h-3.5 text-white" />
   175→    if (status === 'warning') return <AlertTriangle className="w-3.5 h-3.5 text-white" />
   176→    if (status === 'compliant') return <CheckCircle className="w-3.5 h-3.5 text-white" />
   177→    if (status === 'needs_review') return <Info className="w-3.5 h-3.5 text-stone-600 dark:text-stone-300" />
   178→    return null
   179→  }
   180→
   181→  const docIds = documents.map(d => d.id)
   182→  const uniqueDocIds = [...new Set(compliance.map(c => c.documentId).concat(docIds))].slice(0, 5)
   183→
   184→  return (
   185→    <Card className="rounded-xl shadow-sm border-l-4 border-stone-500">
   186→      <CardHeader className="pb-2">
   187→        <CardTitle className="text-sm flex items-center gap-2">
   188→          <BarChart3 className="w-4 h-4 text-amber-600" />
   189→          Матрица соответствия
   190→          <Badge variant="outline" className="text-xs">{uniqueDocIds.length} док. × {HEATMAP_COLUMNS.length} типов</Badge>
   191→        </CardTitle>
   192→      </CardHeader>
   193→      <CardContent className="p-4">
   194→        <div className="overflow-x-auto">
   195→          <table className="w-full">
   196→            <thead>
   197→              <tr>
   198→                <th className="text-left text-xs font-medium text-muted-foreground pb-2 pr-3 min-w-[140px]">Документ</th>
   199→                {HEATMAP_COLUMNS.map(ct => (
   200→                  <th key={ct} className="text-center text-xs font-medium text-muted-foreground pb-2 px-2">
   201→                    <div className="flex items-center justify-center gap-1">
   202→                      {CHECK_TYPES[ct]?.icon}
   203→                      <span>{CHECK_TYPES[ct]?.shortLabel ?? ct}</span>
   204→                    </div>
   205→                  </th>
   206→                ))}
   207→              </tr>
   208→            </thead>
   209→            <tbody>
   210→              {uniqueDocIds.map(docId => {
   211→                const doc = documents.find(d => d.id === docId)
   212→                const docName = doc?.originalName ?? docId
   213→                const shortName = docName.length > 30 ? docName.slice(0, 28) + '…' : docName
   214→                return (
   215→                  <tr key={docId}>
   216→                    <td className="text-xs font-medium pr-3 py-1.5 max-w-[160px] truncate" title={docName}>{shortName}</td>
   217→                    {HEATMAP_COLUMNS.map(ct => {
   218→                      const status = matrix[docId]?.[ct]
   219→                      return (
   220→                        <td key={ct} className="text-center px-2 py-1.5">
   221→                          <TooltipProvider delayDuration={100}>
   222→                            <Tooltip>
   223→                              <TooltipTrigger asChild>
   224→                                <div className={`inline-flex items-center justify-center w-8 h-8 rounded-md transition-colors cursor-default ${cellColor(status)}`}>
   225→                                  {cellIcon(status)}
   226→                                </div>
   227→                              </TooltipTrigger>
   228→                              <TooltipContent side="top" className="text-xs">
   229→                                {status ? `${CHECK_TYPES[ct]?.shortLabel}: ${STATUS[status]?.label ?? status}` : 'Нет данных'}
   230→                              </TooltipContent>
   231→                            </Tooltip>
   232→                          </TooltipProvider>
   233→                        </td>
   234→                      )
   235→                    })}
   236→                  </tr>
   237→                )
   238→              })}
   239→            </tbody>
   240→          </table>
   241→        </div>
   242→        <div className="flex items-center gap-4 mt-3 text-xs text-muted-foreground">
   243→          <div className="flex items-center gap-1"><div className="w-4 h-4 rounded bg-emerald-500/80" /><span>Соответствует</span></div>
