   306→function radarPoint(axisIndex: number, value: number, totalAxes = 6, cx = 120, cy = 120, r = 88): { x: number; y: number } {
   307→  const angle = (axisIndex / totalAxes) * 2 * Math.PI - Math.PI / 2
   308→  return {
   309→    x: cx + (clamp(value, 0, 100) / 100) * r * Math.cos(angle),
   310→    y: cy + (clamp(value, 0, 100) / 100) * r * Math.sin(angle),
   311→  }
   312→}

   314→function radarPolygonPoints(values: number[]): string {
   315→  return values
   316→    .map((v, i) => {
   317→      const p = radarPoint(i, v)
   318→      return `${p.x.toFixed(2)},${p.y.toFixed(2)}`
   319→    })
   320→    .join(' ')
   321→}

   426→function PleaBargainingCalculator() {
   427→  const [articleCode, setArticleCode] = useState<string>(PLEA_ARTICLES[0]!.code)
   428→  const [mitState, setMitState] = useState<Record<string, boolean>>({})
   429→  const [aggState, setAggState] = useState<Record<string, boolean>>({})
   430→
   431→  const article = PLEA_ARTICLES.find(a => a.code === articleCode) ?? PLEA_ARTICLES[0]!
   432→
   433→  const calc = useMemo(() => {
   434→    const mitTotal = PLEA_MITIGATING.reduce((sum, f) => sum + (mitState[f.id] ? f.reduction : 0), 0)
   435→    const aggTotal = PLEA_AGGRAVATING.reduce((sum, f) => sum + (aggState[f.id] ? f.increase : 0), 0)
   436→    const raw = article.baseSentence - mitTotal + aggTotal
   437→    const sentence = Math.round(clamp(raw, article.punishmentMin, article.punishmentMax) * 10) / 10
   438→    const activeMitCount = PLEA_MITIGATING.filter(f => mitState[f.id]).length
   439→    const activeAggCount = PLEA_AGGRAVATING.filter(f => aggState[f.id]).length
   440→
   441→    // Probability heuristics
   442→    const sentenceRatio = sentence / Math.max(article.punishmentMax, 0.001)
   443→    let imprisonmentRaw = 45 + sentenceRatio * 50 - mitTotal * 6 + aggTotal * 6
   444→    let suspendedRaw = 25 + mitTotal * 8 - aggTotal * 5 - Math.max(0, sentence - 3) * 6
   445→    let fineRaw = 30 - sentenceRatio * 25 + (activeMitCount > activeAggCount ? 10 : 0)
   446→
   447→    // Normalize so the three primary outcomes sum to 100%
   448→    const totalRaw = imprisonmentRaw + suspendedRaw + fineRaw
   449→    if (totalRaw > 0) {
   450→      imprisonmentRaw = (imprisonmentRaw / totalRaw) * 100
   451→      suspendedRaw = (suspendedRaw / totalRaw) * 100
   452→      fineRaw = (fineRaw / totalRaw) * 100
   453→    }
   454→    const imprisonment = clamp(imprisonmentRaw, 1, 99)
   455→    const suspended = clamp(suspendedRaw, 1, 99)
   456→    const fine = clamp(fineRaw, 1, 99)
   457→
   458→    // Fine calculation — moves between min and max depending on mitigation
   459→    const fineRatio = clamp(1 - mitTotal * 0.15 + aggTotal * 0.1, 0, 1)
   460→    const fineAmount = Math.round((article.fineMin + (article.fineMax - article.fineMin) * fineRatio) / 1000) * 1000
   461→
   462→    // Category badge logic

   912→function scoreColor(s: number): string {
   913→  if (s >= 75) return '#991b1b'
   914→  if (s >= 50) return '#dc2626'
   915→  if (s >= 25) return '#f59e0b'
   916→  return '#059669'
   917→}
   918→function scoreTextClass(s: number): string {
   919→  if (s >= 75) return 'text-red-800'
   920→  if (s >= 50) return 'text-red-600'
   921→  if (s >= 25) return 'text-amber-500'
   922→  return 'text-emerald-600'
   923→}
   924→function factorProgressClass(s: number): string {
   925→  if (s >= 70) return '[&>div]:bg-red-700'
   926→  if (s >= 50) return '[&>div]:bg-amber-500'
   927→  return '[&>div]:bg-emerald-600'
   928→}
   929→function factorBadgeClass(s: number): string {
   930→  if (s >= 70) return 'bg-red-700 text-white'
   931→  if (s >= 50) return 'bg-amber-600 text-white'
   932→  return 'bg-emerald-700 text-white'
   933→}
   934→function matrixColor(likelihood: number, impact: number): string {
   935→  const sum = Math.ceil(likelihood / 20) + Math.ceil(impact / 20)
   936→  if (sum <= 3) return 'bg-emerald-700/60'
   937→  if (sum <= 5) return 'bg-amber-500/60'
   938→  if (sum <= 7) return 'bg-orange-500/70'
   939→  return 'bg-red-700/80'
   940→}

   942→function RiskRing({ score, level }: { score: number; level: string }) {
   943→  const r = 58, c = 2 * Math.PI * r, off = c - (score / 100) * c
   944→  const color = scoreColor(score)
   945→  return (
   946→    <div className="flex items-center justify-center relative">
   947→      <svg width="140" height="140" className="transform -rotate-90">
   948→        <circle cx="70" cy="70" r={r} stroke="#e5e7eb" strokeWidth="8" fill="none" className="dark:stroke-stone-700" />
   949→        <circle cx="70" cy="70" r={r} stroke={color} strokeWidth="8" fill="none" strokeDasharray={c} strokeDashoffset={off} strokeLinecap="round" className="transition-all duration-700" />
   950→      </svg>
   951→      <div className="absolute flex flex-col items-center">
   952→        <span className="text-2xl font-bold" style={{ color }}>{score}</span>
   953→        <span className="text-xs text-muted-foreground">{LEVEL_LABEL[level] ?? level}</span>
   954→      </div>
   955→    </div>
   956→  )
   957→}
