    80→function getRoleBadge(role: string | null) {
    81→  if (!role) return <Badge variant="outline">—</Badge>
    82→  const color = ROLE_COLORS[role] || 'bg-stone-500 text-white'
    83→  return <Badge className={color}>{role}</Badge>
    84→}
    85→
    86→function getGuiltBadge(guiltLevel: string | undefined) {
    87→  if (!guiltLevel) return <Badge variant="outline">Не оценено</Badge>
    88→  const info = GUILT_LEVELS[guiltLevel]
    89→  if (!info) return <Badge variant="outline">{guiltLevel}</Badge>
    90→  return (
    91→    <Badge style={{ backgroundColor: info.color, color: 'white' }}>
    92→      Виновность: {info.label}
    93→    </Badge>
    94→  )
    95→}
    96→
    97→function getGuiltProgressBar(guiltLevel: string | undefined) {
    98→  if (!guiltLevel) return <Progress value={0} className="h-2" />
    99→  const info = GUILT_LEVELS[guiltLevel]
   100→  if (!info) return <Progress value={0} className="h-2" />
   101→  return (
   102→    <div className="relative">
   103→      <Progress value={info.numeric} className="h-2.5" />
   104→      <div
   105→        className="absolute top-0 left-0 h-2.5 rounded-full transition-all"
   106→        style={{
   107→          width: `${info.numeric}%`,
   108→          backgroundColor: info.color,
   109→          opacity: 0.3,
   110→        }}
   111→      />
   112→    </div>
   113→  )
   114→}
   115→
   116→function getEvidenceStrengthBar(strength: string | undefined) {
   117→  const value = strength === 'strong' ? 85 : strength === 'moderate' ? 50 : strength === 'weak' ? 20 : 0
   118→  const color = EVIDENCE_COLORS[strength || 'none'] || '#525252'
   119→  return (
   120→    <div className="flex items-center gap-2">
   121→      <div className="flex-1 h-2 bg-stone-800 rounded-full overflow-hidden">
   122→        <motion.div
   123→          initial={{ width: 0 }}
   124→          animate={{ width: `${value}%` }}
   125→          transition={{ duration: 0.6 }}
   126→          className="h-full rounded-full"
   127→          style={{ backgroundColor: color }}
   128→        />
   129→      </div>
   130→      <span className="text-xs font-medium" style={{ color }}>{EVIDENCE_LABELS[strength || 'none']}</span>
   131→    </div>
   132→  )
   133→}
