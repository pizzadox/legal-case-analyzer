    81→function exportAsCSV(data: { documents: any[], persons: any[], episodes: any[], dashboard: any }, caseNumber: string, selItems: Set<string>): void {
    82→  const sections: string[] = []
    83→  const hasDocs = DOCUMENT_IDS.some(id => selItems.has(id))
    84→  const hasPersons = PERSON_IDS.some(id => selItems.has(id))
    85→  const hasEpisodes = PERSON_IDS.some(id => selItems.has(id))
    86→
    87→  // Documents section
    88→  if (hasDocs && data.documents.length > 0) {
    89→    sections.push('=== Документы ===')
    90→    const headers = ['Название', 'Тип', 'Статус', 'Дата документа', 'Размер (КБ)', 'Дата загрузки', 'Описание']
    91→    const rows = data.documents.map(d => [
    92→      d.originalName, d.documentType ?? '', d.processingStatus,
    93→      d.documentDate ?? '', Math.round(d.fileSize / 1024),
    94→      new Date(d.uploadedAt).toLocaleDateString('ru-RU'), (d.summary ?? '').substring(0, 200)
    95→    ].join(';'))
    96→    sections.push(headers.join(';'))
    97→    sections.push(...rows)
    98→  }
    99→
   100→  // Persons section
   101→  if (hasPersons && data.persons.length > 0) {
   102→    sections.push('')
   103→    sections.push('=== Участники ===')
   104→    const headers = ['ФИО', 'Роль', 'Статус', 'Описание']
   105→    const rows = data.persons.map(p => [
   106→      p.fullName, p.role ?? '', p.status ?? '', (p.description ?? '').substring(0, 200)
   107→    ].join(';'))
   108→    sections.push(headers.join(';'))
   109→    sections.push(...rows)
   110→  }
   111→
   112→  // Episodes section
   113→  if (hasEpisodes && data.episodes.length > 0) {
   114→    sections.push('')
   115→    sections.push('=== Эпизоды ===')
   116→    const headers = ['Название', 'Описание', 'Дата', 'Тяжесть', 'Статус']
   117→    const rows = data.episodes.map(e => [
   118→      e.title, (e.description ?? '').substring(0, 200), e.date ?? '',
   119→      e.severity ?? '', e.status ?? ''
   120→    ].join(';'))
   121→    sections.push(headers.join(';'))
   122→    sections.push(...rows)
   123→  }
   124→
   125→  const csv = sections.join('\n')
   126→  const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8' })
   127→  const url = URL.createObjectURL(blob)
   128→  const a = document.createElement('a')
   129→  a.href = url
   130→  a.download = `delo-${caseNumber}-export-${new Date().toISOString().slice(0,10)}.csv`
   131→  a.click()
   132→  URL.revokeObjectURL(url)
   133→  toast.success('CSV экспорт выполнен')
   134→}

   246→export function CaseExportCenter({ caseId }: { caseId: string }) {
   247→  const [selFmt,setSelFmt]=useState<ExportFormat>('pdf')
   248→  const [selItems,setSelItems]=useState<Set<string>>(()=>new Set(ALL_IDS.slice(0,12)))
   249→  const [opt,setOpt]=useState<ExportOptions>({language:'ru',includeAI:true,includeCharts:true,pageFormat:'a4',orientation:'portrait',watermark:true,encrypt:false})
   250→  const [isGen,setIsGen]=useState(false)
   251→  const [prog,setProg]=useState(0)
   252→  const [recent,setRecent]=useState<RecentExport[]>([])
   253→  const [previewOpen,setPreviewOpen]=useState(false)
   254→  const [resultOpen,setResultOpen]=useState(false)
   255→  const [genName,setGenName]=useState('')
   256→  const [genSize,setGenSize]=useState(0)
   257→
   258→  // Fetch real case data for exports
   259→  const { data: documents = [] } = useQuery({
   260→    queryKey: ['documents', caseId],
   261→    queryFn: () => caseApi.getDocuments(caseId),
   262→    enabled: !!caseId,
   263→    refetchInterval: 10000,
   264→  })
   265→  const { data: persons = [] } = useQuery({
   266→    queryKey: ['persons', caseId],
   267→    queryFn: () => caseApi.getPersons(caseId),
   268→    enabled: !!caseId,
   269→    refetchInterval: 10000,
   270→  })
   271→  const { data: episodes = [] } = useQuery({
   272→    queryKey: ['episodes', caseId],
   273→    queryFn: () => caseApi.getEpisodes(caseId),
   274→    enabled: !!caseId,
   275→    refetchInterval: 10000,
   276→  })
   277→  const { data: dashboard } = useQuery({
   278→    queryKey: ['dashboard', caseId],
   279→    queryFn: () => caseApi.getDashboardStats(caseId),
   280→    enabled: !!caseId,
   281→    refetchInterval: 10000,
   282→  })
   283→
   284→  const exportData = { documents, persons, episodes, dashboard }
   285→  const caseNumber = dashboard?.summary?.totalDocuments ?? caseId
   286→
   287→  const estSize=useMemo(()=>estimate(selItems,selFmt,opt),[selItems,selFmt,opt])
   288→  const toggle=useCallback((id:string)=>setSelItems(p=>{const n=new Set(p);if(n.has(id))n.delete(id);else n.add(id);return n}),[])
   289→  const selectAll=useCallback(()=>{setSelItems(new Set(ALL_IDS));toast.success(`Выбрано: ${ALL_IDS.length}`)},[])
   290→  const clearAll=useCallback(()=>{setSelItems(new Set());toast.info('Выбор очищен')},[])
   291→  const updOpt=useCallback(<K extends keyof ExportOptions>(k:K,v:ExportOptions[K])=>setOpt(p=>({...p,[k]:v})),[])
   292→
   293→  const generate=useCallback(async()=>{
   294→    if(selItems.size===0){toast.error('Выберите элементы');return}
   295→    setIsGen(true);setProg(0)
   296→    const name=`delo-${caseNumber}-${new Date().toISOString().slice(0,10)}.${F_CFG[selFmt].ext}`
   297→    const size=estSize
   298→    setGenName(name);setGenSize(size)
   299→
   300→    // Simulate progress while actually generating the export
   301→    for(const p of [10,30,60,80,95,100]){setProg(p);await new Promise(r=>setTimeout(r,200))}
   302→
   303→    // Actually generate the export based on format
   304→    const data = { documents, persons, episodes, dashboard }
   305→    switch(selFmt) {
   306→      case 'csv': exportAsCSV(data, caseNumber, selItems); break
   307→      case 'json': exportAsJSON(data, caseNumber, selItems); break
   308→      case 'html': exportAsHTML(data, caseNumber, selItems); break
   309→      case 'pdf': exportAsPDF(data, caseNumber, selItems); break
   310→    }
