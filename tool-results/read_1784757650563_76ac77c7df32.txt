  1405→
  1406→export function CaseViolations() {
  1407→  const [search, setSearch] = useState('')
  1408→  const [severityFilter, setSeverityFilter] = useState<'all' | Severity>('all')
  1409→  const [articleFilter, setArticleFilter] = useState<'all' | string>('all')
  1410→  const [statusFilter, setStatusFilter] = useState<'all' | ViolationStatus>('all')
  1411→  const [sortBy, setSortBy] = useState<SortKey>('date-desc')
  1412→  const [selectedViolation, setSelectedViolation] = useState<Violation | null>(null)
  1413→  const [sheetOpen, setSheetOpen] = useState(false)
  1414→  // По умолчанию в ходатайстве — все нарушения со статусом excludable
  1415→  const [petitionItems, setPetitionItems] = useState<Set<string>>(
  1416→    () => new Set(VIOLATIONS.filter((v) => v.status === 'excludable').map((v) => v.id))
  1417→  )
  1418→
  1419→  const handleSelect = useCallback((v: Violation) => {
  1420→    setSelectedViolation(v)
  1421→    setSheetOpen(true)
  1422→  }, [])
  1423→
  1424→  const handleTogglePetition = useCallback((id: string) => {
  1425→    setPetitionItems((prev) => {
  1426→      const next = new Set(prev)
  1427→      if (next.has(id)) {
  1428→        next.delete(id)
  1429→      } else {
  1430→        next.add(id)
  1431→      }
  1432→      return next
  1433→    })
  1434→  }, [])
  1435→
  1436→  const handleAddToPetition = useCallback(
  1437→    (id: string) => {
  1438→      const wasIn = petitionItems.has(id)
  1439→      handleTogglePetition(id)
  1440→      if (wasIn) {
  1441→        toast.info('Нарушение удалено из ходатайства')
  1442→      } else {
  1443→        const v = VIOLATIONS.find((x) => x.id === id)
  1444→        toast.success(`Нарушение добавлено в ходатайство`, {
  1445→          description: v ? v.type : undefined,
  1446→        })
  1447→      }
  1448→    },
  1449→    [petitionItems, handleTogglePetition]
  1450→  )
  1451→
  1452→  const handleResetFilters = useCallback(() => {
  1453→    setSearch('')
  1454→    setSeverityFilter('all')
  1455→    setArticleFilter('all')
  1456→    setStatusFilter('all')
  1457→    setSortBy('date-desc')
  1458→  }, [])
  1459→
  1460→  const handleGenerate = useCallback(() => {
  1461→    toast.success('Ходатайство сформировано. Открыть редактор?', {
  1462→      description: `Включено нарушений: ${petitionItems.size || VIOLATIONS.filter((v) => v.status === 'excludable').length}`,
  1463→      action: {
  1464→        label: 'Открыть',
  1465→        onClick: () => toast.info('Открытие редактора ходатайств...'),
  1466→      },
  1467→    })
  1468→  }, [petitionItems])
  1469→
  1470→  const handleDownloadTemplate = useCallback(() => {
  1471→    try {
  1472→      const template = `ХОДАТАЙСТВО\nоб исключении доказательств\n\nВ производстве следователя (дознавателя) находится уголовное дело № 2024-00145\nпо обвинению Колесниченко Д.А. по ст. 159 ч.3, ст. 160 ч.2 УК РФ.\n\nЗащита заявляет ходатайство об исключении следующих доказательств\nкак недопустимых в порядке ст. 75, 88 УПК РФ:\n\n${VIOLATIONS.filter((v) => v.status === 'excludable').map((v, i) => `${i + 1}. ${v.type} (${v.article}) — ${v.evidence}`).join('\n')}\n\nОснование: ч. 1 ст. 75 УПК РФ, Постановление Пленума ВС РФ №1 от 31.10.1995.\n\nЗащитник ___________________ /___________________/\n\nДата: «___» ___________ 2024 г.`
  1473→      const blob = new Blob([template], { type: 'text/plain;charset=utf-8' })
  1474→      const url = URL.createObjectURL(blob)
  1475→      const a = document.createElement('a')
  1476→      a.href = url
  1477→      a.download = 'hodataystvo-isobshenie-dokazatelstv.txt'
  1478→      a.click()
  1479→      URL.revokeObjectURL(url)
  1480→      toast.success('Шаблон ходатайства сохранён в загрузки')
  1481→    } catch {
  1482→      toast.error('Не удалось сохранить шаблон')
  1483→    }
  1484→  }, [])
  1485→
  1486→  const handleClearPetition = useCallback(() => {
  1487→    setPetitionItems(new Set())
  1488→    toast.info('Выбор нарушений очищен')
  1489→  }, [])
  1490→
  1491→  // Активные фильтры
  1492→  const activeFilterCount = useMemo(() => {
  1493→    let n = 0
  1494→    if (search.trim()) n++
  1495→    if (severityFilter !== 'all') n++
  1496→    if (articleFilter !== 'all') n++
  1497→    if (statusFilter !== 'all') n++
  1498→    if (sortBy !== 'date-desc') n++
  1499→    return n
  1500→  }, [search, severityFilter, articleFilter, statusFilter, sortBy])
  1501→
  1502→  // Фильтрация и сортировка
  1503→  const filteredViolations = useMemo(() => {
  1504→    const sevRank: Record<Severity, number> = { critical: 3, serious: 2, moderate: 1 }
  1505→    let arr = VIOLATIONS.filter((v) => {
  1506→      if (search.trim()) {
  1507→        const q = search.toLowerCase().trim()
  1508→        const matches =
  1509→          v.type.toLowerCase().includes(q) ||
  1510→          v.description.toLowerCase().includes(q) ||
  1511→          v.article.toLowerCase().includes(q) ||
  1512→          v.evidence.toLowerCase().includes(q)
  1513→        if (!matches) return false
  1514→      }
  1515→      if (severityFilter !== 'all' && v.severity !== severityFilter) return false
  1516→      if (articleFilter !== 'all' && v.articleKey !== articleFilter) return false
  1517→      if (statusFilter !== 'all' && v.status !== statusFilter) return false
  1518→      return true
  1519→    })
  1520→
  1521→    arr = arr.sort((a, b) => {
  1522→      switch (sortBy) {
  1523→        case 'date-desc':
  1524→          return new Date(b.date).getTime() - new Date(a.date).getTime()
  1525→        case 'date-asc':
  1526→          return new Date(a.date).getTime() - new Date(b.date).getTime()
  1527→        case 'severity':
  1528→          return sevRank[b.severity] - sevRank[a.severity]
  1529→        case 'article':
  1530→          return a.articleKey.localeCompare(b.articleKey)
  1531→        default:
  1532→          return 0
  1533→      }
  1534→    })
  1535→
  1536→    return arr
  1537→  }, [search, severityFilter, articleFilter, statusFilter, sortBy])
  1538→
  1539→  return (
  1540→    <div className="space-y-6 pb-8">
  1541→      {/* Секция 1: Шапка-баннер */}
  1542→      <HeaderBanner />
  1543→
  1544→      {/* Секция 2: Диаграммы распределения + Потенциал защиты */}
  1545→      <ChartsSection />
  1546→
  1547→      {/* Секция 3: Панель фильтров */}
  1548→      <FiltersBar
  1549→        search={search}
  1550→        setSearch={setSearch}
  1551→        severityFilter={severityFilter}
  1552→        setSeverityFilter={setSeverityFilter}
  1553→        articleFilter={articleFilter}
  1554→        setArticleFilter={setArticleFilter}
  1555→        statusFilter={statusFilter}
  1556→        setStatusFilter={setStatusFilter}
  1557→        sortBy={sortBy}
  1558→        setSortBy={setSortBy}
  1559→        onReset={handleResetFilters}
  1560→        activeCount={activeFilterCount}
  1561→        totalFiltered={filteredViolations.length}
  1562→      />
  1563→
  1564→      {/* Секция 4: Таблица нарушений */}
  1565→      <ViolationsTable
  1566→        violations={filteredViolations}
  1567→        onSelect={handleSelect}
  1568→        petitionItems={petitionItems}
  1569→        togglePetition={handleTogglePetition}
  1570→      />
  1571→
  1572→      {/* Секция 6: Конструктор ходатайства (под таблицей) */}
  1573→      <PetitionBuilder
  1574→        petitionItems={petitionItems}
  1575→        violations={VIOLATIONS}
  1576→        onGenerate={handleGenerate}
  1577→        onDownloadTemplate={handleDownloadTemplate}
  1578→        onClear={handleClearPetition}
  1579→      />
  1580→
  1581→      {/* Секция 5: Sheet с деталями нарушения (открывается по клику) */}
  1582→      <DetailSheet
  1583→        violation={selectedViolation}
  1584→        open={sheetOpen}
  1585→        onOpenChange={setSheetOpen}
  1586→        onAddToPetition={handleAddToPetition}
  1587→        inPetition={selectedViolation ? petitionItems.has(selectedViolation.id) : false}
  1588→      />
  1589→    </div>
  1590→  )
  1591→}
