   511→function monthToX(month: number): number {
   512→  return LEFT_PADDING + month * MONTH_WIDTH
   513→}
   514→
   515→function formatMonthRange(start: number, duration: number): string {
   516→  const end = start + duration - 1
   517→  if (duration === 1) {
   518→    return MONTH_LABELS[start]
   519→  }
   520→  return `${MONTH_LABELS[start]} — ${MONTH_LABELS[end]}`
   521→}
   522→
   523→function getMoveGradient(side: MoveSide): { id: string; from: string; to: string; stroke: string } {
   524→  if (side === 'prosecution') {
   525→    return { id: 'grad-prosecution', from: '#b91c1c', to: '#991b1b', stroke: '#7f1d1d' }
   526→  }
   527→  return { id: 'grad-defense', from: '#047857', to: '#065f46', stroke: '#064e3b' }
   528→}
   529→
   530→function isFutureMove(move: BattleMove): boolean {
   531→  return move.startMonth >= TODAY_MONTH_INDEX
   532→}
   533→
   534→function visibleMoves(filter: FilterKey): { prosecution: BattleMove[]; defense: BattleMove[] } {
   535→  const all = [...PROSECUTION_MOVES, ...DEFENSE_MOVES]
   536→  const matches = (m: BattleMove) => {
   537→    switch (filter) {
   538→      case 'all':
   539→        return true
   540→      case 'prosecution':
   541→        return m.side === 'prosecution'
   542→      case 'defense':
   543→        return m.side === 'defense'
   544→      case 'completed':
   545→        return m.status === 'completed'
   546→      case 'planned':
   547→        return m.status === 'planned' || m.status === 'active'
   548→    }
   549→  }
   550→  const filtered = all.filter(matches)
   551→  return {
   552→    prosecution: filtered.filter((m) => m.side === 'prosecution'),
   553→    defense: filtered.filter((m) => m.side === 'defense'),
   554→  }
   555→}

   658→function BattleMoveBar({ move, y, isHovered, onHover, onSelect }: BattleMoveBarProps) {

   660→  const width = move.durationMonths * MONTH_WIDTH - 2

   664→  const opacity = isHovered ? 1 : baseOpacity
   665→  const statusCfg = STATUS_CONFIG[move.status]
   666→  const StatusIcon = statusCfg.icon
   667→
   668→  // Текст внутри бара — обрезаем, если не помещается
   669→  const maxChars = Math.floor(width / 6.5)
   670→  const displayTitle =
   671→    move.title.length > maxChars ? move.title.slice(0, Math.max(3, maxChars - 1)) + '…' : move.title
   672→  const showText = width > 50
   673→
   674→  return (
   675→    <g
   676→      onMouseEnter={() => onHover(move.id)}

   678→      onClick={() => onSelect(move)}

   680→      style={{ transition: 'opacity 200ms ease' }}

   682→    >
   683→      {/* Невидимая широкая дорожка для наведения */}
