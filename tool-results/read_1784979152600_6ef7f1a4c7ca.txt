    31→// Format ISO timestamp in Russian style: dd.MM.yyyy HH:mm
    32→function formatRussianDateTime(iso: string): string {
    33→  try {
    34→    const d = new Date(iso)
    35→    if (isNaN(d.getTime())) return iso
    36→    const dd = String(d.getDate()).padStart(2, '0')
    37→    const mm = String(d.getMonth() + 1).padStart(2, '0')
    38→    const yyyy = d.getFullYear()
    39→    const hh = String(d.getHours()).padStart(2, '0')
    40→    const min = String(d.getMinutes()).padStart(2, '0')
    41→    return `${dd}.${mm}.${yyyy} ${hh}:${min}`
    42→  } catch {
    43→    return iso
    44→  }
    45→}

    50→function loadAllAnnotations(): Record<string, Annotation[]> {
    51→  if (typeof window === 'undefined') return {}
    52→  const loaded: Record<string, Annotation[]> = {}
    53→  try {
    54→    for (let i = 0; i < window.localStorage.length; i++) {
    55→      const key = window.localStorage.key(i)
    56→      if (key && key.startsWith(ANNOTATIONS_PREFIX)) {
    57→        const docId = key.replace(ANNOTATIONS_PREFIX, '')
    58→        const stored = window.localStorage.getItem(key)
    59→        if (stored) {
    60→          try {
    61→            const parsed = JSON.parse(stored)
    62→            if (Array.isArray(parsed)) loaded[docId] = parsed
    63→          } catch {
    64→            /* ignore malformed entries */
    65→          }
    66→        }
    67→      }
    68→    }
    69→  } catch {
    70→    /* localStorage unavailable */
    71→  }
    72→  return loaded
    73→}

    84→function fmtSize(b: number) {
    85→  if (b < 1024) return b + ' Б'
    86→  if (b < 1048576) return (b / 1024).toFixed(1) + ' КБ'
    87→  return (b / 1048576).toFixed(1) + ' МБ'
    88→}

   129→function IntegrityRing({ score }: { score: number }) {
   130→  const radius = 18, circ = 2 * Math.PI * radius, offset = circ - (score / 100) * circ
   131→  const color = score >= 75 ? '#059669' : score >= 50 ? '#d97706' : '#dc2626'
   132→  return (
   133→    <div className="relative w-12 h-12 shrink-0">
   134→      <svg width="48" height="48" className="transform -rotate-90"><circle cx="24" cy="24" r={radius} stroke="#e5e7eb" strokeWidth="4" fill="none" className="dark:stroke-stone-700" /><circle cx="24" cy="24" r={radius} stroke={color} strokeWidth="4" fill="none" strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" className="transition-all duration-700" /></svg>
   135→      <div className="absolute inset-0 flex items-center justify-center text-xs font-bold" style={{ color }}>{score}</div>
   136→    </div>

   188→// Export CSV helper
   189→function exportDocumentsCSV(docs: DocumentData[]) {
   190→  const rows = ['Name,Type,Date,Status,Size,Summary']
   191→  docs.forEach(d => {
   192→    rows.push(`"${d.originalName}",${d.documentType ?? ''},${d.documentDate ?? ''},${d.processingStatus},${fmtSize(d.fileSize)},${d.summary ?? ''}`)
   193→  })
   194→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
   195→  const url = URL.createObjectURL(blob)
   196→  const a = document.createElement('a')
   197→  a.href = url
   198→  a.download = 'documents.csv'
   199→  a.click()
   200→  URL.revokeObjectURL(url)
   201→  toast.success('CSV экспорт выполнен')
   202→}
