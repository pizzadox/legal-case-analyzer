    43→// Health score ring component (SVG-based)
    44→function HealthScoreRing({ score }: { score: number }) {
    45→  const radius = 58
    46→  const circumference = 2 * Math.PI * radius
    47→  const offset = circumference - (score / 100) * circumference
    48→  const color = score >= 70 ? '#059669' : score >= 50 ? '#d97706' : '#dc2626'
    49→  return (
    50→    <div className="flex items-center justify-center">
    51→      <svg width="140" height="140" className="transform -rotate-90">
    52→        <circle cx="70" cy="70" r={radius} stroke="#e5e7eb" strokeWidth="8" fill="none" className="dark:stroke-stone-700" />
    53→        <circle cx="70" cy="70" r={radius} stroke={color} strokeWidth="8" fill="none" strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round" className="transition-all duration-700" />
    54→      </svg>
    55→      <div className="absolute flex flex-col items-center"><span className="text-2xl font-bold" style={{ color }}>{score}</span><span className="text-xs text-muted-foreground">из 100</span></div>
