    81→// Custom Recharts tooltip - declared outside component
    82→function CustomTooltip({ active, payload, label }: any) {
    83→  if (active && payload && payload.length) {
    84→    return (
    85→      <div className="p-2 bg-stone-800 rounded-lg border border-stone-700 text-xs">
    86→        <p className="font-medium text-stone-200 mb-1">{label}</p>
    87→        {payload.map((entry: any, i: number) => (
    88→          <p key={i} className="text-stone-400">
    89→            <span className="font-medium" style={{ color: entry.color }}>{entry.name}</span>: {entry.value}%
    90→          </p>
    91→        ))}
    92→      </div>
    93→    )
    94→  }
    95→  return null
    96→}
