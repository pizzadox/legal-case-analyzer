   407→// ===== Helpers =====
   408→function formatFileSize(bytes: number): string {
   409→  if (bytes < 1024) return `${bytes} Б`
   410→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(0)} КБ`
   411→  return `${(bytes / (1024 * 1024)).toFixed(1)} МБ`
   412→}
   413→
   414→function estimateSize(
   415→  selectedItems: Set<string>,
   416→  format: ExportFormat,
   417→  options: ExportOptions,
   418→): number {
   419→  const perItem = FORMAT_CONFIG[format].perItemBytes
   420→  let bytes = selectedItems.size * perItem
   421→  // AI analysis adds ~20% overhead per item
   422→  if (options.includeAI) bytes = Math.round(bytes * 1.2)
   423→  // Charts as images adds ~15% overhead (only PDF/HTML)
   424→  if (options.includeCharts && (format === 'pdf' || format === 'html')) {
   425→    bytes = Math.round(bytes * 1.15)
   426→  }
   427→  // Both languages double the size
   428→  if (options.language === 'both') bytes *= 2
   429→  // Encryption adds small constant overhead
   430→  if (options.encrypt) bytes += 4096
   431→  return bytes
   432→}
   433→
   434→function buildFileName(format: ExportFormat, options: ExportOptions): string {
   435→  const date = new Date()
   436→  const ymd = `${date.getFullYear()}${String(date.getMonth() + 1).padStart(2, '0')}${String(date.getDate()).padStart(2, '0')}`
   437→  const langSuffix = options.language === 'both' ? '-ru-en' : options.language === 'en' ? '-en' : '-ru'
   438→  return `delo-${CASE_NUMBER}-${ymd}${langSuffix}.${FORMAT_CONFIG[format].extension}`
   439→}
   440→
   441→function formatDateRu(iso: string): string {
   442→  try {
   443→    const d = new Date(iso)
   444→    return d.toLocaleString('ru-RU', {
   445→      day: '2-digit',
   446→      month: '2-digit',
   447→      year: 'numeric',
   448→      hour: '2-digit',
   449→      minute: '2-digit',
   450→    })
   451→  } catch {
   452→    return iso
   453→  }
   454→}
