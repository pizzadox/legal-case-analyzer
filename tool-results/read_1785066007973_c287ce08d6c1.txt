    77→    77→function exportAsCSV(data: { documents: any[], persons: any[], episodes: any[], dashboard: any }, caseNumber: string): void {
    78→    78→  const sections: string[] = []
    79→    79→
    80→    80→  // Documents section
    81→    81→  if (data.documents.length > 0) {
    82→    82→    sections.push('=== Документы ===')
    83→    83→    const headers = ['Название', 'Тип', 'Статус', 'Дата документа', 'Размер (КБ)', 'Дата загрузки', 'Описание']
    84→    84→    const rows = data.documents.map(d => [
    85→    85→      d.originalName, d.documentType ?? '', d.processingStatus,
    86→    86→      d.documentDate ?? '', Math.round(d.fileSize / 1024),
    87→    87→      new Date(d.uploadedAt).toLocaleDateString('ru-RU'), (d.summary ?? '').substring(0, 200)
    88→    88→    ].join(';'))
    89→    89→    sections.push(headers.join(';'))
    90→    90→    sections.push(...rows)
    91→    91→  }
    92→    92→
    93→    93→  // Persons section
    94→    94→  if (data.persons.length > 0) {
    95→    95→    sections.push('')
    96→    96→    sections.push('=== Участники ===')
    97→    97→    const headers = ['ФИО', 'Роль', 'Статус', 'Описание']
    98→    98→    const rows = data.persons.map(p => [
    99→    99→      p.fullName, p.role ?? '', p.status ?? '', (p.description ?? '').substring(0, 200)
   100→   100→    ].join(';'))
   101→   101→    sections.push(headers.join(';'))
   102→   102→    sections.push(...rows)
   103→   103→  }
   104→   104→
   105→   105→  // Episodes section
   106→   106→  if (data.episodes.length > 0) {
   107→   107→    sections.push('')
   108→   108→    sections.push('=== Эпизоды ===')
   109→   109→    const headers = ['Название', 'Описание', 'Дата', 'Тяжесть', 'Статус']
   110→   110→    const rows = data.episodes.map(e => [
   111→   111→      e.title, (e.description ?? '').substring(0, 200), e.date ?? '',
   112→   112→      e.severity ?? '', e.status ?? ''
   113→   113→    ].join(';'))
   114→   114→    sections.push(headers.join(';'))
   115→   115→    sections.push(...rows)
   116→   116→  }
   117→   117→
   118→   118→  const csv = sections.join('\n')
   119→   119→  const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8' })
   120→   120→  const url = URL.createObjectURL(blob)
   121→   121→  const a = document.createElement('a')
   122→   122→  a.href = url
   123→   123→  a.download = `delo-${caseNumber}-export-${new Date().toISOString().slice(0,10)}.csv`
   124→   124→  a.click()
   125→   125→  URL.revokeObjectURL(url)
   126→   126→  toast.success('CSV экспорт выполнен')
   127→   127→}
